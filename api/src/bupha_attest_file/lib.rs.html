1//! File adapter of `Attest` — writes an in-toto/SLSA-shaped predicate
2//! JSON per finished run to `<dir>/<repo>-<sha12>.json`. Unsigned v1:
3//! provenance is *recorded*, signing (`bupha-attest-sigstore`) is the
4//! follow-up adapter.
5
6use bupha_attest::{Attest, RunAttestation};
7use std::io::{self, Write};
8use std::path::PathBuf;
9
10/// Writes predicate files under `dir` (created on demand).
11pub struct FileAttest {
12    /// Destination directory for `*.json` predicates.
13    pub dir: PathBuf,
14}
15
16impl Attest for FileAttest {
17    fn attest(&self, a: &RunAttestation) -> io::Result<PathBuf> {
18        std::fs::create_dir_all(&self.dir)?;
19        let clean = |s: &str| s.replace(['/', '"', '\\'], "-");
20        let sha12 = &a.sha[..12.min(a.sha.len())];
21        let path = self.dir.join(format!("{}-{}.json", clean(&a.repo), sha12));
22        let json = format!(
23            r#"{{"_type":"https://in-toto.io/Statement/v1",
24"subject":[{{"name":"{}","digest":{{"gitCommit":"{}"}}}}],
25"predicateType":"https://bupha.dev/attestation/run/v1",
26"predicate":{{"runner":"{}","status":"{}","wall_secs":{}}}}}
27"#,
28            clean(&a.repo),
29            clean(&a.sha),
30            clean(&a.runner),
31            clean(&a.status),
32            a.wall_secs
33        );
34        let mut f = std::fs::File::create(&path)?;
35        f.write_all(json.as_bytes())?;
36        Ok(path)
37    }
38}

40#[cfg(test)]
41mod tests {
42    use super::*;
43
44    #[test]
45    fn predicate_written() {
46        let dir = std::env::temp_dir().join(format!("att-{}", std::process::id()));
47        let a = FileAttest { dir: dir.clone() };
48        let p = a
49            .attest(&RunAttestation {
50                sha: "0123456789abcdef".into(),
51                repo: "org/repo".into(),
52                runner: "shell".into(),
53                status: "success".into(),
54                wall_secs: 42,
55            })
56            .unwrap();
57        let s = std::fs::read_to_string(&p).unwrap();
58        assert!(s.contains("in-toto.io/Statement"));
59        assert_eq!(
60            p.file_name().unwrap().to_string_lossy(),
61            "org-repo-0123456789ab.json"
62        );
63        let _ = std::fs::remove_dir_all(&dir);
64    }
