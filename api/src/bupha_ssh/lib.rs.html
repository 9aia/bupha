1//! SSH impl of `Runner` — ships the detached worktree to a remote
2//! host and runs the shell command there. Sync and run are separate
3//! stages (rsync first, ssh second) so an infra failure — unreachable
4//! host, rsync error — reports "ci infra: …" instead of a pipeline
5//! "ci failed". The returned handle is a local process pair, so the
6//! worker's wait/cancel logic is unchanged — the remote-ness lives
7//! inside the impl.
8//!
9//! Config: `[runner] kind = "ssh", remote = "u@h", remote_dir =
10//! "~/bupha", cmd = "./ci.sh"` — auth via the user's ssh config
11//! (agent/keys), nothing bupha-specific. Each run lands in its own
12//! remote dir `<remote_dir>/<key>-<sha12>` — two runners sharing a
13//! remote never `rsync --delete` each other's tree.
14//!
15//! The run script — including the resolved `token_env` value — is
16//! fed to the remote `bash -s` over ssh's stdin, so the secret never
17//! appears in a remote (or local) `ps` listing.
18//!
19//! Known caveats:
20//! - `kill()` terminates the local ssh and best-effort `pkill`s the
21//!   remote run shell (tagged argv0) + removes the run dir, but a
22//!   remote process group that outlives its shell can still linger.
23//! - The rsynced worktree's `.git` is a *pointer file* into the main
24//!   clone's gitdir — meaningless on the remote, so `git` commands
25//!   inside the remote run fail. Pipelines that need VCS metadata
26//!   must clone or init their own repo remotely.
27//! - The remote shell cleans its run dir after the command; a hard
28//!   kill mid-run leaves cleanup to the `pkill`/`rm -rf` fallback.
29

59/// Which stage of the remote pipeline is currently in flight.
60/// Splitting them is what lets `failure_desc` tell a sync/transport
61/// failure from a run failure.
62enum Stage {
63    /// rsync worktree -> remote run dir.
64    Sync,
65    /// ssh remote `bash -s` (script already fed to stdin).
66    Run,
67}
68
69/// RunHandle over the two-stage remote run. Holds whichever child is
70/// in flight plus everything needed to arm the run stage once the
71/// sync lands.
72struct SshHandle {
73    /// Current stage's child — the kill() target.
74    child: Child,
75    stage: Stage,
76    /// "user@host".
77    remote: String,
78    /// Quoted/`~`-resolved per-run dir for remote shell lines —
79    /// bash cwd, cleanup target.
80    rdir_shell: String,
81    /// Remote bash argv0 marker — `pkill -f` target on cancel.
82    tag: String,
83    /// Payload fed to the remote `bash -s` over ssh's stdin.
84    script: String,
85    /// Sync stage exited non-zero — infra, not the pipeline.
86    sync_failed: bool,
87    /// Terminal status once either stage completes.
88    done: Option<ExitStatus>,
89}
90
91/// A remote-dir/tag/job-name component — `[A-Za-z0-9._-]` only,
92/// everything else maps to `-`. These values land inside remote
93/// shell command strings (pkill patterns, rm -rf targets) where a
94/// stray quote or metachar would inject.
95fn safe_component(s: &str) -> String {
96    s.chars()
97        .map(|c| {
98            if c.is_ascii_alphanumeric() || matches!(c, '_' | '.' | '-') {
99                c
100            } else {
101                '-'
102            }
103        })
104        .collect()
105}
106
107/// An ssh destination ("user@host", "[user@]host:port" forms). Must
108/// never start with `-` (argv option injection on ssh AND rsync) and
109/// never carry characters a shell or ssh config could re-interpret.
110fn is_valid_remote(s: &str) -> bool {
111    !s.is_empty()
112        && !s.starts_with('-')
113        && s.bytes().all(|b| {
114            b.is_ascii_alphanumeric() || matches!(b, b'@' | b'.' | b'-' | b'_' | b':' | b'[' | b']')
115        })
116}
117
118/// `'…'` with embedded quotes escaped — for the remote script's
119/// single-quoted literals.
120fn sh_escape(s: &str) -> String {
121    format!("'{}'", s.replace('\'', "'\\''"))
122}
123
124/// `"…"` for remote shell lines — escapes what double quotes honor.
125fn dq_escape(s: &str) -> String {
126    format!(
127        "\"{}\"",
128        s.replace('\\', "\\\\")
129            .replace('"', "\\\"")
130            .replace('$', "\\$")
131            .replace('`', "\\`")
132    )
133}
134
135/// `rdir` as a remote-shell-safe path — a leading `~` becomes the
136/// concatenation `"$HOME"/"rest"` so it still expands inside quotes
137/// (rsync expands `host:~/…` itself; remote bash does not).
138fn shell_dir(rdir: &str) -> String {
139    match rdir.strip_prefix('~') {
140        Some(rest) if rest.is_empty() => "\"$HOME\"".into(),
141        Some(rest) if rest.starts_with('/') => format!("\"$HOME\"/{}", dq_escape(&rest[1..])),
142        _ => dq_escape(rdir),
143    }
144}
145
146/// The remote `bash -s` payload — the command rides verbatim as a
147/// script line so no quoting layer wraps it. Ends by removing the
148/// run dir (cleanup mirrors the local worktree_remove).
149fn remote_script(rdir_shell: &str, envs: &[(String, String)], cmd: &str) -> String {
150    let mut s = format!("cd {rdir_shell} || exit 125\n");
151    for (k, v) in envs {
152        s.push_str(&format!("export {}={}\n", k, sh_escape(v)));
153    }
154    s.push_str(cmd);
155    s.push_str(&format!("\nrc=$?\ncd /\nrm -rf {rdir_shell}\nexit $rc\n"));
156    s
157}

159impl SshHandle {
160    /// Stage 2: ssh in, hand the remote `bash -s` our script via
161    /// stdin. The `exec -a` gives the remote run shell a tagged
162    /// argv0 so cancel's `pkill -f` can find it.
163    fn spawn_remote(&mut self) -> io::Result<Child> {
164        let mut c = Command::new("ssh")
165            .arg(&self.remote)
166            .arg(format!("bash -c \"exec -a {} bash -s\"", self.tag))
167            .stdin(Stdio::piped())
168            .spawn()?;
169        if let Some(mut si) = c.stdin.take() {
170            // A closed/failed stdin just means the remote shell died
171            // before reading — the child's exit will say so.
172            let _ = si.write_all(self.script.as_bytes());
173        }
174        Ok(c)
175    }
176}

178impl RunHandle for SshHandle {
179    fn try_wait(&mut self) -> io::Result<Option<ExitStatus>> {
180        if let Some(st) = self.done {
181            return Ok(Some(st));
182        }
183        match self.child.try_wait()? {
184            None => Ok(None),
185            Some(st) => match self.stage {
186                Stage::Run => {
187                    self.done = Some(st);
188                    Ok(Some(st))
189                }
190                Stage::Sync => {
191                    if !st.success() {
192                        self.sync_failed = true;
193                        self.done = Some(st);
194                        return Ok(Some(st));
195                    }
196                    self.stage = Stage::Run;
197                    self.child = self.spawn_remote()?;
198                    Ok(None)
199                }
200            },
201        }
202    }
203
204    fn wait(&mut self) -> io::Result<ExitStatus> {
205        loop {
206            if let Some(st) = self.try_wait()? {
207                return Ok(st);
208            }
209            std::thread::sleep(Duration::from_millis(50));
210        }
211    }
212
213    fn kill(&mut self) -> io::Result<()> {
214        let res = self.child.kill();
215        let _ = self.child.wait(); // reap the dead stage
216                                   // Best-effort remote cleanup: kill the tagged run shell and
217                                   // drop the run dir. The bracket keeps pkill's pattern from
218                                   // matching the cleanup command's own argv.
219        if let Ok(mut c) = Command::new("ssh")
220            .arg("-o")
221            .arg("ConnectTimeout=5")
222            .arg("-o")
223            .arg("BatchMode=yes")
224            .arg(&self.remote)
225            .arg(format!(
226                "pkill -f '[{}]{}' 2>/dev/null; rm -rf {}",
227                &self.tag[..1],
228                &self.tag[1..],
229                self.rdir_shell
230            ))
231            .stdin(Stdio::null())
232            .stdout(Stdio::null())
233            .stderr(Stdio::null())
234            .spawn()
235        {
236            let _ = c.wait();
237        }
238        res
239    }
240
241    fn failure_desc(&mut self) -> Option<String> {
242        if self.sync_failed {
243            Some("ci infra: sync failed".into())
244        } else if self.done.is_some_and(|s| s.code() == Some(255)) {
245            // ssh exits 255 on transport failure — the pipeline
246            // itself never ran.
247            Some("ci infra: unreachable".into())
248        } else {
249            None
250        }
251    }
252}
253
254impl Drop for SshHandle {
255    fn drop(&mut self) {
256        // Dropped before completion — the remote run would orphan
257        // without the kill+rm its cancel path does.
258        if self.done.is_none() {
259            let _ = self.kill();
260        }
261    }
262}

264impl Runner for SshRunner {
265    fn name(&self) -> &'static str {
266        "ssh"
267    }
268
269    fn spawn(&self, ctx: &RunCtx) -> io::Result<Box<dyn RunHandle>> {
270        let remote = self.cfg.remote.as_deref().ok_or_else(|| {
271            io::Error::new(
272                io::ErrorKind::InvalidInput,
273                "runner kind=ssh needs `remote` (user@host)",
274            )
275        })?;
276        if !is_valid_remote(remote) {
277            return Err(io::Error::new(
278                io::ErrorKind::InvalidInput,
279                format!("runner remote {remote:?} is not a valid user@host"),
280            ));
281        }
282        if !bupha_core::is_valid_env_name(ctx.token_env) {
283            return Err(io::Error::new(
284                io::ErrorKind::InvalidInput,
285                format!("token_env {:?} is not a valid env-var name", ctx.token_env),
286            ));
287        }
288        let base_dir = self.cfg.remote_dir.as_deref().unwrap_or("~/bupha");
289        if base_dir.bytes().any(|b| b < 0x20) {
290            // Control chars (esp. newline) would break out of the
291            // quoted path inside the remote script.
292            return Err(io::Error::new(
293                io::ErrorKind::InvalidInput,
294                "runner remote_dir contains control characters",
295            ));
296        }
297        let cmd = self.cfg.cmd.as_deref().unwrap_or("./ci.sh");
298        let sha12 = &ctx.sha[..12.min(ctx.sha.len())];
299        // key+sha name the remote dir and the pkill tag — both ride
300        // inside remote shell command strings, so force them through
301        // the same allowlist the systemd-unit name uses.
302        let leaf = format!("{}-{}", safe_component(ctx.key), safe_component(sha12));
303        let rdir = format!("{}/{}", base_dir.trim_end_matches('/'), leaf);
304        let tag = format!("bupha-run-{leaf}");
305
306        // Stage 1: ship the worktree. `~/` stays literal — remote
307        // rsync expands it.
308        let target = format!("{remote}:{rdir}/");
309        let child = Command::new("rsync")
310            .args([
311                "-a",
312                "--delete",
313                &format!("{}/", ctx.workdir.display()),
314                &target,
315            ])
316            .spawn()?;
317
318        // The resolved token rides in the stdin script, never argv —
319        // env NAME=value over ssh's command line would sit in remote
320        // `ps`. SendEnv needs AcceptEnv server-side, so it isn't used.
321        let envs: Vec<(String, String)> = match std::env::var(ctx.token_env) {
322            Ok(v) if !v.is_empty() => vec![(ctx.token_env.to_string(), v)],
323            _ => vec![],
324        };
325        let rdir_shell = shell_dir(&rdir);
326        Ok(Box::new(SshHandle {
327            child,
328            stage: Stage::Sync,
329            remote: remote.into(),
330            script: remote_script(&rdir_shell, &envs, cmd),
331            rdir_shell,
332            tag,
333            sync_failed: false,
334            done: None,
335        }))
336    }
337}

339#[cfg(test)]
340mod tests {
341    use super::*;

343    #[test]
344    fn remote_rejects_option_injection() {
345        assert!(is_valid_remote("ci@arm64-box"));
346        assert!(is_valid_remote("host:2222"));
347        assert!(!is_valid_remote("-oProxyCommand=evil"));
348        assert!(!is_valid_remote("--upload-pack=x"));
349        assert!(!is_valid_remote("h;touch /tmp/x"));
350        assert!(!is_valid_remote("h`id`"));
351        assert!(!is_valid_remote(""));
352    }
353
354    #[test]
355    fn leaf_components_are_shell_safe() {
356        assert_eq!(safe_component("owner/name:gate"), "owner-name-gate");
357        assert_eq!(safe_component("a'; rm -rf /;'"), "a---rm--rf----");
358        assert_eq!(safe_component("$(id)"), "--id--");
359    }
