1//! SSH impl of `Runner` — ships the detached worktree to a remote
2//! host and runs the shell command there. Sync and run are separate
3//! stages (rsync first, ssh second) so an infra failure — unreachable
4//! host, rsync error — reports "ci infra: …" instead of a pipeline
5//! "ci failed". The returned handle is a local process pair, so the
6//! worker's wait/cancel logic is unchanged — the remote-ness lives
7//! inside the impl.
8//!
9//! Config: `[runner] kind = "ssh", remote = "u@h", remote_dir =
10//! "~/bupha", cmd = "./ci.sh"` — auth via the user's ssh config
11//! (agent/keys), nothing bupha-specific. Each run lands in its own
12//! remote dir `<remote_dir>/<key>-<sha12>` — two runners sharing a
13//! remote never `rsync --delete` each other's tree.
14//!
15//! The run script — including the resolved `token_env` value — is
16//! fed to the remote `bash -s` over ssh's stdin, so the secret never
17//! appears in a remote (or local) `ps` listing.
18//!
19//! Known caveats:
20//! - `kill()` terminates the local ssh and best-effort `pkill`s the
21//!   remote run shell (tagged argv0) + removes the run dir, but a
22//!   remote process group that outlives its shell can still linger.
23//! - The rsynced worktree's `.git` is a *pointer file* into the main
24//!   clone's gitdir — meaningless on the remote, so `git` commands
25//!   inside the remote run fail. Pipelines that need VCS metadata
26//!   must clone or init their own repo remotely.
27//! - The remote shell cleans its run dir after the command; a hard
28//!   kill mid-run leaves cleanup to the `pkill`/`rm -rf` fallback.
29

59/// Which stage of the remote pipeline is currently in flight.
60/// Splitting them is what lets `failure_desc` tell a sync/transport
61/// failure from a run failure.
62enum Stage {
63    /// rsync worktree -> remote run dir.
64    Sync,
65    /// ssh remote `bash -s` (script already fed to stdin).
66    Run,
67}
68
69/// RunHandle over the two-stage remote run. Holds whichever child is
70/// in flight plus everything needed to arm the run stage once the
71/// sync lands.
72struct SshHandle {
73    /// Current stage's child — the kill() target.
74    child: Child,
75    stage: Stage,
76    /// "user@host".
77    remote: String,
78    /// Quoted/`~`-resolved per-run dir for remote shell lines —
79    /// bash cwd, cleanup target.
80    rdir_shell: String,
81    /// Remote bash argv0 marker — `pkill -f` target on cancel.
82    tag: String,
83    /// Payload fed to the remote `bash -s` over ssh's stdin.
84    script: String,
85    /// Sync stage exited non-zero — infra, not the pipeline.
86    sync_failed: bool,
87    /// Terminal status once either stage completes.
88    done: Option<ExitStatus>,
89}
90
91/// `'…'` with embedded quotes escaped — for the remote script's
92/// single-quoted literals.
93fn sh_escape(s: &str) -> String {
94    format!("'{}'", s.replace('\'', "'\\''"))
95}
96
97/// `"…"` for remote shell lines — escapes what double quotes honor.
98fn dq_escape(s: &str) -> String {
99    format!(
100        "\"{}\"",
101        s.replace('\\', "\\\\")
102            .replace('"', "\\\"")
103            .replace('$', "\\$")
104            .replace('`', "\\`")
105    )
106}
107
108/// `rdir` as a remote-shell-safe path — a leading `~` becomes the
109/// concatenation `"$HOME"/"rest"` so it still expands inside quotes
110/// (rsync expands `host:~/…` itself; remote bash does not).
111fn shell_dir(rdir: &str) -> String {
112    match rdir.strip_prefix('~') {
113        Some(rest) if rest.is_empty() => "\"$HOME\"".into(),
114        Some(rest) if rest.starts_with('/') => format!("\"$HOME\"/{}", dq_escape(&rest[1..])),
115        _ => dq_escape(rdir),
116    }
117}
118
119/// The remote `bash -s` payload — the command rides verbatim as a
120/// script line so no quoting layer wraps it. Ends by removing the
121/// run dir (cleanup mirrors the local worktree_remove).
122fn remote_script(rdir_shell: &str, envs: &[(String, String)], cmd: &str) -> String {
123    let mut s = format!("cd {rdir_shell} || exit 125\n");
124    for (k, v) in envs {
125        s.push_str(&format!("export {}={}\n", k, sh_escape(v)));
126    }
127    s.push_str(cmd);
128    s.push_str(&format!("\nrc=$?\ncd /\nrm -rf {rdir_shell}\nexit $rc\n"));
129    s
130}

132impl SshHandle {
133    /// Stage 2: ssh in, hand the remote `bash -s` our script via
134    /// stdin. The `exec -a` gives the remote run shell a tagged
135    /// argv0 so cancel's `pkill -f` can find it.
136    fn spawn_remote(&mut self) -> io::Result<Child> {
137        let mut c = Command::new("ssh")
138            .arg(&self.remote)
139            .arg(format!("bash -c \"exec -a {} bash -s\"", self.tag))
140            .stdin(Stdio::piped())
141            .spawn()?;
142        if let Some(mut si) = c.stdin.take() {
143            // A closed/failed stdin just means the remote shell died
144            // before reading — the child's exit will say so.
145            let _ = si.write_all(self.script.as_bytes());
146        }
147        Ok(c)
148    }
149}

151impl RunHandle for SshHandle {
152    fn try_wait(&mut self) -> io::Result<Option<ExitStatus>> {
153        if let Some(st) = self.done {
154            return Ok(Some(st));
155        }
156        match self.child.try_wait()? {
157            None => Ok(None),
158            Some(st) => match self.stage {
159                Stage::Run => {
160                    self.done = Some(st);
161                    Ok(Some(st))
162                }
163                Stage::Sync => {
164                    if !st.success() {
165                        self.sync_failed = true;
166                        self.done = Some(st);
167                        return Ok(Some(st));
168                    }
169                    self.stage = Stage::Run;
170                    self.child = self.spawn_remote()?;
171                    Ok(None)
172                }
173            },
174        }
175    }
176
177    fn wait(&mut self) -> io::Result<ExitStatus> {
178        loop {
179            if let Some(st) = self.try_wait()? {
180                return Ok(st);
181            }
182            std::thread::sleep(Duration::from_millis(50));
183        }
184    }
185
186    fn kill(&mut self) -> io::Result<()> {
187        let res = self.child.kill();
188        let _ = self.child.wait(); // reap the dead stage
189        // Best-effort remote cleanup: kill the tagged run shell and
190        // drop the run dir. The bracket keeps pkill's pattern from
191        // matching the cleanup command's own argv.
192        if let Ok(mut c) = Command::new("ssh")
193            .arg("-o")
194            .arg("ConnectTimeout=5")
195            .arg("-o")
196            .arg("BatchMode=yes")
197            .arg(&self.remote)
198            .arg(format!(
199                "pkill -f '[{}]{}' 2>/dev/null; rm -rf {}",
200                &self.tag[..1],
201                &self.tag[1..],
202                self.rdir_shell
203            ))
204            .stdin(Stdio::null())
205            .stdout(Stdio::null())
206            .stderr(Stdio::null())
207            .spawn()
208        {
209            let _ = c.wait();
210        }
211        res
212    }
213
214    fn failure_desc(&mut self) -> Option<String> {
215        if self.sync_failed {
216            Some("ci infra: sync failed".into())
217        } else if self.done.is_some_and(|s| s.code() == Some(255)) {
218            // ssh exits 255 on transport failure — the pipeline
219            // itself never ran.
220            Some("ci infra: unreachable".into())
221        } else {
222            None
223        }
224    }
225}
226
227impl Drop for SshHandle {
228    fn drop(&mut self) {
229        // Dropped before completion — the remote run would orphan
230        // without the kill+rm its cancel path does.
231        if self.done.is_none() {
232            let _ = self.kill();
233        }
234    }
235}

237impl Runner for SshRunner {
238    fn name(&self) -> &'static str {
239        "ssh"
240    }
241
242    fn spawn(&self, ctx: &RunCtx) -> io::Result<Box<dyn RunHandle>> {
243        let remote = self.cfg.remote.as_deref().ok_or_else(|| {
244            io::Error::new(
245                io::ErrorKind::InvalidInput,
246                "runner kind=ssh needs `remote` (user@host)",
247            )
248        })?;
249        let base_dir = self.cfg.remote_dir.as_deref().unwrap_or("~/bupha");
250        let cmd = self.cfg.cmd.as_deref().unwrap_or("./ci.sh");
251        let sha12 = &ctx.sha[..12.min(ctx.sha.len())];
252        let leaf = format!(
253            "{}-{}",
254            ctx.key.replace('/', "-").replace(':', "-"),
255            sha12
256        );
257        let rdir = format!("{}/{}", base_dir.trim_end_matches('/'), leaf);
258        let tag = format!("bupha-run-{leaf}");
259
260        // Stage 1: ship the worktree. `~/` stays literal — remote
261        // rsync expands it.
262        let target = format!("{remote}:{rdir}/");
263        let child = Command::new("rsync")
264            .args([
265                "-a",
266                "--delete",
267                &format!("{}/", ctx.workdir.display()),
268                &target,
269            ])
270            .spawn()?;
271
272        // The resolved token rides in the stdin script, never argv —
273        // env NAME=value over ssh's command line would sit in remote
274        // `ps`. SendEnv needs AcceptEnv server-side, so it isn't used.
275        let envs: Vec<(String, String)> = match std::env::var(ctx.token_env) {
276            Ok(v) if !v.is_empty() => vec![(ctx.token_env.to_string(), v)],
277            _ => vec![],
278        };
279        let rdir_shell = shell_dir(&rdir);
280        Ok(Box::new(SshHandle {
281            child,
282            stage: Stage::Sync,
283            remote: remote.into(),
284            script: remote_script(&rdir_shell, &envs, cmd),
285            rdir_shell,
286            tag,
287            sync_failed: false,
288            done: None,
289        }))
290    }
