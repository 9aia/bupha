1//! bupha-deploy — install-time helpers so users never copy files:
2//! `bupha deploy service` writes + enables the systemd unit,
3//! `bupha deploy webhook` registers the GitHub hook, and
4//! `bupha deploy env` seeds the secrets env file.

6use std::io;
7use std::path::{Path, PathBuf};
8use std::process::Command;

10fn home() -> PathBuf {
11    std::env::var("HOME")
12        .map(PathBuf::from)
13        .unwrap_or(".".into())
14}

16/// Render the user unit. `unit` is a systemd user service running
17/// `bupha {verb}` ("watch" for the listener, "worker" for a fleet
18/// puller); the watcher spawns the configured tunnel itself.
19pub fn write_service_unit(
20    unit: &str,
21    bin: &Path,
22    env_file: &Path,
23    verb: &str,
24    config: Option<&Path>,
25) -> io::Result<PathBuf> {
26    let dir = home().join(".config/systemd/user");
27    std::fs::create_dir_all(&dir)?;
28    let path = dir.join(unit);
29    std::fs::write(
30        &path,
31        format!(
32            "[Unit]
33Description=bupha — headless CI {verb} (webhook -> queue -> runner)
34After=network-online.target
35Wants=network-online.target
36
37[Service]
38EnvironmentFile={}
39ExecStart={} {}{}
40Restart=always
41RestartSec=5
42
43[Install]
44WantedBy=default.target
45",
46            env_file.display(),
47            bin.display(),
48            verb,
49            config
50                .map(|c| format!(" --config {}", c.display()))
51                .unwrap_or_default()
52        ),
53    )?;
54    Ok(path)
55}

57/// systemctl --user daemon-reload + enable --now. Needs `loginctl
58/// enable-linger` (once) to survive logout/boot.
59pub fn enable_service(unit: &str) -> io::Result<()> {
60    run("systemctl", &["--user", "daemon-reload"])?;
61    run("systemctl", &["--user", "enable", "--now", unit])
62}

64/// Seed ~/.config/bupha/env with the secret names (empty values —
65/// never invent secrets). `names` is every env var the config
66/// resolves secrets from (global + per-repo + token).
67pub fn write_env_skeleton(dir: &Path, names: &[&str]) -> io::Result<PathBuf> {
68    std::fs::create_dir_all(dir)?;
69    let path = dir.join("env");
70    if !path.exists() {
71        let mut body = "# Secrets only — everything else lives in bupha.toml\n".to_string();
72        for n in names {
73            body.push_str(n);
74            body.push_str("=\n");
75        }
76        body.push_str("# CODECOV_TOKEN=\n");
77        std::fs::write(&path, body)?;
78    }
79    #[cfg(unix)]
80    {
81        use std::os::unix::fs::PermissionsExt;
82        // token-bearing file — owner-only regardless of umask.
83        std::fs::set_permissions(&path, std::fs::Permissions::from_mode(0o600))?;
84    }
85    Ok(path)
86}

88/// systemctl --user disable --now — stop now, skip at login. Keeps
89/// the unit file so `enable_service` brings it back without a
90/// reinstall (unlike `disable_service`, which removes it).
91pub fn stop_service(unit: &str) -> io::Result<()> {
92    run("systemctl", &["--user", "disable", "--now", unit])
93}

95/// One-line "enabled=<state> active=<state>" for `service status` —
96/// is-enabled/is-active exit non-zero on disabled/inactive units,
97/// so failures report the state instead of throwing.
98pub fn unit_status(unit: &str) -> io::Result<String> {
99    let probe = |args: &[&str]| -> String {
100        Command::new("systemctl")
101            .args(args)
102            .output()
103            .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string())
104            .unwrap_or_else(|_| "unknown".into())
105    };
106    Ok(format!(
107        "enabled={} active={}",
108        probe(&["--user", "is-enabled", unit]),
109        probe(&["--user", "is-active", unit])
110    ))
111}

113/// systemctl --user restart — bounce the watcher; enablement and the
114/// unit file are untouched.

116    run("systemctl", &["--user", "restart", unit])

119/// `service logs` — journal tail for the unit. `follow` hands the
120/// terminal to `journalctl -f` (inherited stdio) and returns its exit
121/// code; non-follow captures + prints the last 50 lines.
122pub fn service_logs(unit: &str, follow: bool) -> io::Result<i32> {
123    if follow {
124        let st = Command::new("journalctl")
125            .args(["--user", "-u", unit, "-f"])
126            .status()?;
127        return Ok(st.code().unwrap_or(-1));
128    }
129    let out = Command::new("journalctl")
130        .args(["--user", "-u", unit, "-n", "50", "--no-pager"])
131        .output()?;
132    print!("{}", String::from_utf8_lossy(&out.stdout));
133    if !out.status.success() {
134        eprint!("{}", String::from_utf8_lossy(&out.stderr));
135    }
136    Ok(out.status.code().unwrap_or(-1))
137}

139/// Disable + remove the user unit. Best-effort stop first — an absent
140/// or never-enabled unit still gets its file removed.
141pub fn disable_service(unit: &str) -> io::Result<PathBuf> {
142    let _ = run("systemctl", &["--user", "disable", "--now", unit]);
143    let path = home().join(".config/systemd/user").join(unit);
144    if path.exists() {
145        std::fs::remove_file(&path)?;
146    }
147    run("systemctl", &["--user", "daemon-reload"])?;
148    Ok(path)
149}
150
151fn run(bin: &str, args: &[&str]) -> io::Result<()> {
152    let st = Command::new(bin).args(args).status()?;
153    if st.success() {
154        Ok(())
155    } else {
156        Err(io::Error::other(format!("{bin} {args:?} -> {st}")))
157    }
158}

160/// systemd impl of the `Service` port.
161pub struct SystemdService {
162    /// Unit name, e.g. "bupha-watch.service".
163    pub unit: String,
164    /// Binary to ExecStart.
165    pub bin: PathBuf,
166    /// EnvironmentFile (secrets), e.g. ~/.config/bupha/env.
167    pub env: PathBuf,
168    /// Optional --config arg.
169    pub config: Option<PathBuf>,
170    /// Subcommand the unit runs — "watch" or "worker".
171    pub verb: String,
172}
173
174impl bupha_service::Service for SystemdService {
175    fn write(&self) -> io::Result<PathBuf> {
176        write_service_unit(
177            &self.unit,
178            &self.bin,
179            &self.env,
180            &self.verb,
181            self.config.as_deref(),
182        )
183    }
184    fn enable(&self) -> io::Result<()> {
185        enable_service(&self.unit)
186    }
187    fn disable(&self) -> io::Result<PathBuf> {
188        disable_service(&self.unit)
189    }
190}
