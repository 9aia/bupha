6/// Where pushes come from. Implementations parse their own payload
7/// format; the queue only sees shas.
8pub trait Trigger: Send + Sync {
9    /// Is this request the kind of event we run CI on (e.g. a push)?
10    fn is_event(&self, headers: &[(String, String)]) -> bool;
11    /// Cheap peek at `repository.full_name` — untrusted input used
12    /// only to pick the watched repo (and its secret) before verify.
13    fn repo_hint(&self, body: &[u8]) -> Option<String>;
14    /// Verify the request signature over the raw body with `secret`
15    /// (resolved per repo by the caller via `repo_hint`).
16    fn verify(&self, secret: &str, body: &[u8], headers: &[(String, String)]) -> bool;
17    /// Parse the payload — `Push` to enqueue, `Skip` for well-formed
18    /// events with nothing to run (e.g. branch deletion), `None` for
19    /// malformed bodies.
20    fn parse_push(&self, body: &[u8]) -> Option<PushOutcome>;
21    /// Unique delivery id from the request headers (e.g. GitHub's
22    /// `X-GitHub-Delivery`). Used for replay dedup — `None` means the
23    /// trigger has no such concept.
24    fn delivery_id(&self, _headers: &[(String, String)]) -> Option<String> {
25        None
26    }
27}
28
29/// What a parsed payload asks the watcher to do.
30pub enum PushOutcome {
31    /// (repo_name, git_ref, sha, sender) — repo is
32    /// `repository.full_name`; the caller filters ref per watched
33    /// repo. `sender` is the forge-attested identity that triggered
34    /// the event (empty string if the forge does not provide one).
35    Push {
36        repo: String,
37        git_ref: String,
38        sha: String,
39        sender: String,
40    },
41    /// Well-formed event that needs no run (deleted branch, zero sha).
42    Skip,
43}

45/// Forge management — webhook registration and lifecycle ops that
46/// live outside request handling. The impl for a forge owns its API
47/// surface; the watcher calls this instead of knowing GitHub.
48pub trait ForgeAdmin: Send + Sync {
49    /// Register the push webhook for `repo` pointing at `url`.
50    /// `token`/`secret` are resolved values (env/keyring/file).
51    fn register_hook(&self, url: &str, secret: &str, token: &str) -> Result<String, String>;
52    /// Remove the webhook at `url` for `repo` (exact-match).
53    fn remove_hook(&self, url: &str, token: &str) -> Result<String, String>;
54    /// The repo this admin instance manages ("owner/name").
55    fn repo(&self) -> &str;
56}
