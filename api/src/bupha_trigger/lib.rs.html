1//! Source port — where pushes come from.
2//!
3//! bupha is a queue engine: `source → queue → effect`. A `Source`
4//! is anything that lands (repo, ref, sha, sender) pushes — a forge
5//! webhook listener, a WAL poller, cron; the queue only sees
6//! `Push`es. `Trigger` is the narrower per-request surface a
7//! webhook `Source` drives (parse/verify one signed payload);
8//! implementations parse their own payload format (GitHub today —
9//! gitea/gitlab = a new impl crate). `ForgeAdmin` is forge
10//! management — webhook registration outside request handling.
11
12use std::sync::Arc;
13
14/// One normalized push — (repo, ref, sha, sender) is all the queue
15/// ever sees. `sender` is the source-attested identity that
16/// triggered the event (empty string if the source has no such
17/// concept).
18pub struct Push {
19    /// `repository.full_name` ("owner/name") — the caller filters
20    /// ref per watched repo.
21    pub repo: String,
22    pub git_ref: String,
23    pub sha: String,
24    pub sender: String,
25}
26
27/// The queue's answer to an emitted push — each source maps it to
28/// its own ack surface (HTTP status codes, log lines).
29pub enum Verdict {
30    /// At least one job queued the sha.
31    Queued,
32    /// Every matched job already had it pending/running.
33    Duplicate,
34    /// Well-formed push, but no job watches that (repo, ref).
35    NoMatch,
36    /// The push itself failed validation (e.g. sha isn't path/argv
37    /// safe) — never reaches the queue.
38    Rejected,
39    /// Access policy refused the sender.
40    Denied,
41}
42
43/// Where emitted pushes land — the shared ingest path into the
44/// queues. Shared across a source's threads (webhook request
45/// threads all emit through the same one).
46pub type Emit = Arc<dyn Fn(&Push) -> Verdict + Send + Sync>;
47
48/// Shutdown probe — `run` polls it and unwinds (e2e graceful
49/// shutdown without signal-handling deps).
50pub type Stop = Arc<dyn Fn() -> bool + Send + Sync>;
51
52/// A push source — `bupha watch` runs every configured source and
53/// feeds their pushes into one shared queue. The github webhook
54/// listener and the sturn WAL poller are today's impls; cron,
55/// poll-ref, file-watch plug in here.
56pub trait Source: Send + Sync {
57    /// Stable id for logs and config errors ("github-webhook").
58    fn name(&self) -> &str;
59    /// Drive the source until `stop` reports true — blocking;
60    /// emitted pushes go through `emit`.
61    fn run(&self, emit: Emit, stop: Stop);
62}
63
64/// Per-request surface a webhook `Source` drives — is this our
65/// event, which repo (untrusted hint), does the signature verify,
66/// what does the payload ask for. Implementations parse their own
67/// payload format; the queue only sees shas.
68pub trait Trigger: Send + Sync {
69    /// Is this request the kind of event we run CI on (e.g. a push)?
70    fn is_event(&self, headers: &[(String, String)]) -> bool;
71    /// Is this request the forge's liveness probe (github's
72    /// `ping` event)? The webhook source short-circuits it with a
73    /// `pong` — no signature, no queue.
74    fn is_ping(&self, _headers: &[(String, String)]) -> bool {
75        false
76    }
77    /// Cheap peek at `repository.full_name` — untrusted input used
78    /// only to pick the watched repo (and its secret) before verify.
79    fn repo_hint(&self, body: &[u8]) -> Option<String>;
80    /// Verify the request signature over the raw body with `secret`
81    /// (resolved per repo by the caller via `repo_hint`).
82    fn verify(&self, secret: &str, body: &[u8], headers: &[(String, String)]) -> bool;
83    /// Parse the payload — `Push` to enqueue, `Skip` for well-formed
84    /// events with nothing to run (e.g. branch deletion), `None` for
85    /// malformed bodies.
86    fn parse_push(&self, body: &[u8]) -> Option<PushOutcome>;
87    /// Unique delivery id from the request headers (e.g. GitHub's
88    /// `X-GitHub-Delivery`). Used for replay dedup — `None` means the
89    /// trigger has no such concept.
90    fn delivery_id(&self, _headers: &[(String, String)]) -> Option<String> {
91        None
92    }
93}
94
95/// What a parsed payload asks the watcher to do.
96pub enum PushOutcome {
97    /// The push to enqueue — repo is `repository.full_name`; the
98    /// caller filters ref per watched repo.
99    Push(Push),
100    /// Well-formed event that needs no run (deleted branch, zero sha).
101    Skip,
102}
103
104/// Forge management — webhook registration and lifecycle ops that
105/// live outside request handling. The impl for a forge owns its API
106/// surface; the watcher calls this instead of knowing GitHub.
107pub trait ForgeAdmin: Send + Sync {
108    /// Register the push webhook for `repo` pointing at `url`.
109    /// `token`/`secret` are resolved values (env/keyring/file).
110    fn register_hook(&self, url: &str, secret: &str, token: &str) -> Result<String, String>;
111    /// Remove the webhook at `url` for `repo` (exact-match).
112    fn remove_hook(&self, url: &str, token: &str) -> Result<String, String>;
113    /// The repo this admin instance manages ("owner/name").
114    fn repo(&self) -> &str;
115}
