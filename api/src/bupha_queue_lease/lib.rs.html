1//! Queue-lease port extension — multi-worker fencing. Single-box
2//! stores (redb) always claim; networked stores compare-and-set.
3
4/// A fencing token — monotonic counter making stale claims detectable.
5pub struct Fence(pub u64);
6
7/// Try to claim a sha for `worker`. `Some(fence)` = this worker owns
8/// the run; `None` = another worker holds it.
9pub trait TryClaim: Send + Sync {
10    fn try_claim(&self, repo: &str, sha: &str, worker: &str) -> std::io::Result<Option<Fence>>;
11    /// Extend a held claim's TTL. Long runs renew periodically so the
12    /// lease doesn't expire mid-build. `false` = the claim was lost
13    /// (expired or stolen). Default: always-held (single-box stores).
14    fn renew(&self, repo: &str, sha: &str, worker: &str) -> std::io::Result<bool> {
15        let _ = (repo, sha, worker);
16        Ok(true)
17    }
18    /// Has this sha already been run to completion by ANY worker?
19    /// Peers drop their redundant queue entries on `true` — without
20    /// it a sha a peer already ran would sit at the front of the
21    /// queue until the lease expired, then re-run. Default: never
22    /// (single box — no peer could have finished it).
23    fn is_done(&self, repo: &str, sha: &str) -> std::io::Result<bool> {
24        let _ = (repo, sha);
25        Ok(false)
26    }
27    /// Mark a sha complete — peers drop their redundant entries.
28    /// Default no-op (single box).
29    fn mark_done(&self, repo: &str, sha: &str) -> std::io::Result<()> {
30        let _ = (repo, sha);
31        Ok(())
32    }
33}
34
35/// Single-box claim — always succeeds with a zero fence. Used when
36/// the store can't be shared (redb): one worker means no contention.
37/// The done set is in-memory so is_done/mark_done still behave
38/// correctly within the process (and in tests).
39#[derive(Default)]
40pub struct LocalClaim {
41    done: std::sync::Mutex<std::collections::HashSet<(String, String)>>,
42}
43
44impl TryClaim for LocalClaim {
45    fn try_claim(&self, _repo: &str, _sha: &str, _worker: &str) -> std::io::Result<Option<Fence>> {
46        Ok(Some(Fence(0)))
47    }
48
49    fn is_done(&self, repo: &str, sha: &str) -> std::io::Result<bool> {
50        Ok(self
51            .done
52            .lock()
53            .unwrap()
54            .contains(&(repo.to_string(), sha.to_string())))
55    }
56
57    fn mark_done(&self, repo: &str, sha: &str) -> std::io::Result<()> {
58        self.done
59            .lock()
60            .unwrap()
61            .insert((repo.to_string(), sha.to_string()));
62        Ok(())
63    }
64}
