1//! bupha-runner-dagger — `dagger call <fn>` in the worktree. Dagger (and
2//! its container engine) is only needed when this runner is selected;
3//! the port knows nothing about it.
4
5use bupha_core::RunnerCfg;
6use bupha_runner::{RunCtx, Runner};
7use std::io;
8use std::process::{Child, Command};
9
10pub struct DaggerRunner {
11    cfg: RunnerCfg,
12    /// `--cli-source <dir>` from `[runner] cli_source` — for repos
13    /// whose pipeline builds against a sibling checkout.
14    cli_source: Option<String>,
15}
16
17impl DaggerRunner {
18    pub fn new(cfg: RunnerCfg, cli_source: Option<String>) -> Self {
19        Self { cfg, cli_source }
20    }
21}
22
23impl Runner for DaggerRunner {
24    fn name(&self) -> &'static str {
25        "dagger"
26    }
27
28    fn spawn(&self, ctx: &RunCtx) -> io::Result<Child> {
29        let c = &self.cfg;
30        let mut cmd = Command::new("dagger");
31        // Flags beyond shards/jobs/token are only emitted when set —
32        // `dagger call` rejects args the module's function doesn't
33        // declare, so the emitted set must match the module signature.
34        cmd.current_dir(ctx.workdir).args([
35            "call", &c.function,
36            "--shards", &c.shards, "--jobs", &c.jobs,
37            "--token", &format!("env://{}", ctx.token_env),
38        ]);
39        if !c.dogfood_jobs.is_empty() {
40            cmd.args(["--dogfood-jobs", &c.dogfood_jobs]);
41        }
42        if let Some(conf) = c.sccache_conf.as_deref() {
43            cmd.args(["--sccache-conf", &format!("file://{conf}")]);
44        }
45        if c.sequential {
46            cmd.arg("--sequential");
47        }
48        if let Some(src) = &self.cli_source {
49            cmd.args(["--cli-source", src]);
50        }
51        cmd.args(&c.extra_args);
52        cmd.spawn()
53    }
54}
