1//! File impl of `ConfigSource` — `bupha.yaml`/`bupha.yml` preferred,
2//! `bupha.toml` permanent. Explicit path → $BUPHA_CONFIG →
3//! ./bupha.* → ~/.config/bupha/bupha.*; at each search step YAML wins
4//! when both formats exist. Parsed `Config`s get `~` expanded in
5//! local-path fields (`Config::expand_tilde`).
6
7use bupha_config::ConfigSource;
8use serde::Deserialize;
9use std::path::{Path, PathBuf};
10
11pub struct TomlSource {
12    /// Explicit `--config` path, or None for the default search.
13    pub path: Option<String>,
14}
15
16/// File names tried, in order, inside each searched directory —
17/// YAML first (preferred), TOML last (permanent fallback).
18const NAMES: [&str; 3] = ["bupha.yaml", "bupha.yml", "bupha.toml"];
19
20/// Which syntax a config path carries — `.yaml`/`.yml` parse as
21/// YAML, everything else as TOML (back-compat for explicit paths).
22fn is_yaml(p: &Path) -> bool {
23    matches!(
24        p.extension().and_then(|e| e.to_str()),
25        Some("yaml") | Some("yml")
26    )
27}
28
29fn parse(text: &str, path: &Path) -> Result<bupha_core::Config, String> {
30    let mut cfg = if is_yaml(path) {
31        let v: serde_norway::Value =
32            serde_norway::from_str(text).map_err(|e| format!("parse {}: {e}", path.display()))?;
33        bupha_core::Config::deserialize(v).map_err(|e| format!("parse {}: {e}", path.display()))?
34    } else {
35        toml::from_str(text).map_err(|e| format!("parse {}: {e}", path.display()))?
36    };
37    cfg.expand_tilde();
38    Ok(cfg)
39}
40
41impl ConfigSource for TomlSource {
42    fn load(&self) -> Result<bupha_core::Config, String> {
43        let candidates: Vec<PathBuf> = match self
44            .path
45            .clone()
46            .or_else(|| std::env::var("BUPHA_CONFIG").ok())
47        {
48            Some(p) => vec![PathBuf::from(p)],
49            None => {
50                let home = std::env::var("HOME")
51                    .map(PathBuf::from)
52                    .unwrap_or(".".into())
53                    .join(".config/bupha");
54                let mut v = Vec::with_capacity(NAMES.len() * 2);
55                for n in NAMES {
56                    v.push(PathBuf::from(n));
57                }
58                for n in NAMES {
59                    v.push(home.join(n));
60                }
61                v
62            }
63        };
64        for p in candidates {
65            if p.exists() {
66                let text = std::fs::read_to_string(&p)
67                    .map_err(|e| format!("read {}: {e}", p.display()))?;
68                return parse(&text, &p);
69            }
70        }
71        Err("no bupha.{yaml,yml,toml} found (search: $BUPHA_CONFIG, ., ~/.config/bupha/)".into())
72    }
73}
