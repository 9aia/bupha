1//! OpenMetrics adapter of `Telemetry` — writes a textfile that
2//! node_exporter/Prometheus scrapes (`bupha_runs_total`,
3//! `bupha_run_wall_seconds`, `bupha_queue_depth`).
4
5use bupha_telemetry::{RunEvent, Telemetry};
6use std::io::Write;
7use std::path::PathBuf;
8
9/// Writes `/path/bupha.prom` atomically on each event.
10pub struct OtelTelemetry {
11    pub path: PathBuf,
12    runs: std::sync::Mutex<u64>,
13}
14
15impl OtelTelemetry {
16    pub fn new(path: impl Into<PathBuf>) -> Self {
17        Self {
18            path: path.into(),
19            runs: std::sync::Mutex::new(0),
20        }
21    }
22}
23
24impl Telemetry for OtelTelemetry {
25    fn event(&self, e: &RunEvent) {
26        let mut n = self.runs.lock().unwrap();
27        if matches!(e, RunEvent::Finished { .. }) {
28            *n += 1;
29        }
30        let mut s = String::from(
31            "# HELP bupha_runs_total Completed CI runs.\n# TYPE bupha_runs_total counter\n",
32        );
33        s.push_str(&format!("bupha_runs_total {}\n", *n));
34        if let RunEvent::Finished { wall, .. } = e {
35            s.push_str(&format!("bupha_run_wall_seconds {}\n", wall.as_secs()));
36        }
37        // atomic write via tmp+rename
38        let tmp = self.path.with_extension("tmp");
39        if std::fs::File::create(&tmp)
40            .and_then(|mut f| f.write_all(s.as_bytes()))
41            .is_ok()
42        {
43            let _ = std::fs::rename(tmp, &self.path);
44        }
45    }
46}
