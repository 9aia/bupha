1//! Audit adapter (ADR-0002 §2) — appends every access decision and
2//! run outcome to `<worktree.base>/audit.log` as JSONL. Two shapes
3//! on one writer: [`AuditLog::entry`] for direct events (access
4//! deny, runner decisions) and [`AuditSink`] — a `StatusSink`
5//! decorator so status posts leave an audit line too.
6
7use std::fs::OpenOptions;
8use std::io::Write;
9use std::path::{Path, PathBuf};
10use std::sync::{Arc, Mutex};
11
12use bupha_status::StatusSink;
13
14/// Append-only JSONL writer. One `Mutex` serializes lines between
15/// the listener thread (access decisions) and run workers (statuses).
16pub struct AuditLog {
17    path: PathBuf,
18    lock: Mutex<()>,
19}
20
21impl AuditLog {
22    /// Log at `path`; the parent dir must exist (worktree.base does —
23    /// the watcher creates it on boot).
24    pub fn new(path: impl Into<PathBuf>) -> Arc<Self> {
25        Arc::new(Self {
26            path: path.into(),
27            lock: Mutex::new(()),
28        })
29    }
30
31    pub fn path(&self) -> &Path {
32        &self.path
33    }
34
35    /// Append one record. Missing fields serialize as `""`, never
36    /// dropped — the log is grep/jq-friendly either way.
37    pub fn entry(&self, sha: &str, repo: &str, sender: &str, outcome: &str, desc: &str) {
38        let ts = std::time::SystemTime::now()
39            .duration_since(std::time::UNIX_EPOCH)
40            .map(|d| d.as_secs())
41            .unwrap_or(0);
42        let line = serde_json::json!({
43            "ts": ts,
44            "sha": sha,
45            "repo": repo,
46            "sender": sender,
47            "outcome": outcome,
48            "desc": desc,
49        })
50        .to_string();
51        // Audit must never break the pipeline — a write error is a
52        // stderr note, not a failure.
53        let _g = self.lock.lock().unwrap();
54        match OpenOptions::new()
55            .create(true)
56            .append(true)
57            .open(&self.path)
58        {
59            Ok(mut f) => {
60                if let Err(e) = writeln!(f, "{line}") {
61                    eprintln!("audit: write {}: {e}", self.path.display());
62                }
63            }
64            Err(e) => eprintln!("audit: open {}: {e}", self.path.display()),
65        }
66    }
67}
68
69/// StatusSink decorator — records `post()` as an audit entry under
70/// this repo, then delegates to the wrapped sink. `sender` isn't in
71/// the `post` signature; status lines leave it empty (access-deny
72/// lines, written via [`AuditLog::entry`] directly, carry it).
73pub struct AuditSink {
74    inner: Arc<dyn StatusSink>,
75    repo: String,
76    log: Arc<AuditLog>,
77}
78
79impl AuditSink {
80    pub fn wrap(inner: Arc<dyn StatusSink>, repo: impl Into<String>, log: Arc<AuditLog>) -> Self {
81        Self {
82            inner,
83            repo: repo.into(),
84            log,
85        }
86    }
87}
88
89impl StatusSink for AuditSink {
90    fn context(&self) -> &str {
91        self.inner.context()
92    }
93
94    fn post(&self, sha: &str, state: &str, desc: &str) {
95        self.log.entry(sha, &self.repo, "", state, desc);
96        self.inner.post(sha, state, desc);
97    }
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103
104    struct Spy(std::sync::Mutex<Vec<(String, String, String)>>);
105    impl StatusSink for Spy {
106        fn context(&self) -> &str {
107            "ci/test"
108        }
109        fn post(&self, sha: &str, state: &str, desc: &str) {
110            self.0
111                .lock()
112                .unwrap()
113                .push((sha.into(), state.into(), desc.into()));
114        }
115    }
116
117    #[test]
118    fn entry_writes_jsonl() {
119        let dir = std::env::temp_dir().join(format!("bupha-audit-{}", std::process::id()));
120        std::fs::create_dir_all(&dir).unwrap();
121        let log = AuditLog::new(dir.join("audit.log"));
122        log.entry("abc", "o/r", "alice", "denied", "sender not allowed");
123        let body = std::fs::read_to_string(log.path()).unwrap();
124        let v: serde_json::Value = serde_json::from_str(body.trim()).unwrap();
125        assert_eq!(v["sha"], "abc");
126        assert_eq!(v["repo"], "o/r");
127        assert_eq!(v["sender"], "alice");
128        assert_eq!(v["outcome"], "denied");
129        assert!(v["ts"].as_u64().unwrap() > 0);
130        std::fs::remove_dir_all(&dir).ok();
131    }
132
133    #[test]
134    fn sink_records_then_delegates() {
135        let dir = std::env::temp_dir().join(format!("bupha-audits-{}", std::process::id()));
136        std::fs::create_dir_all(&dir).unwrap();
137        let log = AuditLog::new(dir.join("audit.log"));
138        let spy = Arc::new(Spy(std::sync::Mutex::new(vec![])));
139        let s = AuditSink::wrap(spy.clone(), "o/r", log.clone());
140        s.post("def", "success", "all green");
141        assert_eq!(spy.0.lock().unwrap().len(), 1);
142        let body = std::fs::read_to_string(log.path()).unwrap();
143        let v: serde_json::Value = serde_json::from_str(body.trim()).unwrap();
144        assert_eq!(v["sha"], "def");
145        assert_eq!(v["outcome"], "success");
146        std::fs::remove_dir_all(&dir).ok();
147    }
148}
