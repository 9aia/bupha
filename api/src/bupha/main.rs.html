189fn build_runner(rc: &bupha_core::RunnerCfg, cache: &bupha_core::CacheCfg) -> Arc<dyn Runner> {
190    // Impls own their config shape — `rc.extra` deserializes into the
191    // matched impl's cfg struct. `[cache]` wins over a legacy
192    // `sccache_conf` entry in `extra`; the resolved conf is injected
193    // back so impls receive data, never backend objects.
194    let mut extra = rc.extra.clone();
195    let cache_impl = SccacheConf {
196        conf: cache.conf.clone().or_else(|| {
197            extra
198                .remove("sccache_conf")
199                .and_then(|v| v.as_str().map(str::to_string))
200        }),
201    };
202    if cache.kind != "none" {
203        if let Some(conf) = cache_impl.conf() {
204            extra.insert("sccache_conf".into(), toml::Value::String(conf));
205        }
206    }
207    let table = toml::Value::Table(extra);
208    match rc.kind.as_str() {
209        "dagger" => {
210            let dc: bupha_dagger::DaggerCfg = table.clone().try_into().unwrap_or_else(|e| {
211                eprintln!("runner config (dagger): {e}");
212                std::process::exit(2)
213            });
214            let cs = dc.cli_source.clone();
215            Arc::new(DaggerRunner::new(dc, cs))
216        }
217        "shell" | "script" => {
218            let sc: bupha_shell::ShellCfg = table.clone().try_into().unwrap_or_else(|e| {
219                eprintln!("runner config (shell): {e}");
220                std::process::exit(2)
221            });
222            Arc::new(ShellRunner::new(sc))
223        }
224        "ssh" => {
225            let sc: bupha_ssh::SshCfg = table.clone().try_into().unwrap_or_else(|e| {
226                eprintln!("runner config (ssh): {e}");
227                std::process::exit(2)
228            });
229            Arc::new(bupha_ssh::SshRunner::new(sc))
230        }
231        "container" => {
232            let cc: ContainerCfg = table.clone().try_into().unwrap_or_else(|e| {
233                eprintln!("runner config (container): {e}");
234                std::process::exit(2)
235            });
236            Arc::new(ContainerRunner::new(cc))
237        }
238        "http" => {
239            let hc: bupha_http::HttpCfg = table.clone().try_into().unwrap_or_else(|e| {
240                eprintln!("runner config (http): {e}");
241                std::process::exit(2)
242            });
243            let s = secret(&hc.secret_env).unwrap_or_else(|| {
244                panic!("{} required (env, keyring or env file)", hc.secret_env)
245            });
246            Arc::new(bupha_http::HttpRunner::new(hc, s))
247        }
248
249        // Coverage builds only: a runner whose handle fails `try_wait`
250        // — the only way e2e tests can reach run_job's wait-error arm
251        // (a real `Child::try_wait` can't fail once the spawn worked).
252        #[cfg(coverage)]
253        "cov-fail-wait" => Arc::new(FailWaitRunner),
254        other => {
255            eprintln!("unknown runner kind '{other}' (dagger|shell|ssh|container)");
256            std::process::exit(2);
257        }
258    }
259}

298fn cfg_dir() -> PathBuf {
299    // Env-only resolution: dirs::config_dir() would fall back to the
300    // passwd entry when HOME is unset, so on a headless box with no
301    // HOME we'd write to the wrong home. XDG_CONFIG_HOME wins, then
302    // HOME/.config, then ./.config.
303    let base = std::env::var("XDG_CONFIG_HOME")
304        .ok()
305        .filter(|s| !s.is_empty())
306        .map(PathBuf::from)
307        .or_else(|| {
308            std::env::var("HOME")
309                .ok()
310                .filter(|s| !s.is_empty())
311                .map(|h| PathBuf::from(h).join(".config"))
312        })
313        .unwrap_or_else(|| PathBuf::from(".").join(".config"));
314    base.join("bupha")
315}

365/// Poll ticks between fleet-claim renewals (~500 ms each, so the
366/// 5-min lease is renewed ~every 2 min). `BUPHA_RENEW_TICKS`
367/// overrides it — e2e tests set it low to exercise renew paths
368/// without a 2-minute build.
369fn renew_ticks() -> u32 {
370    std::env::var("BUPHA_RENEW_TICKS")
371        .ok()
372        .and_then(|v| v.parse::<u32>().ok())
373        .filter(|n| *n > 0)
374        .unwrap_or(240)
375}

377/// How a run ended — the caller needs to know whether THIS worker
378/// may mark the sha done (a lost claim means a peer owns it now).
379enum JobEnd {
380    /// Run completed locally; `false` = failed.
381    Finished(bool),
382    /// Lease lost mid-run — a peer owns the sha; don't mark done.
383    ClaimLost,
384}

386fn run_job(
387    cfg: &Config,
388    repo: &bupha_core::WatchedRepo,
389    sink: &dyn StatusSink,
390    runner: &dyn Runner,
391    shared: &Mutex<Shared>,
392    worker: &Worker<'_>,
393    sha: &str,
394) -> JobEnd {
395    // Implicit jobs keep <base>-<repo>-<sha12>; named jobs insert
396    // the job — sha stays last so `explain`'s ends_with(sha) and
397    // `destroy worktrees`' prefix match still work.
398    let wt = PathBuf::from(match repo.job.is_empty() {
399        true => format!(
400            "{}-{}-{}",
401            cfg.worktree.base,
402            repo.name.replace('/', "-"),
403            &sha[..12.min(sha.len())]
404        ),
405        false => format!(
406            "{}-{}-{}-{}",
407            cfg.worktree.base,
408            repo.name.replace('/', "-"),
409            repo.job,
410            &sha[..12.min(sha.len())]
411        ),
412    });
413    let _ = std::fs::remove_dir_all(&wt);
414    let vcs = GitVcs;
415    vcs.worktree_prune(&repo.dir);
416    // The pushed sha usually isn't in the local clone yet.
417    if !vcs.has_commit(&repo.dir, sha) {
418        vcs.sync(&repo.dir);
419    }
420    if !vcs.worktree_add(&repo.dir, &wt, sha) {
421        eprintln!("worktree add failed for {sha}");
422        sink.post(sha, "error", "ci infra: worktree failed");
423        return JobEnd::Finished(false);
424    }
425    sink.post(sha, "pending", "ci running");
426    println!("ci start {} {sha:.12} runner={}", repo.key, runner.name());
427
428    let ctx = RunCtx {
429        workdir: &wt,
430        token_env: &repo.runner.token_env,
431        key: &repo.key,
432        sha,
433    };
434    /// One poll of the run handle — kept scoped so the child lock
435    /// isn't held across the claim renew or the sleep.
436    enum Poll {
437        Done(std::process::ExitStatus, Option<String>),
438        Running,
439        WaitErr(std::io::Error),
440    }
441    let (end, fail_desc) = match runner.spawn(&ctx) {
442        Ok(ch) => {
443            *worker.child.lock().unwrap() = Some(ch);
444            let mut ticks = 0u32;
445            loop {
446                let poll = {
447                    let mut slot = worker.child.lock().unwrap();
448                    let h = slot.as_mut().unwrap();
449                    match h.try_wait() {
450                        Ok(Some(s)) => Poll::Done(
451                            s,
452                            if s.success() {
453                                None
454                            } else {
455                                h.failure_desc()
456                            },
457                        ),
458                        Ok(None) => Poll::Running,
459                        Err(e) => Poll::WaitErr(e),
460                    }
461                };
462                match poll {
463                    Poll::Done(s, fd) => break (JobEnd::Finished(s.success()), fd),
464                    Poll::Running => {
465                        // Renew the fleet claim every ~2 min — the
466                        // 5-min lease must out-live long builds.
467                        ticks += 1;
468                        if ticks.is_multiple_of(renew_ticks()) {
469                            match worker.claim.renew(&repo.key, sha, worker.id) {
470                                Ok(true) => {}
471                                Ok(false) => {
472                                    // A peer owns the sha now —
473                                    // kill the local run; the peer
474                                    // reports the real result.
475                                    eprintln!(
476                                        "claim lost for {sha:.12} — killing run"
477                                    );
478                                    if let Some(h) =
479                                        worker.child.lock().unwrap().as_mut()
480                                    {
481                                        let _ = h.kill();
482                                    }
483                                    break (JobEnd::ClaimLost, None);
484                                }
485                                Err(e) => eprintln!("claim renew failed: {e}"),
486                            }
487                        }
488                        std::thread::sleep(std::time::Duration::from_millis(500));
489                    }
490                    Poll::WaitErr(e) => {
491                        eprintln!("runner wait failed: {e}");
492                        break (JobEnd::Finished(false), None);
493                    }
494                }
495            }
496        }
497        Err(e) => {
498            eprintln!("runner spawn failed: {e}");
499            (
500                JobEnd::Finished(false),
501                Some("ci infra: spawn failed".to_string()),
502            )
503        }
504    };
505    *worker.child.lock().unwrap() = None;
506    // A killed child exits non-zero; report cancelled, not failed.
507    let cancelled = shared
508        .lock()
509        .unwrap()
510        .queues
511        .get(&repo.key)
512        .map(|q| q.cancelled)
513        .unwrap_or(false);
514    let (state, desc) = match &end {
515        JobEnd::ClaimLost => ("error", "ci claim lost".to_string()),
516        JobEnd::Finished(true) => ("success", "ci passed".to_string()),
517        JobEnd::Finished(false) if cancelled => {
518            ("error", "ci cancelled (superseded)".to_string())
519        }
520        JobEnd::Finished(false) => (
521            "failure",
522            fail_desc.unwrap_or_else(|| "ci failed".into()),
523        ),
524    };
525    sink.post(sha, state, &desc);
526    println!("ci {desc} {sha:.12}");
527    vcs.worktree_remove(&repo.dir, &wt);
528    end
529}

540/// Persist ONE repo's queue. A whole-map write here would clobber a
541/// peer watcher's entries in a shared store — this watcher only
542/// holds its own repos' queues, so persists stay scoped to the key
543/// that actually changed.
544fn persist_queue(g: &Shared, store: &dyn QueueStore, key: &str) {
545    if let Some(q) = g.queues.get(key) {
546        let _ = store.persist_key(
547            key,
548            &PersistedQueue {
549                pending: q.pending.clone(),
550                running: q.running.clone(),
551            },
552        );
553    }
554}

657    wall: std::time::Duration,
658) {
659    let status = match end {
660        JobEnd::Finished(true) => "success",
661        JobEnd::Finished(false) => "failure",
662        JobEnd::ClaimLost => "claim-lost",
663    };
664    telemetry.event(&RunEvent::Finished {
665        repo: key.to_string(),
666        sha: sha.to_string(),
667        status,
668        wall,
669    });
670    let _ = bupha_history::record(
671        &PathBuf::from(&cfg.worktree.base).join("history.jsonl"),
672        &RunRecord {
673            sha: sha.to_string(),
674            repo: key.to_string(),
675            status: status.into(),
676            wall_secs: wall.as_secs(),
677        },
678    );
679    if let Some(att) = attest {
680        let _ = att.attest(&RunAttestation {
681            sha: sha.to_string(),
682            repo: key.to_string(),
683            runner: runner_name.into(),
684            status: status.into(),
685            wall_secs: wall.as_secs(),
686        });
687    }

690fn cmd_watch(cfg: Arc<Config>) {
691    let wh_secret = secret(&cfg.trigger.secret_env).unwrap_or_else(|| {
692        panic!(
693            "{} required (env, keyring or env file)",
694            cfg.trigger.secret_env
695        )
696    });
697    let token = secret(&cfg.runner.token_env).unwrap_or_default();
698
699    let watched: Vec<bupha_core::WatchedRepo> = cfg.watched().unwrap_or_else(|e| {
700        eprintln!("config: {e}");
701        std::process::exit(2);
702    });
703    // Per-repo webhook secrets resolved once at boot — a repo with
704    // `secret_env` set fails fast like the global when the name
705    // doesn't resolve; repos without it share `wh_secret`.

718        "github" => Arc::new(GitHubTrigger),
719        other => {
720            eprintln!("unknown trigger '{other}' (github)");
721            std::process::exit(2);
722        }
723    };
724    // Who may trigger — forge-attested sender checked against policy.
725    let access: Arc<dyn Access> = match cfg.access.kind.as_str() {
726        "all" => Arc::new(AllowAll),

730        "forge" => Arc::new(ForgeAccess::new(&cfg.trigger.api_base, &token)),

732            eprintln!("unknown access kind '{other}' (all|file|forge)");
733            std::process::exit(2);

738    let attest = build_attest(&cfg);
739    let sinks = build_sinks(&cfg, &watched, &token);
740    // Runner per job (cli-sibling detection is repo-relative).
741    let runners = build_runners(&cfg, &watched);
742
743    // Optional tunnel — the watcher stays decoupled from ingress.
744    let tunnel: Option<Box<dyn Tunnel>> = match cfg.tunnel.provider.as_str() {
745        "cloudflared" => Some(Box::new(CloudflaredTunnel::new(cfg.tunnel.clone()))),
746        "none" | "direct" => None,
747        other => {
748            eprintln!("unknown tunnel provider '{other}'");
749            std::process::exit(2);
750        }

752    if let Some(t) = tunnel {
753        std::thread::spawn(move || loop {
754            let Ok(Some(mut c)) = t.spawn() else {
755                break; // no-op provider or spawn fail
756            };
757            println!("tunnel {} started", t.name());
758            let _ = c.wait(); // reap, then restart
759            eprintln!("tunnel {} exited — restarting in 5s", t.name());
760            std::thread::sleep(std::time::Duration::from_secs(5));
761        });
762    }

765    // worktrees live under the base — make sure it exists so a fresh
766    // config boots without a manual mkdir (a file there still fails
767    // at the queue-store open below).
768    let _ = std::fs::create_dir_all(&cfg.worktree.base);
769    let qfile = PathBuf::from(&cfg.worktree.base).join("queue.json");
770    let mut shared = Shared::default();
771    for r in watched.iter() {
772        shared
773            .queues
774            .insert(r.key.clone(), Queue::with_policy(r.policy));
775    }
776    let mut claim: Arc<dyn TryClaim> = Arc::new(LocalClaim::default());
777    // The fleet pull queue — only the shared store feeds it; remote
778    // `bupha worker` processes claim what lands here.
779    let mut workq: Option<Arc<dyn WorkQueue>> = None;
780    let store: Arc<dyn QueueStore> = match cfg.queue.kind.as_str() {
781        "redb" | "" => Arc::new(
782            RedbQueueStore::open(qfile.with_extension("redb"))
783                .unwrap_or_else(|e| panic!("queue store: {e}")),
784        ),
785        "redis" => {
786            let url = cfg.queue.url.clone().unwrap_or_else(|| {
787                eprintln!("[queue] kind=redis needs `url`");
788                std::process::exit(2);
789            });
790            let redis = Arc::new(
791                bupha_queue_redis::RedisQueueStore::open(&url)
792                    .unwrap_or_else(|e| panic!("redis: {e}")),
793            );
794            claim = redis.clone();
795            workq = Some(redis.clone());
796            redis
797        }
798        other => {
799            eprintln!("unknown queue kind '{other}' (redb|redis)");
800            std::process::exit(2);
801        }
802    };
803    // A load failure means "no saved work" — same as an empty map.
804    for (name, mut pq) in store.load().unwrap_or_default() {
805        let Some(q) = shared.queues.get_mut(&name) else {
806            continue; // saved entry for a repo we no longer watch
807        };
808        // A run interrupted mid-flight re-queues at the FRONT of
809        // pending — it never finished, so it gets another shot (the
810        // PersistedQueue doc's promise). A peer that completed it in
811        // the meantime is caught by the done-marker drop at dequeue.
812        if let Some(r) = pq.running.take() {
813            q.pending.push_front(r);
814        }
815        q.pending.extend(pq.pending);
816        if !q.pending.is_empty() {
817            println!("restored {} pending sha(s) for {name}", q.pending.len());
818        }
819    }
820
821    let sinks = Arc::new(sinks);
822    let shared = Arc::new(Mutex::new(shared));
823    let cv = Arc::new(Condvar::new());
824    let child: ChildSlot = Arc::new(Mutex::new(None));
825    {
826        let shared = Arc::clone(&shared);
827        let cv = Arc::clone(&cv);
828        let child = Arc::clone(&child);
829        let cfg = Arc::clone(&cfg);
830        let watched = Arc::clone(&watched);
831        let sinks = Arc::clone(&sinks);
832        let telemetry = telemetry.clone();
833        let attest = attest.clone();
834        let worker_id = worker_id();
835        std::thread::spawn({
836            let store = store.clone();
837            let claim = claim.clone();
838            move || loop {
839                let (repo_name, sha) = {
840                    let mut g = shared.lock().unwrap();
841                    loop {
842                        // Peek the front pending sha per repo; claim it
843                        // in the shared store before dequeuing so a
844                        // second watcher can't double-run it. A sha a
845                        // peer already finished drops off the front —
846                        // otherwise it would sit there until its lease
847                        // expired, then re-run pointlessly.
848                        let hit = g.queues.iter_mut().find_map(|(n, q)| {
849                            let front = q.pending.front()?.clone();
850                            if claim.is_done(n, &front).unwrap_or(false) {
851                                q.dequeue();
852                                return Some((n.clone(), front, false));
853                            }
854                            match claim.try_claim(n, &front, &worker_id) {
855                                Ok(Some(_)) => {
856                                    let s = q.dequeue().unwrap();
857                                    q.running = Some(s.clone());
858                                    q.cancelled = false;
859                                    Some((n.clone(), s, true))
860                                }
861                                _ => None, // claimed elsewhere or store down
862                            }
863                        });
864                        match hit {
865                            // Dropped a peer-completed sha — persist
866                            // and scan again.
867                            Some((n, _, false)) => persist_queue(&g, &*store, &n),
868                            Some((n, s, true)) => {
869                                persist_queue(&g, &*store, &n);
870                                break (n, s);
871                            }
872                            None => g = cv.wait(g).unwrap(),
873                        }
874                    }
875                };
876                let repo = watched.iter().find(|r| r.key == repo_name).unwrap();
877                let sink = &*sinks[&repo_name];
878                let runner = &*runners[&repo_name];
879                telemetry.event(&RunEvent::Started {
880                    repo: repo_name.clone(),
881                    sha: sha.clone(),
882                });
883                let t0 = std::time::Instant::now();
884                let end = run_job(
885                    &cfg,
886                    repo,
887                    sink,
888                    runner,
889                    &shared,
890                    &Worker {
891                        child: &child,
892                        claim: &*claim,
893                        id: &worker_id,
894                    },
895                    &sha,
896                );
897                let wall = t0.elapsed();
898                let status = match &end {
899                    JobEnd::Finished(true) => "success",
900                    JobEnd::Finished(false) => "failure",
901                    JobEnd::ClaimLost => "claim-lost",
902                };
903                telemetry.event(&RunEvent::Finished {
904                    repo: repo_name.clone(),
905                    sha: sha.clone(),
906                    status,
907                    wall,
908                });
909                // Settle the queue BEFORE the observability writes —
910                // anything watching history.jsonl then also sees the
911                // persisted queue drained.
912                {
913                    let mut g = shared.lock().unwrap();
914                    let _ = g.queues.get_mut(&repo_name).map(|q| q.running = None);
915                    persist_queue(&g, &*store, &repo_name);
916                }
917                // Completion marker: peers holding a redundant copy of
918                // this sha drop it instead of re-running. A lost claim
919                // means a peer owns the sha — IT marks done.
920                if !matches!(end, JobEnd::ClaimLost) {
921                    let _ = claim.mark_done(&repo_name, &sha);
922                }
923                // Observability write-paths — failures never break the run.
924                let _ = bupha_history::record(
925                    &PathBuf::from(&cfg.worktree.base).join("history.jsonl"),
926                    &RunRecord {
927                        sha: sha.clone(),
928                        repo: repo_name.clone(),
929                        status: status.into(),
930                        wall_secs: wall.as_secs(),
931                    },
932                );
933                if let Some(att) = &attest {
934                    let _ = att.attest(&RunAttestation {
935                        sha: sha.clone(),
936                        repo: repo_name.clone(),
937                        runner: runner.name().into(),
938                        status: status.into(),
939                        wall_secs: wall.as_secs(),
940                    });
941                }
942            }
943        });
944    }
945
946    let server =
947        Arc::new(tiny_http::Server::http(&cfg.trigger.listen).expect("failed to bind listener"));
948    println!("bupha watch listening on {}", cfg.trigger.listen);
949
950    // e2e hook: with BUPHA_WATCH_SHUTDOWN_FILE set, a watcher exits
951    // the accept loop once that file appears — spawned test binaries
952    // can then shut down gracefully (no signal-handling deps needed).
953    if let Ok(f) = std::env::var("BUPHA_WATCH_SHUTDOWN_FILE") {
954        let srv = Arc::clone(&server);
955        std::thread::spawn(move || {
956            while !Path::new(&f).exists() {
957                std::thread::sleep(std::time::Duration::from_millis(50));
958            }
959            srv.unblock();
960        });
961    }
962    #[cfg(coverage)]
963    {
964        // Second e2e hook (coverage builds only): end the accept loop
965        // after N ms. Lets a test reach the loop *without* the
966        // shutdown-file hook — covering its no-env path — while the
967        // spawned binary still unwinds cleanly and flushes its profile.
968        if let Ok(ms) = std::env::var("BUPHA_WATCH_SHUTDOWN_MS") {
969            let srv = Arc::clone(&server);
970            std::thread::spawn(move || {
971                let ms = ms.parse().unwrap_or(500);
972                std::thread::sleep(std::time::Duration::from_millis(ms));
973                srv.unblock();
974            });
975        }
976    }
977
978    for req in server.incoming_requests() {
979        // Per-request thread — a stalled client must not block the
980        // accept loop (slowloris). Body cap still applies inside.
981        let shared = shared.clone();
982        let cv = cv.clone();
983        let trigger = trigger.clone();
984        let sinks = Arc::clone(&sinks);
985        let child = Arc::clone(&child);
986        let store = store.clone();
987        let watched = Arc::clone(&watched);
988        let secrets = Arc::clone(&secrets);
989        let wh_secret = wh_secret.clone();
990        let access = access.clone();
991        let telemetry = telemetry.clone();
992        let workq = workq.clone();
993        let cfg = Arc::clone(&cfg);
994        std::thread::spawn(move || {
995            let mut req = req;
996            let headers: Vec<(String, String)> = req
997                .headers()
998                .iter()
999                .map(|h| (h.field.as_str().to_string(), h.value.as_str().to_string()))
1000                .collect();
1001            // Only the hook path exists — everything else 404s.
1002            if req.url().split('?').next() != Some("/hook") {
1003                let _ = req
1004                    .respond(tiny_http::Response::from_string("not found").with_status_code(404));
1005                return;
1006            }
1007
1008            let mut body = Vec::new();
1009            let _ = req.as_reader().take(2 * 1024 * 1024).read_to_end(&mut body);
1010
1011            let is_ping = headers
1012                .iter()
1013                .any(|(k, v)| k.eq_ignore_ascii_case("x-github-event") && v == "ping");
1014            let (code, msg) = if is_ping {
1015                (200, "pong")
1016            } else if !trigger.is_event(&headers) {
1017                (400, "not a push event")
1018            } else {
1019                // repo_hint is untrusted until verify passes — it
1020                // only picks which watched repo (and secret) applies.
1021                let repo = trigger
1022                    .repo_hint(&body)
1023                    .and_then(|n| watched.iter().find(|r| r.name == n));
1024                match repo {
1025                    // An unwatched repo can't verify — no secret of
1026                    // ours was used to sign it. Reject, not ignore.
1027                    None => (404, "unwatched repo"),
1028                    Some(r)
1029                        if !trigger.verify(
1030                            secrets.get(&r.name).unwrap_or(&wh_secret),
1031                            &body,
1032                            &headers,
1033                        ) =>
1034                    {
1035                        (401, "bad signature")
1036                    }
1037                    Some(_)
1038                        if trigger
1039                            .delivery_id(&headers)
1040                            .map(|id| !shared.lock().unwrap().seen.insert(id))
1041                            .unwrap_or(false) =>
1042                    {
1043                        // replayed signed payload: same delivery id
1044                        (200, "duplicate delivery")
1045                    }
1046                    Some(_) => {
1047                        match trigger.parse_push(&body) {
1048                            Some(PushOutcome::Skip) => (200, "ignored: no-op event"),
1049                            Some(PushOutcome::Push {
1050                                repo: repo_name,
1051                                git_ref,
1052                                sha,
1053                                sender,
1054                            }) => {
1055                                // Every job of the repo watching this
1056                                // ref gets the push — each keeps its
1057                                // own queue, policy and runner.
1058                                let matched: Vec<&bupha_core::WatchedRepo> = watched
1059                                    .iter()
1060                                    .filter(|w| w.name == repo_name && w.branch == git_ref)
1061                                    .collect();
1062                                if matched.is_empty() {
1063                                    (200, "ignored: not the watched branch")
1064                                } else if !access.allow(&sender, &repo_name) {
1065                                    println!("denied: {sender} may not trigger {repo_name}");
1066                                    (403, "sender not allowed")
1067                                } else {
1068                                    // Lock: mutate queues + decide, then drop.
1069                                    // HTTP posts and process kills happen after.
1070                                    let mut olds: Vec<(String, String)> = vec![];
1071                                    let mut queued = false;
1072                                    let mut kill = false;
1073                                    {
1074                                        let mut g = shared.lock().unwrap();
1075                                        for w in &matched {
1076                                            // The local queue only
1077                                            // holds jobs THIS box can
1078                                            // run — a job wanting
1079                                            // labels we lack belongs
1080                                            // to the fleet pull queue.
1081                                            if !w
1082                                                .runs_on
1083                                                .iter()
1084                                                .all(|l| cfg.worker.labels.contains(l))
1085                                            {
1086                                                continue;
1087                                            }
1088                                            let q = g.queues.get_mut(&w.key).unwrap();
1089                                            // Snapshot before enqueue:
1090                                            // only THIS job's in-flight
1091                                            // run is killable — another
1092                                            // job's running entry must
1093                                            // not be touched.
1094                                            let running = q.running.is_some();
1095                                            match q.enqueue(sha.clone()) {
1096                                                EnqueueResult::Duplicate => {}
1097                                                EnqueueResult::Queued { superseded } => {
1098                                                    queued = true;
1099                                                    telemetry.event(&RunEvent::Queued {
1100                                                        repo: w.key.clone(),
1101                                                        sha: sha.clone(),
1102                                                    });
1103                                                    if let Some(old) = superseded {
1104                                                        olds.push((w.key.clone(), old));
1105                                                    }
1106                                                    if w.runner.cancel_running && running {
1107                                                        q.cancelled = true;
1108                                                        kill = true;
1109                                                    }
1110                                                }
1111                                            }
1112                                        }
1113                                        for w in &matched {
1114                                            persist_queue(&g, &*store, &w.key);
1115                                        }
1116                                    }
1117                                    // Fleet visibility: the shared
1118                                    // pull queue gets every matched
1119                                    // job regardless of local labels.
1120                                    if let Some(wq) = &workq {
1121                                        for w in &matched {
1122                                            let j = Job {
1123                                                key: w.key.clone(),
1124                                                sha: sha.clone(),
1125                                                runs_on: w.runs_on.clone(),
1126                                            };
1127                                            if wq.enqueue(&j).unwrap_or(false) {
1128                                                telemetry.event(&RunEvent::Queued {
1129                                                    repo: j.key.clone(),
1130                                                    sha: sha.clone(),
1131                                                });
1132                                                queued = true;
1133                                            }
1134                                        }
1135                                    }
1136                                    for (key, old) in olds {
1137                                        telemetry.event(&RunEvent::Superseded {
1138                                            repo: key.clone(),
1139                                            sha: old.clone(),
1140                                        });
1141                                        sinks[&key].post(
1142                                            &old,
1143                                            "error",
1144                                            &format!("superseded by {}", &sha[..12.min(sha.len())]),
1145                                        );
1146                                        println!("superseded {key} {old:.12} -> {:.12}", sha);
1147                                    }
1148                                    if kill {
1149                                        let _ = child.lock().unwrap().as_mut().map(|c| {
1150                                            let _ = c.kill();
1151                                            println!("cancelled in-flight run");
1152                                        });
1153                                    }
1154                                    if queued {
1155                                        cv.notify_one();
1156                                        (202, "queued")
1157                                    } else {
1158                                        (200, "already queued")
1159                                    }
1160                                }
1161                            }
1162                            None => (400, "unparseable payload"),
1163                        }
1164                    }
1165                }
1166            };
1167            let _ = req.respond(tiny_http::Response::from_string(msg).with_status_code(code));
1168        });
1169    }
1170}

1172/// `bupha worker` — the fleet puller: claims jobs from the shared
1173/// pull queue ([queue] kind = "redis") and runs them locally. No
1174/// listener, tunnel, webhook secrets or trigger — a worker box only
1175/// needs [[repo]] dirs, [runner], [status], [worktree], [worker].
1176fn cmd_worker(cfg: Arc<Config>) {
1177    let watched: Vec<bupha_core::WatchedRepo> = cfg.watched().unwrap_or_else(|e| {
1178        eprintln!("config: {e}");
1179        std::process::exit(2);
1180    });
1181    let watched = Arc::new(watched);
1182    let token = secret(&cfg.runner.token_env).unwrap_or_default();
1183    let (workq, claim): (Arc<dyn WorkQueue>, Arc<dyn TryClaim>) =
1184        match cfg.queue.kind.as_str() {
1185            "redis" => {
1186                let url = cfg.queue.url.clone().unwrap_or_else(|| {
1187                    eprintln!("[queue] kind=redis needs `url`");
1188                    std::process::exit(2);
1189                });
1190                let r = Arc::new(
1191                    bupha_queue_redis::RedisQueueStore::open(&url)
1192                        .unwrap_or_else(|e| panic!("redis: {e}")),
1193                );
1194                (r.clone(), r)
1195            }
1196            other => {
1197                eprintln!(
1198                    "`bupha worker` needs a shared pull queue — [queue] kind = \
1199                     \"redis\", got '{other}'"
1200                );
1201                std::process::exit(2);
1202            }
1203        };
1204    let sinks = build_sinks(&cfg, &watched, &token);
1205    let runners = build_runners(&cfg, &watched);
1206    let telemetry = build_telemetry(&cfg);
1207    let attest = build_attest(&cfg);
1208    let id = worker_id();
1209    let keys: Vec<KeyPolicy> = watched
1210        .iter()
1211        .map(|w| KeyPolicy {
1212            key: w.key.clone(),
1213            latest: w.policy == bupha_core::QueuePolicy::Latest,
1214        })
1215        .collect();
1216    let shared = Mutex::new(Shared::default());
1217    let child: ChildSlot = Arc::new(Mutex::new(None));
1218    let _ = std::fs::create_dir_all(&cfg.worktree.base);
1219    let poll_ms: u64 = std::env::var("BUPHA_WORKER_POLL_MS")
1220        .ok()
1221        .and_then(|v| v.parse().ok())
1222        .unwrap_or(1000);
1223    println!("bupha worker claiming jobs ({id})");
1224    loop {
1225        // Same e2e hook as watch — spawned test binaries stop cleanly.
1226        if std::env::var("BUPHA_WATCH_SHUTDOWN_FILE")
1227            .ok()
1228            .is_some_and(|f| Path::new(&f).exists())
1229        {
1230            break;
1231        }
1232        match workq.claim_next(&id, &cfg.worker.labels, &keys) {
1233            Ok(Some(Claim::Won { job, superseded })) => {
1234                for j in &superseded {
1235                    sinks[&j.key].post(&j.sha, "error", "superseded by newer push");
1236                }
1237                let repo = watched.iter().find(|r| r.key == job.key).unwrap();
1238                let sink = &sinks[&job.key];
1239                let runner = &runners[&job.key];
1240                telemetry.event(&RunEvent::Started {
1241                    repo: job.key.clone(),
1242                    sha: job.sha.clone(),
1243                });
1244                let t0 = std::time::Instant::now();
1245                let end = run_job(
1246                    &cfg,
1247                    repo,
1248                    &**sink,
1249                    &**runner,
1250                    &shared,
1251                    &Worker {
1252                        child: &child,
1253                        claim: &*claim,
1254                        id: &id,
1255                    },
1256                    &job.sha,
1257                );
1258                let wall = t0.elapsed();
1259                // A lost claim means a peer owns the job — IT settles.
1260                if !matches!(end, JobEnd::ClaimLost) {
1261                    let _ = workq.complete(&job, &id);
1262                }
1263                settle_run(&cfg, &*telemetry, &attest, runner.name(), &job.key, &job.sha, &end, wall);
1264            }
1265            Ok(Some(Claim::Pruned(sup))) => {
1266                for j in &sup {
1267                    if let Some(s) = sinks.get(&j.key) {
1268                        s.post(&j.sha, "error", "superseded by newer push");
1269                    }
1270                }
1271            }
1272            Ok(None) => {}
1273            Err(e) => eprintln!("claim scan: {e}"),
1274        }
1275        std::thread::sleep(std::time::Duration::from_millis(poll_ms));
1276    }
1277}

1376                    u
1377                }
1378            })

1387        println!("wrote {}", main_py.display());
1388    }
1389    println!("next: `dagger call ci --token env://BUPHA_TOKEN` to run the gate");
1390}

1583        ),

1594        if let Some(free) = out

1831        DeploySub::Webhook => {

1835                std::process::exit(1);
1836            }
1837            let watched = cfg.watched().unwrap_or_else(|e| {
1838                eprintln!("{e}");
1839                std::process::exit(1);

1851                    let env = r.secret_env.as_deref().unwrap_or(&cfg.trigger.secret_env);

1853                        eprintln!("{env} required (env, keyring or env file)");

1856                    let msg = (GitHubAdmin {
1857                        repo: r.name.clone(),

1861                    .unwrap_or_else(|e| {
1862                        eprintln!("webhook {}: {e}", r.name);
1863                        std::process::exit(1);
1864                    });
1865                    println!("{msg}");
1866                });
1867        }

1876/// Seed the release flow: publish every workspace member in dep
1877/// order, then tag `{repo}-v{version}` and push the tag. Explicit —
1878/// this is the only irreversible step in bupha's lifecycle.
1879fn cmd_deploy_release(dry_run: bool) {
1880    let repo_dir = std::env::current_dir().unwrap_or(PathBuf::from("."));
1881    if std::env::var("CARGO_REGISTRY_TOKEN")
1882        .unwrap_or_default()
1883        .is_empty()
1884        && !dry_run
1885    {
1886        eprintln!("CARGO_REGISTRY_TOKEN required");
1887        std::process::exit(1);
1888    }
1889    let out = Command::new("cargo")
1890        .args(["metadata", "--format-version", "1", "--no-deps"])
1891        .current_dir(&repo_dir)
1892        .output()
1893        .unwrap_or_else(|e| {
1894            eprintln!("cargo metadata: {e}");
1895            std::process::exit(1)
1896        });
1897    let meta: serde_json::Value = serde_json::from_slice(&out.stdout).unwrap_or_else(|e| {
1898        eprintln!("metadata json: {e}");
1899        std::process::exit(1)
1900    });
1901    let empty: Vec<serde_json::Value> = vec![];
1902    let member_ids: std::collections::HashSet<String> = meta["workspace_members"]
1903        .as_array()
1904        .unwrap_or(&empty)
1905        .iter()
1906        .filter_map(|m| m.as_str().map(String::from))
1907        .collect();
1908    let pkgs: Vec<&serde_json::Value> = meta["packages"]
1909        .as_array()
1910        .unwrap_or(&empty)
1911        .iter()
1912        .filter(|p| member_ids.contains(p["id"].as_str().unwrap_or("")))
1913        .collect();
1914    let members: Vec<String> = pkgs
1915        .iter()
1916        .filter_map(|p| p["name"].as_str().map(String::from))
1917        .collect();
1918    let deps: Vec<(String, String)> = pkgs
1919        .iter()
1920        .flat_map(|p| {
1921            let a = p["name"].as_str().unwrap_or("").to_string();
1922            p["dependencies"]
1923                .as_array()
1924                .unwrap_or(&empty)
1925                .iter()
1926                .filter(|d| d["path"].is_string())
1927                .filter_map(|d| d["name"].as_str().map(|n| (a.clone(), n.to_string())))
1928                .collect::<Vec<_>>()
1929        })
1930        .collect();
1931    let order = bupha_core::publish_order(&members, &deps).unwrap_or_else(|e| {
1932        eprintln!("{e}");
1933        std::process::exit(1)
1934    });
1935    let versions: std::collections::HashSet<String> = pkgs
1936        .iter()
1937        .filter_map(|p| p["version"].as_str().map(String::from))
1938        .collect();
1939    if versions.len() != 1 {
1940        eprintln!(
1941            "members have mixed versions {versions:?} — \
1942                   release needs one version across the workspace"
1943        );
1944        std::process::exit(1);
1945    }
1946    let ver = versions.into_iter().next().unwrap();
1947    let tag = format!(
1948        "{}-v{ver}",
1949        repo_dir
1950            .file_name()
1951            .map(|n| n.to_string_lossy().to_string())
1952            .unwrap_or("release".into())
1953    );
1954    if dry_run {
1955        println!("publish order: {}", order.join(" -> "));
1956        println!("tag: {tag}");
1957        return;
1958    }
1959    for name in &order {
1960        let mut ok = false;
1961        for _ in 0..5 {
1962            let st = Command::new("cargo")
1963                .args(["publish", "-p", name, "--allow-dirty"])
1964                .current_dir(&repo_dir)
1965                .output();
1966            let text = st
1967                .map(|o| {
1968                    format!(
1969                        "{}{}",
1970                        String::from_utf8_lossy(&o.stdout),
1971                        String::from_utf8_lossy(&o.stderr)
1972                    )
1973                })
1974                .unwrap_or_default();
1975            // "Finished" also appears for a successful package build
1976            // whose upload then fails — only the upload counts.
1977            if text.contains("Uploaded") || text.contains("already") {
1978                println!("published {name}");
1979                ok = true;
1980                break;
1981            }
1982            std::thread::sleep(std::time::Duration::from_secs(20));
1983        }
1984        if !ok {
1985            eprintln!("publish failed: {name}");
1986            std::process::exit(1);
1987        }
1988    }
1989    let vcs = GitVcs;
1990    if !vcs.tag_exists(&repo_dir, &tag) && !vcs.tag(&repo_dir, &tag) {
1991        eprintln!("git tag {tag} failed");
1992        std::process::exit(1);
1993    }
1994    if !vcs.push_tag(&repo_dir, &tag) {
1995        eprintln!("tag push failed — publish done, push {tag} by hand");
1996        std::process::exit(1);
1997    }
1998    println!(
1999        "released {tag} — pushes to main now release via \
2000              the watch loop"
2001    );
2002}

2004fn cmd_destroy(cfg: &Config, sub: &DestroySub) {
2005    match sub {
2006        DestroySub::Service { worker } => {
2007            let svc = deploy::SystemdService {
2008                unit: if *worker {
2009                    "bupha-worker.service".into()
2010                } else {
2011                    "bupha-watch.service".into()
2012                },
2013                bin: PathBuf::new(),
2014                env: cfg_dir().join("env"),
2015                config: None,
2016                verb: "watch".into(),
2017            };
2018            match svc.disable() {
2019                Ok(p) => println!("removed {}", p.display()),
2020                Err(e) => {
2021                    eprintln!("service: {e}");
2022                    std::process::exit(1);
2023                }
2024            }
2025        }
2026        DestroySub::Webhook => {
2027            if cfg.tunnel.public_url.is_empty() {
2028                eprintln!(
2029                    "tunnel.public_url not set — nothing to \
2030                           match hooks against"
2031                );
2032                std::process::exit(1);
2033            }
2034            let watched = cfg.watched().unwrap_or_else(|e| {
2035                eprintln!("{e}");
2036                std::process::exit(1);
2037            });
2038            let token = secret(&cfg.runner.token_env).unwrap_or_default();
2039            // One hook per repo — jobs share it.
2040            let mut seen: std::collections::HashSet<&str> = Default::default();
2041            watched
2042                .iter()
2043                .filter(|r| seen.insert(r.name.as_str()))
2044                .for_each(|r| {
2045                    let msg = (GitHubAdmin {
2046                        repo: r.name.clone(),
2047                        api_base: cfg.trigger.api_base.clone(),
2048                    })
2049                    .remove_hook(&cfg.tunnel.public_url, &token)
2050                    .unwrap_or_else(|e| {
2051                        eprintln!("webhook {}: {e}", r.name);
2052                        std::process::exit(1);
2053                    });
2054                    println!("{msg}");
2055                });
2056        }
2057        DestroySub::Worktrees => {
2058            // worktrees land at <base>-<repo>[-<job>]-<sha12>;
2059            // queue.json in base
2060            let base = PathBuf::from(&cfg.worktree.base);
2061            let prefix = base
2062                .file_name()
2063                .map(|n| format!("{}-", n.to_string_lossy()))
2064                .unwrap_or_default();
2065            if let Some(parent) = base.parent() {
2066                if let Ok(rd) = std::fs::read_dir(parent) {
2067                    for e in rd.flatten() {
2068                        // .git must be a FILE (worktree pointer), not a
2069                        // dir — protects foreign dirs sharing the prefix.
2070                        if e.file_name().to_string_lossy().starts_with(&prefix)
2071                            && e.path().is_dir()
2072                            && e.path().join(".git").is_file()
2073                        {
2074                            let _ = std::fs::remove_dir_all(e.path());
2075                            println!("removed {}", e.path().display());
2076                        }
2077                    }
2078                }
2079            }
2080            let q = base.join("queue.json");
2081            if q.exists() {
2082                let _ = std::fs::remove_file(&q);
2083                println!("removed {}", q.display());
2084            }
2085            if let Ok(ws) = cfg.watched() {
2086                // One prune per repo — jobs share the clone.
2087                let mut seen: std::collections::HashSet<&str> = Default::default();
2088                for r in ws.iter().filter(|r| seen.insert(r.name.as_str())) {
2089                    GitVcs.worktree_prune(&r.dir);
2090                }
2091            }
2092        }
2093    }
2094}

2110    let env_file = std::fs::read_to_string(home.join(".config/bupha/env")).unwrap_or_default();
2111    for name in webhook_secret_names(cfg) {
2112        let secret_ok = secret(name).is_some()
2113            || env_file

2115                .any(|l| l.starts_with(&format!("{name}=")) && l.len() > name.len() + 1);

2117            "webhook secret set",
2118            secret_ok,
2119            &format!("{name} (env or env file)"),
2120            &mut fails,
2121        );
2122    }
2123    let token_ok = secret(&cfg.runner.token_env).is_some()
2124        || env_file.lines().any(|l| {
2125            l.starts_with(&format!("{}=", cfg.runner.token_env))
2126                && l.len() > cfg.runner.token_env.len() + 1

2128    check(
2129        "token set",
2130        token_ok,
2131        &format!("{} (env or env file)", cfg.runner.token_env),
2132        &mut fails,

2135    // is a note, not a failure.
2136    if cfg.tunnel.public_url.is_empty() {
2137        check(
2138            "tunnel.public_url",
2139            true,
2140            "unset (needed only for webhook registration)",
2141            &mut fails,

2145        check(
2146            "endpoint reachable",
2147            code.is_some(),
2148            &format!("{} -> {:?}", cfg.tunnel.public_url, code),
2149            &mut fails,
2150        );
2151        if cfg.tunnel.public_url.contains(".trycloudflare.com") {
2152            println!(
2153                "  note: trycloudflare quick tunnels are flaky — \
2154                prefer a named tunnel"
2155            );
2156        }
2157    }
2158    if fails == 0 {
2159        println!("deploy check: all green");
2160    } else {
2161        println!("{fails} problem(s)");
2162        std::process::exit(1);

2166fn ureq_probe(url: &str) -> Option<u16> {
2167    // Any HTTP answer (even a 4xx) proves the endpoint is reachable;
2168    // only transport-level failure is None.
2169    let agent = ureq::Agent::config_builder()
2170        .timeout_global(Some(std::time::Duration::from_secs(10)))
2171        .build()
2172        .new_agent();
2173    match agent
2174        .post(&format!("{}/hook", url.trim_end_matches('/')))
2175        .header("User-Agent", "bupha")
2176        .send_empty()
2177    {
2178        Ok(r) => Some(r.status().as_u16()),
2179        Err(ureq::Error::StatusCode(c)) => Some(c),
2180        Err(_) => None,
2181    }
2182}

2188fn cmd_runs(cfg: &Config, repo: Option<&str>) {
2189    let p = history_path(cfg);
2190    let Ok(s) = std::fs::read_to_string(&p) else {
2191        println!("no runs yet ({})", p.display());
2192        return;
2193    };
2194    println!("{:<12} {:<30} {:<9} {:>6}", "sha", "repo", "status", "wall");
2195    for line in s.lines() {
2196        let Ok(v) = serde_json::from_str::<serde_json::Value>(line) else {
2197            continue;
2198        };
2199        let r = v["repo"].as_str().unwrap_or_default();
2200        // A bare repo filter also matches its job keys — `runs o/r`
2201        // catches `o/r:gate`; `runs o/r:gate` stays exact.
2202        if repo.is_some_and(|w| w != r && !r.starts_with(&format!("{w}:"))) {
2203            continue;
2204        }
2205        println!(
2206            "{:<12} {:<30} {:<9} {:>5}s",
2207            &v["sha"].as_str().unwrap_or("?")[..12.min(v["sha"].as_str().unwrap_or("?").len())],
2208            r,
2209            v["status"].as_str().unwrap_or("?"),
2210            v["wall_secs"].as_u64().unwrap_or(0),
2211        );
2212    }
2213}

2230fn cmd_explain(cfg: &Config, sha: &str) {
2231    // 1. History — did it run?
2232    if let Ok(s) = std::fs::read_to_string(history_path(cfg)) {
2233        for line in s.lines().rev() {
2234            let Ok(v) = serde_json::from_str::<serde_json::Value>(line) else {
2235                continue;
2236            };
2237            if v["sha"].as_str().unwrap_or_default().starts_with(sha) {
2238                println!(
2239                    "ran: {} ({}) in {}s",
2240                    v["status"].as_str().unwrap_or("?"),
2241                    v["repo"].as_str().unwrap_or("?"),
2242                    v["wall_secs"].as_u64().unwrap_or(0)
2243                );
2244                return;
2245            }
2246        }
2247    }
2248    // 2. Queue — is it pending/running?
2249    let qf = PathBuf::from(&cfg.worktree.base).join("queue.redb");
2250    if let Ok(store) = RedbQueueStore::open(qf) {
2251        let map = store.load().unwrap_or_default();
2252        for (name, q) in &map {
2253            if q.running.as_deref().is_some_and(|r| r.starts_with(sha)) {
2254                println!("running now on {name}");
2255                return;
2256            }
2257            if q.pending.iter().any(|p| p.starts_with(sha)) {
2258                println!(
2259                    "queued on {name} ({} ahead in line)",
2260                    q.pending
2261                        .iter()
2262                        .position(|p| p.starts_with(sha))
2263                        .unwrap_or(0)
2264                );
2265                return;
2266            }
2267        }
2268    }
2269    // 3. Worktree — did a run start but never finish?
2270    let base = &cfg.worktree.base;
2271    if let Ok(rd) = std::fs::read_dir(base) {
2272        for e in rd.flatten() {
2273            if e.file_name().to_string_lossy().ends_with(sha) {
2274                println!(
2275                    "worktree exists ({}), no result — run died mid-flight",
2276                    e.path().display()
2277                );
2278                return;
2279            }
2280        }
2281    }
2282    println!("never seen — not in history, queue, or worktrees");
