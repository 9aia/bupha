1//! bupha — "the oxpecker on your repo's back": headless CI that
2//! watches pushes, batches them latest-wins, runs a pluggable runner
3//! per sha, and reports statuses. No server, no UI.
4//!
5//! Config: bupha.toml (secrets come from env). See bupha-core::Config.

34#[derive(Subcommand)]
35enum Cmd {
36    /// Serve the webhook → queue → runner loop.
37    Watch,
38    /// Scaffold CI in a repo (dagger module install or script skeleton).
39    Init {
40        /// Repo dir to initialize (default: cwd).
41        #[arg(default_value = ".")]
42        dir: PathBuf,
43    },
44    /// Check what the loaded config actually needs on this box.
45    Doctor,
46    /// Run the pipeline locally (the runner's invocation, current dir).

48        /// Function for the dagger runner (default: from config).

53        #[command(subcommand)]

56    /// Teardown counterparts of `deploy`.

58        #[command(subcommand)]

97fn build_runner(rc: &bupha_core::RunnerCfg) -> Arc<dyn Runner> {
98    match rc.kind.as_str() {
99        "dagger" => Arc::new(DaggerRunner::new(
100            rc.clone(), rc.cli_source.clone())),
101        "script" => Arc::new(ScriptRunner::new(rc.clone())),
102        other => {
103            eprintln!("unknown runner kind '{other}' (dagger|script)");
104            std::process::exit(2);
105        }
106    }
107}

109/// Secret resolution: env var → OS keyring (libsecret) → env file.
110/// Headless boxes without dbus/keyring still work via the env file;
111/// desktops stop keeping tokens in plaintext files.
112fn secret(name: &str) -> Option<String> {
113    if let Ok(v) = std::env::var(name) {
114        if !v.is_empty() { return Some(v); }
115    }
116    if let Ok(o) = Command::new("secret-tool")
117        .args(["lookup", "service", "bupha", "name", name])
118        .output()
119    {
120        if o.status.success() {
121            let v = String::from_utf8_lossy(&o.stdout)
122                .trim().to_string();
123            if !v.is_empty() { return Some(v); }
124        }
125    }
126    let f = std::fs::read_to_string(
127        std::env::var("HOME").map(PathBuf::from)
128            .unwrap_or_else(|_| ".".into())
129            .join(".config/bupha/env")).unwrap_or_default();
130    f.lines().find_map(|l| {
131        let (k, v) = l.split_once('=')?;
132        (k.trim() == name && !v.trim().is_empty())
133            .then(|| v.trim().to_string())
134    })
135}

145fn run_job(
146    cfg: &Config,
147    repo: &bupha_core::WatchedRepo,
148    sink: &dyn StatusSink,
149    runner: &dyn Runner,
150    shared: &Mutex<Shared>,
151    child: &ChildSlot,
152    sha: &str,
153) -> bool {
154    let wt = PathBuf::from(format!(
155        "{}-{}-{}", cfg.worktree.base,
156        repo.name.replace('/', "-"), &sha[..12.min(sha.len())]));
157    let _ = std::fs::remove_dir_all(&wt);
158    git(&repo.dir, &["worktree", "prune"]);
159    // The pushed sha usually isn't in the local clone yet.
160    if !git(&repo.dir, &["cat-file", "-e", sha]) {
161        git(&repo.dir, &["fetch", "origin", "--prune"]);
162    }
163    if !git(&repo.dir,
164            &["worktree", "add", "--detach", wt.to_str().unwrap(), sha]) {
165        eprintln!("worktree add failed for {sha}");
166        sink.post(sha, "error", "ci infra: worktree failed");
167        return false;
168    }
169    sink.post(sha, "pending", "ci running");
170    println!("ci start {} {sha:.12} runner={}", repo.name, runner.name());
171
172    let ctx = RunCtx { workdir: &wt, token_env: &repo.runner.token_env };
173    let ok = match runner.spawn(&ctx) {
174        Ok(ch) => {
175            *child.lock().unwrap() = Some(ch);
176            loop {
177                let done = child.lock().unwrap()
178                    .as_mut().unwrap().try_wait();
179                match done {
180                    Ok(Some(s)) => break s.success(),
181                    Ok(None) => std::thread::sleep(
182                        std::time::Duration::from_millis(500)),
183                    Err(e) => {
184                        eprintln!("runner wait failed: {e}");
185                        break false;
186                    }
187                }
188            }
189        }
190        Err(e) => {
191            eprintln!("runner spawn failed: {e}");
192            false
193        }
194    };
195    *child.lock().unwrap() = None;
196    // A killed child exits non-zero; report cancelled, not failed.
197    let cancelled = shared.lock().unwrap()
198        .queues.get(&repo.name).map(|q| q.cancelled).unwrap_or(false);
199    let (state, desc) = if ok {
200        ("success", "ci passed")
201    } else if cancelled {
202        ("error", "ci cancelled (superseded)")
203    } else {
204        ("failure", "ci failed")
205    };
206    sink.post(sha, state, desc);
207    println!("ci {desc} {sha:.12}");
208    git(&repo.dir,
209        &["worktree", "remove", "--force", wt.to_str().unwrap()]);
210    ok
211}

213/// All watched repos share one worker (no overlapping builds on one
214/// box) but keep independent pending slots — latest-wins per repo.
215#[derive(Default)]
216struct Shared {
217    queues: std::collections::HashMap<String, Queue>,
218    /// dedup for replayed webhook deliveries (X-GitHub-Delivery)
219    seen: bupha_core::Seen,
220}
221
222fn persist_queue(g: &Shared, path: &Path) {
223    let v: serde_json::Map<_, _> = g.queues.iter().map(|(n, q)| {
224        (n.clone(), serde_json::json!({
225            "pending": q.pending, "running": q.running }))
226    }).collect();
227    // tmp + rename so a crash mid-write can't lose accepted work.
228    let tmp = path.with_extension("json.tmp");
229    if std::fs::write(&tmp, serde_json::Value::Object(v).to_string())
230        .is_ok()
231    {
232        let _ = std::fs::rename(&tmp, path);
233    }
234}

236fn cmd_watch(cfg: Arc<Config>) {
237    let wh_secret = secret(&cfg.trigger.secret_env)
238        .unwrap_or_else(|| panic!("{} required (env, keyring or env file)", cfg.trigger.secret_env));
239    let token = secret(&cfg.runner.token_env).unwrap_or_default();
240
241    let watched: Vec<bupha_core::WatchedRepo> = cfg.watched()
242        .unwrap_or_else(|e| {
243            eprintln!("config: {e}");
244            std::process::exit(2);
245        });
246    let watched = Arc::new(watched);
247
248    let trigger: Box<dyn Trigger> = match cfg.trigger.kind.as_str() {
249        "github" => Box::new(GitHubTrigger { secret: wh_secret }),
250        other => {
251            eprintln!("unknown trigger '{other}' (github)");
252            std::process::exit(2);
253        }
254    };
255    let mut sinks: std::collections::HashMap<String, Arc<dyn StatusSink>> =
256        std::collections::HashMap::new();
257    for r in watched.iter() {
258        let s: Arc<dyn StatusSink> = match cfg.status.kind.as_str() {
259            "github" => Arc::new(GitHubStatus {
260                token: token.clone(),
261                repo: r.status_repo.clone(),
262                context: cfg.status.context.clone(),
263            }),
264            _ => Arc::new(LogStatus {
265                context: cfg.status.context.clone(),
266            }),
267        };
268        sinks.insert(r.name.clone(), s);
269    }
270
271    // Runner per repo dir (cli-sibling detection is repo-relative).
272    let mut runners: std::collections::HashMap<String, Arc<dyn Runner>> =
273        std::collections::HashMap::new();
274    for r in watched.iter() {
275        runners.insert(r.name.clone(), build_runner(&r.runner));
276    }
277
278    // Optional tunnel — the watcher stays decoupled from ingress.
279    let tunnel: Option<Box<dyn Tunnel>> = match cfg.tunnel.provider.as_str() {
280        "cloudflared" => Some(Box::new(CloudflaredTunnel::new(cfg.tunnel.clone()))),
281        "none" | "direct" => None,
282        other => {
283            eprintln!("unknown tunnel provider '{other}'");
284            std::process::exit(2);
285        }
286    };
287    if let Some(t) = tunnel {
288        std::thread::spawn(move || loop {
289            match t.spawn() {
290                Ok(Some(mut c)) => {
291                    println!("tunnel {} started", t.name());
292                    let _ = c.wait();            // reap, then restart
293                    eprintln!("tunnel {} exited — restarting in 5s", t.name());
294                }
295                Ok(None) | Err(_) => break,    // no-op provider or spawn fail
296            }
297            std::thread::sleep(std::time::Duration::from_secs(5));
298        });
299    }
300
301    // Restore queued work from a previous life.
302    let qfile = PathBuf::from(&cfg.worktree.base).join("queue.json");
303    let mut shared = Shared::default();
304    for r in watched.iter() {
305        shared.queues.insert(r.name.clone(),
306            Queue::with_policy(r.policy));
307    }
308    if let Ok(text) = std::fs::read_to_string(&qfile) {
309        if let Ok(serde_json::Value::Object(m)) =
310            serde_json::from_str::<serde_json::Value>(&text)
311        {
312            for (name, st) in m {
313                if let Some(q) = shared.queues.get_mut(&name) {
314                    if let Some(arr) = st.get("pending")
315                        .and_then(|x| x.as_array())
316                    {
317                        for v in arr {
318                            if let Some(sha) = v.as_str() {
319                                q.pending.push_back(sha.to_string());
320                            }
321                        }
322                    }
323                    if let Some(sha) = st.get("running")
324                        .and_then(|x| x.as_str())
325                    {   // interrupted run re-queues at the front
326                        q.pending.push_front(sha.to_string());
327                    }
328                    if !q.pending.is_empty() {
329                        println!("restored {} pending sha(s) for {name}",
330                            q.pending.len());
331                    }
332                }
333            }
334        }
335    }
336
337    let sinks = Arc::new(sinks);
338    let shared = Arc::new(Mutex::new(shared));
339    let cv = Arc::new(Condvar::new());
340    let child: ChildSlot = Arc::new(Mutex::new(None));
341    {
342        let shared = Arc::clone(&shared);
343        let cv = Arc::clone(&cv);
344        let child = Arc::clone(&child);
345        let cfg = Arc::clone(&cfg);
346        let watched = Arc::clone(&watched);
347        let sinks = Arc::clone(&sinks);
348        let qfile = qfile.clone();
349        std::thread::spawn(move || {
350            loop {
351                let (repo_name, sha) = {
352                    let mut g = shared.lock().unwrap();
353                    loop {
354                        let hit = g.queues.iter_mut().find_map(|(n, q)| {
355                            q.dequeue().map(|s| {
356                                q.running = Some(s.clone());
357                                q.cancelled = false;
358                                (n.clone(), s)
359                            })
360                        });
361                        if let Some(x) = hit {
362                            persist_queue(&g, &qfile);
363                            break x;
364                        }
365                        g = cv.wait(g).unwrap();
366                    }
367                };
368                let repo = watched.iter()
369                    .find(|r| r.name == repo_name).unwrap();
370                let sink = &*sinks[&repo_name];
371                let runner = &*runners[&repo_name];
372                run_job(&cfg, repo, sink, runner, &shared, &child, &sha);
373                let mut g = shared.lock().unwrap();
374                if let Some(q) = g.queues.get_mut(&repo_name) {
375                    q.running = None;
376                }
377                persist_queue(&g, &qfile);
378            }
379        });
380    }
381
382    let server = tiny_http::Server::http(&cfg.trigger.listen)
383        .expect("failed to bind listener");
384    println!("bupha watch listening on {}", cfg.trigger.listen);
385
386    for mut req in server.incoming_requests() {
387        let headers: Vec<(String, String)> = req.headers().iter()
388            .map(|h| (h.field.as_str().to_string(),
389                      h.value.as_str().to_string()))
390            .collect();
391        // Only the hook path exists — everything else 404s.
392        if req.url().split('?').next() != Some("/hook") {
393            let _ = req.respond(tiny_http::Response::from_string(
394                "not found").with_status_code(404));
395            continue;
396        }
397
398        let mut body = Vec::new();
399        let _ = req.as_reader().take(2 * 1024 * 1024)
400            .read_to_end(&mut body);
401
402        let is_ping = headers.iter().any(|(k, v)| {
403            k.eq_ignore_ascii_case("x-github-event") && v == "ping"
404        });
405        let (code, msg) = if is_ping {
406            (200, "pong")
407        } else if !trigger.is_event(&headers) {
408            (400, "not a push event")
409        } else if !trigger.verify(&body, &headers) {
410            (401, "bad signature")
411        } else if trigger.delivery_id(&headers).map(|id| {
412            !shared.lock().unwrap().seen.insert(id)
413        }).unwrap_or(false) {   // replayed signed payload: same delivery id
414            (200, "duplicate delivery")
415        } else {
416            match trigger.parse_push(&body) {
417                Some(PushOutcome::Skip) => (200, "ignored: no-op event"),
418                Some(PushOutcome::Push { repo: repo_name, git_ref, sha }) => {
419                    let repo = watched.iter()
420                        .find(|r| r.name == repo_name);
421                    match repo {
422                        None => (200, "ignored: unwatched repo"),
423                        Some(r) if git_ref != r.branch =>
424                            (200, "ignored: not the watched branch"),
425                        Some(r) => {
426                            // Lock: mutate queue + decide, then drop.
427                            // HTTP posts and process kills happen after.
428                            let mut old = None;
429                            let mut kill = false;
430                            let (code, msg) = {
431                                let mut g = shared.lock().unwrap();
432                                let (res, running) = {
433                                    let q = g.queues
434                                        .get_mut(&repo_name).unwrap();
435                                    (q.enqueue(sha.clone()),
436                                     q.running.is_some())
437                                };
438                                persist_queue(&g, &qfile);
439                                match res {
440                                    EnqueueResult::Duplicate =>
441                                        (200, "already queued"),
442                                    EnqueueResult::Queued { superseded } => {
443                                        old = superseded;
444                                        kill = r.runner.cancel_running
445                                            && running;
446                                        if kill {
447                                            g.queues.get_mut(&repo_name)
448                                                .unwrap().cancelled = true;
449                                        }
450                                        (202, "queued")
451                                    }
452                                }
453                            };
454                            if let Some(old) = old {
455                                sinks[&repo_name].post(&old, "error",
456                                    &format!("superseded by {}",
457                                        &sha[..12.min(sha.len())]));
458                                println!("superseded {repo_name} {old:.12} -> {:.12}", sha);
459                            }
460                            if kill {
461                                if let Some(c) =
462                                    child.lock().unwrap().as_mut()
463                                {
464                                    let _ = c.kill();
465                                    println!("cancelled in-flight run");
466                                }
467                            }
468                            if code == 202 { cv.notify_one(); }
469                            (code, msg)
470                        }
471                    }
472                }
473                None => (400, "unparseable payload"),
474            }
475        };
476        let _ = req.respond(tiny_http::Response::from_string(msg)
477            .with_status_code(code));
478    }
479}

481fn cmd_init(dir: &Path) {
482    let dagger_json = dir.join("dagger.json");
483    if !dagger_json.exists() {
484        match Command::new("dagger")
485            .args(["init", "--sdk=python", "--name=ci"])
486            .current_dir(dir).status()
487        {
488            Ok(s) if s.success() => {}
489            other => {
490                eprintln!(
491                    "dagger init failed: {other:?} — is dagger installed?");
492                std::process::exit(1);
493            }
494        }
495    }
496    // Pinned to a sha, not @main — reproducible installs. Bump to the
497    // release tag once v0.1.0 exists.
498    match Command::new("dagger")
499        .args(["install",
500               &format!("github.com/9aia/bupha/modules/dagger@{MODULE_SHA}")])
501        .current_dir(dir).status()
502    {
503        Ok(s) if s.success() => {
504            println!("installed github.com/9aia/bupha/modules/dagger@{MODULE_SHA}")
505        }
506        _ => eprintln!("warning: dagger install failed — check network/auth"),
507    }
508    // Delegate template — thin module composing shared primitives.
509    // The repo name is just the mount point under /work; the dir name
510    // is a sensible default.
511    let main_py = dir.join("dagger/src/ci/main.py");
512    if !main_py.exists() {
513        if let Some(parent) = main_py.parent() {
514            let _ = std::fs::create_dir_all(parent);
515        }
516        let repo_name = dir.canonicalize().ok()
517            .and_then(|d| d.file_name().map(|n| n.to_string_lossy()
518                .replace([' ', '/', '\\'], "-")))
519            .unwrap_or_else(|| "repo".into());
520        // patch URL for self-patch: prefer the git remote, else the
521        // module derives it from Cargo.toml's `repository` field.
522        let patch_host = Command::new("git")
523            .args(["-C"]).arg(dir)
524            .args(["remote", "get-url", "origin"]).output().ok()
525            .filter(|o| o.status.success())
526            .map(|o| {
527                let u = String::from_utf8_lossy(&o.stdout)
528                    .trim().to_string();
529                let u = u.strip_suffix(".git")
530                    .map(|s| s.to_string()).unwrap_or(u);
531                if let Some(rest) = u.strip_prefix("git@") {
532                    rest.replacen(':', "/", 1)
533                        .replacen("", "https://", 1)
534                } else { u }
535            }).unwrap_or_default();
536        std::fs::write(&main_py,
537            DELEGATE_TEMPLATE.replace("REPO_NAME", &repo_name)
538                .replace("PATCH_HOST", &patch_host))
539            .unwrap_or_else(|e| eprintln!("template: {e}"));
540        println!("wrote {}", main_py.display());
541    }
542    println!("next: `dagger call ci --token env://BUPHA_TOKEN` to run the gate");
543}

575fn check(name: &str, ok: bool, detail: &str, failures: &mut usize) {
576    let mark = if ok { "ok " } else { "FAIL" };
577    if !ok { *failures += 1; }
578    println!("  [{mark}] {name}: {detail}");
579}

581fn cmd_doctor(cfg: &Config) {
582    let mut failures = 0usize;
583    println!("bupha doctor");
584
585    check("git", Command::new("git").arg("--version")
586        .output().map(|o| o.status.success()).unwrap_or(false),
587        "on PATH", &mut failures);
588    match cfg.watched() {
589        Ok(ws) => for r in &ws {
590            check(&format!("repo {}", r.name), r.dir.is_dir(),
591                &r.dir.display().to_string(), &mut failures);
592        },
593        Err(e) => check("repos", false, &e, &mut failures),
594    }
595
596    match cfg.runner.kind.as_str() {
597        "dagger" => {
598            let v = Command::new("dagger").arg("version").output();
599            check("dagger cli", v.as_ref().map(|o| o.status.success())
600                .unwrap_or(false),
601                &v.map(|o| String::from_utf8_lossy(&o.stdout)
602                    .trim().to_string()).unwrap_or_default(),
603                &mut failures);
604            check("docker", Command::new("docker").arg("version")
605                .output().map(|o| o.status.success()).unwrap_or(false),
606                "daemon reachable (dagger runner needs it)", &mut failures);
607        }
608        "script" => {
609            let cmd = cfg.runner.cmd.as_deref().unwrap_or("./ci.sh");
610            check("run cmd", true, cmd, &mut failures);
611        }
612        other => check("runner kind", false,
613            &format!("unknown '{other}'"), &mut failures),
614    }
615
616    // env or ~/.config/bupha/env — the unit loads the file, so both
617    // count as set (same fallback deploy check uses).
618    let env_file = std::fs::read_to_string(
619        std::env::var("HOME").map(PathBuf::from)
620            .unwrap_or_else(|_| ".".into())
621            .join(".config/bupha/env")).unwrap_or_default();
622    let has = |name: &str| secret(name).is_some()
623        || env_file.lines().any(|l| l.starts_with(&format!("{name}="))
624            && l.len() > name.len() + 1);
625    check("webhook secret", has(&cfg.trigger.secret_env),
626        &cfg.trigger.secret_env, &mut failures);
627    check("token", has(&cfg.runner.token_env),
628        &cfg.runner.token_env, &mut failures);
629
630    if cfg.status.kind == "github"
631        && cfg.repos.is_empty() && cfg.repo.is_some() {
632        check("status.repo", !cfg.status.repo.is_empty(),
633            "owner/name for commit statuses", &mut failures);
634    }
635    if let Some(conf) = &cfg.runner.sccache_conf {
636        check("sccache conf", Path::new(conf).exists(),
637            conf, &mut failures);
638    }
639    match cfg.tunnel.provider.as_str() {
640        "cloudflared" => check("cloudflared",
641            Command::new("cloudflared").arg("--version")
642                .output().map(|o| o.status.success()).unwrap_or(false),
643            "on PATH", &mut failures),
644        "none" | "direct" => check("tunnel", true,
645            &format!("provider '{}'", cfg.tunnel.provider), &mut failures),
646        other => check("tunnel", false,
647            &format!("unknown provider '{other}'"), &mut failures),
648    }
649
650    if let Ok(o) = Command::new("df").args(["-m", &cfg.worktree.base])
651        .output()
652    {
653        let out = String::from_utf8_lossy(&o.stdout);
654        if let Some(free) = out.lines().nth(1)
655            .and_then(|l| l.split_whitespace().nth(3))
656            .and_then(|n| n.parse::<u64>().ok())
657        {
658            check("disk", free > 10_000,
659                &format!("{free} MB free under {}", cfg.worktree.base),
660                &mut failures);
661        }
662    }
663
664    if failures == 0 {
665        println!("all green");
666    } else {
667        println!("{failures} failure(s)");
668        std::process::exit(1);
669    }
670}

672fn cmd_run(cfg: &Config, function: Option<String>) {
673    let mut repo = cfg.watched().ok().and_then(|w| w.into_iter().next())
674        .unwrap_or_else(|| {
675            eprintln!("no repo configured");
676            std::process::exit(2);
677        });
678    if let Some(f) = function {
679        repo.runner.function = f;
680    }
681    let runner = build_runner(&repo.runner);
682    let ctx = RunCtx {
683        workdir: &repo.dir,
684        token_env: &repo.runner.token_env,
685    };
686    match runner.spawn(&ctx).and_then(|mut c| c.wait()) {
687        Ok(s) if s.success() => println!("run passed"),
688        Ok(s) => {
689            eprintln!("run failed: {s}");
690            std::process::exit(s.code().unwrap_or(1));
691        }
692        Err(e) => {
693            eprintln!("spawn failed: {e}");
694            std::process::exit(1);
695        }
696    }
697}

699fn main() {
700    let cli = Cli::parse();
701    match &cli.cmd {
702        Cmd::Init { dir } => cmd_init(dir),
703        // Release is repo-local (cwd) — no bupha.toml needed.
704        Cmd::Deploy { sub: DeploySub::Release { dry_run } } =>
705            cmd_deploy_release(*dry_run),
706        _ => {
707            let cfg = Arc::new(Config::load(cli.config.as_deref())
708                .unwrap_or_else(|e| {
709                    eprintln!("config: {e}");
710                    std::process::exit(2);
711                }));
712            match &cli.cmd {
713                Cmd::Watch => cmd_watch(cfg),
714                Cmd::Doctor => cmd_doctor(&cfg),
715                Cmd::Run { function } => cmd_run(&cfg, function.clone()),
716                Cmd::Deploy { sub } => cmd_deploy(&cfg, sub, cli.config.clone()),
717                Cmd::Destroy { sub } => cmd_destroy(&cfg, sub),
718                Cmd::Init { .. } => unreachable!(),
719            }
720        }
721    }
722}

724fn cmd_deploy(cfg: &Config, sub: &DeploySub, cli_config: Option<String>) {
725    let cli_config = cli_config;
726    let home = std::env::var("HOME").map(PathBuf::from)
727        .unwrap_or_else(|_| ".".into());
728    let cfg_dir = home.join(".config/bupha");
729    match sub {
730        DeploySub::Env => {
731            match deploy::write_env_skeleton(&cfg_dir) {
732                Ok(p) => println!("env file: {} (fill the secrets — or set them via secret-tool)", p.display()),
733                Err(e) => { eprintln!("env: {e}"); std::process::exit(1); }
734            }
735            // If values are already in the environment, migrate them
736            // into the OS keyring (resolution order: env > keyring > file).
737            for name in [&cfg.trigger.secret_env, &cfg.runner.token_env] {
738                if let Ok(v) = std::env::var(name) {
739                    if v.is_empty() { continue; }
740                    let ok = Command::new("secret-tool")
741                        .args(["store", "--label", &format!("bupha: {name}"),
742                               "service", "bupha", "name", name])
743                        .stdin(std::process::Stdio::piped())
744                        .spawn()
745                        .and_then(|mut c| {
746                            use std::io::Write;
747                            c.stdin.as_mut().unwrap().write_all(v.as_bytes())?;
748                            c.wait()
749                        }).map(|s| s.success()).unwrap_or(false);
750                    println!("{name}: {}", if ok { "stored in keyring" }
751                        else { "keyring unavailable (env file is the fallback)" });
752                }
753            }
754        }
755        DeploySub::Service { no_start } => {
756            #[cfg(not(target_os = "linux"))]
757            {
758                eprintln!("deploy service requires systemd (Linux). \
759                    bupha supports Linux only for now — macOS/Windows \
760                    may follow.");
761                std::process::exit(1);
762            }
763            let bin = std::env::current_exe().unwrap_or_else(
764                |_| PathBuf::from("bupha"));
765            let cfg_path = cli_config.clone().map(PathBuf::from);
766            match deploy::write_service_unit(
767                &bin, &cfg_dir.join("env"), cfg_path.as_deref()) {
768                Ok(p) => println!("unit: {}", p.display()),
769                Err(e) => { eprintln!("unit: {e}"); std::process::exit(1); }
770            }
771            if !no_start {
772                if let Err(e) = deploy::enable_service("bupha-watch.service") {
773                    eprintln!("enable: {e}");
774                    std::process::exit(1);
775                }
776                println!("bupha-watch enabled — run 'loginctl enable-linger' once for boot persistence");
777            }
778        }
779        DeploySub::Check => deploy_check(cfg),
780        DeploySub::Webhook => {
781            let wh_secret = secret(&cfg.trigger.secret_env)
782                .unwrap_or_else(|| {
783                    eprintln!("{} required (env, keyring or env file)", cfg.trigger.secret_env);
784                    std::process::exit(1);
785                });
786            let url = cfg.tunnel.public_url.clone();
787            if url.is_empty() {
788                eprintln!("tunnel.public_url not set — point the webhook                            at <public-url>/hook manually");
789                std::process::exit(1);
790            }
791            let watched = cfg.watched().unwrap_or_else(|e| {
792                eprintln!("{e}");
793                std::process::exit(1);
794            });
795            let token = secret(&cfg.runner.token_env)
796                .unwrap_or_default();
797            for r in &watched {
798                match bupha_github::register_webhook(
799                    &token, &r.name, &url, &wh_secret)
800                {
801                    Ok(msg) => println!("{msg}"),
802                    Err(e) => { eprintln!("webhook {}: {e}", r.name);
803                                std::process::exit(1); }
804                }
805            }
806        }
807        DeploySub::Release { .. } => unreachable!(),
808    }
809}

811/// Seed the release flow: publish every workspace member in dep
812/// order, then tag `{repo}-v{version}` and push the tag. Explicit —
813/// this is the only irreversible step in bupha's lifecycle.
814fn cmd_deploy_release(dry_run: bool) {
815    let repo_dir = std::env::current_dir()
816        .unwrap_or_else(|_| PathBuf::from("."));
817    if std::env::var("CARGO_REGISTRY_TOKEN").unwrap_or_default()
818        .is_empty() && !dry_run
819    {
820        eprintln!("CARGO_REGISTRY_TOKEN required");
821        std::process::exit(1);
822    }
823    let out = Command::new("cargo")
824        .args(["metadata", "--format-version", "1", "--no-deps"])
825        .current_dir(&repo_dir).output()
826        .unwrap_or_else(|e| { eprintln!("cargo metadata: {e}");
827                              std::process::exit(1) });
828    let meta: serde_json::Value = serde_json::from_slice(&out.stdout)
829        .unwrap_or_else(|e| { eprintln!("metadata json: {e}");
830                              std::process::exit(1) });
831    let empty: Vec<serde_json::Value> = vec![];
832    let member_ids: std::collections::HashSet<String> =
833        meta["workspace_members"].as_array().unwrap_or(&empty)
834            .iter().filter_map(|m| m.as_str().map(String::from))
835            .collect();
836    let pkgs: Vec<&serde_json::Value> =
837        meta["packages"].as_array().unwrap_or(&empty).iter()
838            .filter(|p| member_ids.contains(
839                p["id"].as_str().unwrap_or("")))
840            .collect();
841    let members: Vec<String> = pkgs.iter()
842        .filter_map(|p| p["name"].as_str().map(String::from)).collect();
843    let deps: Vec<(String, String)> = pkgs.iter().flat_map(|p| {
844        let a = p["name"].as_str().unwrap_or("").to_string();
845        p["dependencies"].as_array().unwrap_or(&empty).iter()
846            .filter(|d| d["path"].is_string())
847            .filter_map(|d| d["name"].as_str()
848                .map(|n| (a.clone(), n.to_string())))
849            .collect::<Vec<_>>()
850    }).collect();
851    let order = deploy::publish_order(&members, &deps)
852        .unwrap_or_else(|e| { eprintln!("{e}"); std::process::exit(1) });
853    let versions: std::collections::HashSet<String> = pkgs.iter()
854        .filter_map(|p| p["version"].as_str().map(String::from))
855        .collect();
856    if versions.len() != 1 {
857        eprintln!("members have mixed versions {versions:?} — \
858                   release needs one version across the workspace");
859        std::process::exit(1);
860    }
861    let ver = versions.into_iter().next().unwrap();
862    let tag = format!("{}-v{ver}", repo_dir.file_name()
863        .map(|n| n.to_string_lossy().to_string())
864        .unwrap_or_else(|| "release".into()));
865    if dry_run {
866        println!("publish order: {}", order.join(" -> "));
867        println!("tag: {tag}");
868        return;
869    }
870    for name in &order {
871        let mut ok = false;
872        for _ in 0..5 {
873            let st = Command::new("cargo")
874                .args(["publish", "-p", name, "--allow-dirty"])
875                .current_dir(&repo_dir).output();
876            let text = st.map(|o| format!("{}{}",
877                String::from_utf8_lossy(&o.stdout),
878                String::from_utf8_lossy(&o.stderr)))
879                .unwrap_or_default();
880            // "Finished" also appears for a successful package build
881            // whose upload then fails — only the upload counts.
882            if text.contains("Uploaded") || text.contains("already") {
883                println!("published {name}");
884                ok = true;
885                break;
886            }
887            std::thread::sleep(std::time::Duration::from_secs(20));
888        }
889        if !ok {
890            eprintln!("publish failed: {name}");
891            std::process::exit(1);
892        }
893    }
894    if !git(&repo_dir, &["rev-parse", "-q", "--verify",
895                         &format!("refs/tags/{tag}")]) {
896        if !git(&repo_dir, &["tag", &tag]) {
897            eprintln!("git tag {tag} failed");
898            std::process::exit(1);
899        }
900    }
901    if !git(&repo_dir, &["push", "origin", &tag]) {
902        eprintln!("tag push failed — publish done, push {tag} by hand");
903        std::process::exit(1);
904    }
905    println!("released {tag} — pushes to main now release via \
906              the watch loop");
907}

909fn cmd_destroy(cfg: &Config, sub: &DestroySub) {
910    match sub {
911        DestroySub::Service => {
912            match deploy::disable_service("bupha-watch.service") {
913                Ok(p) => println!("removed {}", p.display()),
914                Err(e) => { eprintln!("service: {e}");
915                            std::process::exit(1); }
916            }
917        }
918        DestroySub::Webhook => {
919            if cfg.tunnel.public_url.is_empty() {
920                eprintln!("tunnel.public_url not set — nothing to \
921                           match hooks against");
922                std::process::exit(1);
923            }
924            let watched = cfg.watched().unwrap_or_else(|e| {
925                eprintln!("{e}");
926                std::process::exit(1);
927            });
928            let token = secret(&cfg.runner.token_env)
929                .unwrap_or_default();
930            for r in &watched {
931                match bupha_github::unregister_webhook(
932                    &token, &r.name, &cfg.tunnel.public_url)
933                {
934                    Ok(msg) => println!("{msg}"),
935                    Err(e) => { eprintln!("webhook {}: {e}", r.name);
936                                std::process::exit(1); }
937                }
938            }
939        }
940        DestroySub::Worktrees => {
941            // worktrees land at <base>-<repo>-<sha12>; queue.json in base
942            let base = PathBuf::from(&cfg.worktree.base);
943            let prefix = base.file_name()
944                .map(|n| format!("{}-", n.to_string_lossy()))
945                .unwrap_or_default();
946            if let Some(parent) = base.parent() {
947                if let Ok(rd) = std::fs::read_dir(parent) {
948                    for e in rd.flatten() {
949                        // .git must be a FILE (worktree pointer), not a
950                        // dir — protects foreign dirs sharing the prefix.
951                        if e.file_name().to_string_lossy()
952                                .starts_with(&prefix)
953                            && e.path().is_dir()
954                            && e.path().join(".git").is_file()
955                        {
956                            let _ = std::fs::remove_dir_all(e.path());
957                            println!("removed {}", e.path().display());
958                        }
959                    }
960                }
961            }
962            let q = base.join("queue.json");
963            if q.exists() {
964                let _ = std::fs::remove_file(&q);
965                println!("removed {}", q.display());
966            }
967            if let Ok(ws) = cfg.watched() {
968                for r in &ws {
969                    git(&r.dir, &["worktree", "prune"]);
970                }
971            }
972        }
973    }
974}

976fn deploy_check(cfg: &Config) {
977    let mut fails = 0usize;
978    let home = std::env::var("HOME").map(PathBuf::from)
979        .unwrap_or_else(|_| ".".into());
980    check("unit file", home.join(
981            ".config/systemd/user/bupha-watch.service").exists(),
982        "bupha-watch.service", &mut fails);
983    let env_ok = home.join(".config/bupha/env").exists();
984    check("env file", env_ok, "~/.config/bupha/env", &mut fails);
985    let env_file = std::fs::read_to_string(
986            home.join(".config/bupha/env")).unwrap_or_default();
987    let secret_ok = secret(&cfg.trigger.secret_env).is_some()
988        || env_file.lines().any(|l| l.starts_with(
989            &format!("{}=", cfg.trigger.secret_env))
990            && l.len() > cfg.trigger.secret_env.len() + 1);
991    check("webhook secret set", secret_ok,
992        &format!("{} (env or env file)", cfg.trigger.secret_env), &mut fails);
993    let token_ok = secret(&cfg.runner.token_env).is_some()
994        || env_file.lines().any(|l| l.starts_with(
995            &format!("{}=", cfg.runner.token_env))
996            && l.len() > cfg.runner.token_env.len() + 1);
997    check("token set", token_ok,
998        &format!("{} (env or env file)", cfg.runner.token_env), &mut fails);
999    // public_url is only required to run `deploy webhook`; its absence
1000    // is a note, not a failure.
1001    if cfg.tunnel.public_url.is_empty() {
1002        check("tunnel.public_url", true,
1003            "unset (needed only for webhook registration)", &mut fails);
1004    } else {
1005        let code = ureq_probe(&cfg.tunnel.public_url);
1006        check("endpoint reachable", code.is_some(),
1007            &format!("{} -> {:?}", cfg.tunnel.public_url, code), &mut fails);
1008        if cfg.tunnel.public_url.contains(".trycloudflare.com") {
1009            println!("  note: trycloudflare quick tunnels are flaky — \
1010                prefer a named tunnel");
1011        }
1012    }
1013    if fails == 0 { println!("deploy check: all green"); }
1014    else { println!("{fails} problem(s)"); std::process::exit(1); }
1015}

1017fn ureq_probe(url: &str) -> Option<u16> {
1018    // Any HTTP answer (even a 4xx) proves the endpoint is reachable;
1019    // only transport-level failure is None.
1020    let agent = ureq::Agent::config_builder()
1021        .timeout_global(Some(std::time::Duration::from_secs(10)))
1022        .build().new_agent();
1023    match agent.post(&format!("{}/hook", url.trim_end_matches('/')))
1024        .header("User-Agent", "bupha")
1025        .send_empty()
1026    {
1027        Ok(r) => Some(r.status().as_u16()),
1028        Err(ureq::Error::StatusCode(c)) => Some(c),
1029        Err(_) => None,
1030    }
