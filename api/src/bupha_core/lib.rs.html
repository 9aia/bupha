1//! bupha-core — domain types, `bupha.{yaml,yml,toml}` config, and the latest-wins
2//! queue every runner impl shares.
3//!
4//! The queue is the batch semantic for fast-pushing repos: at most one
5//! run in flight plus one pending sha — newer pushes supersede the
6//! pending one (skipped shas should get a "superseded" status from the
7//! caller). N commits during a run = 1 follow-up run on the newest.
8

13/// A revision is path- and argv-safe — rejects path-injection payloads
14/// before the value reaches worktree paths or status URLs, and
15/// option-injection payloads before it reaches a `git` argv position
16/// (a leading `-` would parse as a flag, not a revision). Deliberately
17/// not hex-only: the shape comes from the forge/VCS, not the domain.
18pub fn is_valid_sha(sha: &str) -> bool {
19    (7..=64).contains(&sha.len())
20        && !sha.starts_with('-')
21        && !sha.contains("..")
22        && !sha.contains('/')
23        && sha
24            .bytes()
25            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_' | b'.'))
26}

28/// A job key ("owner/name" or "owner/name:job") is path- and
29/// shell-safe — it lands in worktree paths, remote run dirs, claim
30/// keys and status contexts. `..` is rejected even though `/` is a
31/// legal segment separator.
32pub fn is_valid_key(key: &str) -> bool {
33    !key.is_empty()
34        && !key.contains("..")
35        && key
36            .bytes()
37            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_' | b'.' | b'/' | b':'))
38}

40/// An env-var NAME — the shape remote run scripts need for
41/// `export K=V` lines. `[A-Za-z_][A-Za-z0-9_]*`.
42pub fn is_valid_env_name(s: &str) -> bool {
43    !s.is_empty()
44        && !s.bytes().next().is_some_and(|b| b.is_ascii_digit())
45        && s.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'_')
46}

48/// How a repo's pending work collapses: `latest` keeps one pending
49/// slot (newest wins — CI gates), `all` runs every push in order
50/// (FIFO, bounded). Per-repo via `[[repo]] queue = "..."`.
51#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize, Default)]
52#[serde(rename_all = "lowercase")]
53pub enum QueuePolicy {
54    #[default]
55    Latest,
56    All,
57}
58
59/// Backlog cap for `all` — overflow evicts the oldest pending sha.
60pub const MAX_PENDING: usize = 32;
61
62/// Bounded dedup set for delivery ids — a replayed webhook reuses the
63/// same id, so `insert` returning false means "already handled".
64/// Oldest ids are evicted past `cap`; memory-only (replays matter
65/// within process uptime).
66pub struct Seen {
67    ids: std::collections::HashSet<String>,
68    order: VecDeque<String>,
69    cap: usize,
70}
71
72impl Seen {
73    pub fn new(cap: usize) -> Self {
74        Seen {
75            ids: Default::default(),
76            order: VecDeque::new(),
77            cap,
78        }
79    }
80    /// true = fresh, false = replay/duplicate.
81    pub fn insert(&mut self, id: String) -> bool {
82        if !self.ids.insert(id.clone()) {
83            return false;
84        }
85        self.order.push_back(id);
86        while self.order.len() > self.cap {
87            if let Some(old) = self.order.pop_front() {
88                self.ids.remove(&old);
89            }
90        }
91        true
92    }
93}
94
95impl Default for Seen {
96    fn default() -> Self {
97        Self::new(10_000)
98    }
99}

101/// Queue state. The running child handle lives in the runner/CLI layer
102/// (`ChildSlot`) so enqueue can kill it without blocking on this lock.
103#[derive(Default)]
104pub struct Queue {
105    pub pending: VecDeque<String>,
106    pub running: Option<String>,
107    pub cancelled: bool,
108    pub policy: QueuePolicy,
109}
110
111impl Queue {
112    pub fn with_policy(policy: QueuePolicy) -> Self {
113        Queue {
114            policy,
115            ..Default::default()
116        }
117    }
118
119    /// `latest`: replace the pending sha (old is returned for a
120    /// superseded status). `all`: append; a full backlog evicts the
121    /// oldest, also returned as superseded. `Duplicate` = deduped.
122    pub fn enqueue(&mut self, sha: String) -> EnqueueResult {
123        if self.running.as_deref() == Some(sha.as_str()) || self.pending.iter().any(|s| s == &sha) {
124            return EnqueueResult::Duplicate;
125        }
126        let superseded = match self.policy {
127            QueuePolicy::Latest => self.pending.pop_front(),
128            QueuePolicy::All if self.pending.len() >= MAX_PENDING => self.pending.pop_front(),
129            QueuePolicy::All => None,
130        };
131        self.pending.push_back(sha);
132        EnqueueResult::Queued { superseded }
133    }
134
135    /// Next sha to run, FIFO.
136    pub fn dequeue(&mut self) -> Option<String> {
137        self.pending.pop_front()
138    }
139
140    /// Re-append a sha that just failed — the retry path (ADR-0002).
141    /// Unlike `enqueue` it never supersedes the front: the retry goes
142    /// behind already-queued work under both policies.
143    pub fn requeue(&mut self, sha: &str) {
144        if !self.pending.iter().any(|s| s == sha) {
145            self.pending.push_back(sha.to_string());
146        }
147    }
148}
149
150pub enum EnqueueResult {
151    Queued { superseded: Option<String> },
152    Duplicate,
153}

155/// Expand a leading `~` (alone or `~/…`) to `$HOME` — applied to
156/// local-path config fields by [`Config::expand_tilde`] so `dir:
157/// ~/GitHub/x` works in every source format. `~user` and anything
158/// else passes through untouched, as does `~` when `$HOME` is unset.
159pub fn expand_tilde(p: &str) -> String {
160    let Some(rest) = p.strip_prefix('~') else {
161        return p.into();
162    };
163    if !rest.is_empty() && !rest.starts_with('/') {
164        return p.into();
165    }
166    match std::env::var("HOME") {
167        Ok(h) if !h.is_empty() => format!("{h}{rest}"),
168        _ => p.into(),
169    }
170}
171
172/// Config fields that are local paths — expanded in place by
173/// [`Config::expand_tilde`]. Impl-owned extras are whitelisted by
174/// key (`sccache_conf`, `cli_source`): remote-path extras like the
175/// ssh runner's `remote_dir` must keep a literal `~` for the remote
176/// shell, and command strings expand `~` themselves.
177fn expand_extra(extra: &mut toml::Table) {
178    for key in ["sccache_conf", "cli_source"] {
179        if let Some(v) = extra.get_mut(key) {
180            if let Some(s) = v.as_str() {
181                *v = expand_tilde(s).into();
182            }
183        }
184    }
185}

582impl Config {
583    /// Post-load normalization: expand a leading `~` in every
584    /// local-path field (`repo.dir`, `worktree.base`, `audit.path`,
585    /// `telemetry.path`, `attest.dir`, `cache.conf`, plus the
586    /// `sccache_conf`/`cli_source` extras at every runner level).
587    /// Sources call this after parsing so TOML and YAML configs get
588    /// identical semantics.
589    pub fn expand_tilde(&mut self) {
590        fn runner(r: &mut RunnerCfg) {
591            expand_extra(&mut r.extra);
592        }
593        for r in &mut self.repos {
594            let dir = expand_tilde(&r.dir.to_string_lossy());
595            r.dir = dir.into();
596            if let Some(rr) = &mut r.runner {
597                runner(rr);
598            }
599            for j in &mut r.jobs {
600                if let Some(jr) = &mut j.runner {
601                    runner(jr);
602                }
603            }
604        }
605        runner(&mut self.runner);
606        self.worktree.base = expand_tilde(&self.worktree.base);
607        self.audit.path = expand_tilde(&self.audit.path);
608        if let Some(p) = &mut self.telemetry.path {
609            *p = expand_tilde(p);
610        }
611        if let Some(d) = &mut self.attest.dir {
612            *d = expand_tilde(d);
613        }
614        if let Some(c) = &mut self.cache.conf {
615            *c = expand_tilde(c);
616        }
617    }
618
619    /// Expand `[[repo]]` entries into one WatchedRepo per job —
620    /// a repo without `[[repo.jobs]]` yields its single implicit
621    /// pipeline (job = "", key = name).
622    pub fn watched(&self) -> Result<Vec<WatchedRepo>, String> {
623        let canon = |p: &str| std::fs::canonicalize(p).unwrap_or_else(|_| p.into());
624        if self.repos.is_empty() {
625            return Err("no [[repo]] entries configured".into());
626        }
627        // Two [[repo]] tables with the same `name` used to shadow
628        // silently in the name-keyed maps — that shape is what
629        // [[repo.jobs]] is for, so reject it loudly now.
630        for (i, r) in self.repos.iter().enumerate() {
631            if self.repos[..i].iter().any(|o| o.name == r.name) {
632                return Err(format!(
633                    "repo {}: declared more than once — merge \
634                     pipelines into one [[repo]] with [[repo.jobs]] \
635                     entries",
636                    r.name
637                ));
638            }
639        }
640        let mut out = Vec::new();
641        for r in &self.repos {
642            // `name`/`status_repo` land in worktree paths, remote run
643            // dirs and the statuses API URL — path/argv/shell-safe
644            // only (no `:` — that's the job separator). Slash is NOT
645            // required: sturn-only repos use bare names.
646            for (field, v) in [
647                ("name", Some(r.name.as_str())),
648                ("status_repo", r.status_repo.as_deref()),
649            ]
650            .into_iter()
651            .filter_map(|(f, v)| v.map(|v| (f, v)))
652            {
653                if !is_valid_key(v) || v.contains(':') {
654                    return Err(format!(
655                        "repo {field} {v:?} invalid — \
656                         [A-Za-z0-9._/-]+, no \"..\" or \":\""
657                    ));
658                }
659            }
660            if r.jobs.is_empty() {
661                out.push(self.watch_one(r, None, &canon)?);
662                continue;
663            }
664            let mut seen: std::collections::HashSet<&str> = Default::default();
665            for j in &r.jobs {
666                if !is_valid_job_name(&j.name) {
667                    return Err(format!(
668                        "repo {}: invalid job name {:?} — use \
669                         [A-Za-z0-9._-]+ (lands in paths, claim keys \
670                         and status contexts)",
671                        r.name, j.name
672                    ));
673                }
674                if !seen.insert(j.name.as_str()) {
675                    return Err(format!("repo {}: duplicate job name {:?}", r.name, j.name));
676                }
677                out.push(self.watch_one(r, Some(j), &canon)?);
678            }
679        }
680        Ok(out)
681    }
682
683    /// Resolve one repo(+job) pair — every field inherits
684    /// job → repo → global.
685    fn watch_one(
686        &self,
687        r: &RepoEntry,
688        job: Option<&JobEntry>,
689        canon: &dyn Fn(&str) -> PathBuf,
690    ) -> Result<WatchedRepo, String> {
691        let runner = job
692            .and_then(|j| j.runner.clone())
693            .or_else(|| r.runner.clone())
694            .unwrap_or_else(|| self.runner.clone());
695        let policy = job.and_then(|j| j.queue).unwrap_or(r.queue);
696        // `all` keeps every push — killing the in-flight run
697        // would drop a commit that was never superseded.
698        if policy == QueuePolicy::All && runner.cancel_running {
699            return Err(format!(
700                "repo {}: queue = \"all\" conflicts with \
701                 runner.cancel_running{}",
702                r.name,
703                job.map(|j| format!(" (job {:?})", j.name))
704                    .unwrap_or_default()
705            ));
706        }
707        Ok(WatchedRepo {
708            name: r.name.clone(),
709            job: job.map(|j| j.name.clone()).unwrap_or_default(),
710            key: job
711                .map(|j| format!("{}:{}", r.name, j.name))
712                .unwrap_or_else(|| r.name.clone()),
713            dir: canon(&r.dir.to_string_lossy()),
714            branch: job
715                .and_then(|j| j.branch.clone())
716                .unwrap_or_else(|| r.branch.clone()),
717            status_repo: r.status_repo().to_string(),
718            context: job
719                .and_then(|j| j.context.clone())
720                .or_else(|| job.map(|j| format!("{}/{}", self.status.context, j.name)))
721                .unwrap_or_else(|| self.status.context.clone()),
722            runner,
723            policy,
724            secret_env: r.secret_env.clone(),
725            runs_on: job
726                .and_then(|j| j.runs_on.clone())
727                .unwrap_or_else(|| r.runs_on.clone()),
728            retry: {
729                let n = job.and_then(|j| j.retry).unwrap_or(r.retry);
730                if n > 3 {
731                    return Err(format!("repo {}: retry = {n} — max 3 (ADR-0002)", r.name));
732                }
733                n
734            },
735            sturn: r.sturn,
736        })
737    }
738}

910/// Publish order for workspace members: members sorted so every
911/// member's path-deps on other members come first. Returns an error
912/// name on a dependency cycle.
913pub fn publish_order(members: &[String], deps: &[(String, String)]) -> Result<Vec<String>, String> {
914    let member_set: std::collections::HashSet<&str> = members.iter().map(String::as_str).collect();
915    let mut done: std::collections::HashSet<String> = Default::default();
916    let mut out = Vec::with_capacity(members.len());
917    while out.len() < members.len() {
918        let mut progressed = false;
919        for m in members {
920            if done.contains(m) {
921                continue;
922            }
923            let ready = deps
924                .iter()
925                .filter(|(a, _)| a == m)
926                .all(|(_, d)| !member_set.contains(d.as_str()) || done.contains(d));
927            if ready {
928                done.insert(m.clone());
929                out.push(m.clone());
930                progressed = true;
931            }
932        }
933        if !progressed {
934            return Err("dependency cycle in workspace".into());
935        }
936    }
937    Ok(out)
938}
