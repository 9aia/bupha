1//! bupha-core — domain types, `bupha.toml` config, and the latest-wins
2//! queue every runner impl shares.
3//!
4//! The queue is the batch semantic for fast-pushing repos: at most one
5//! run in flight plus one pending sha — newer pushes supersede the
6//! pending one (skipped shas should get a "superseded" status from the
7//! caller). N commits during a run = 1 follow-up run on the newest.
8
9use serde::Deserialize;
10use std::collections::VecDeque;
11use std::path::PathBuf;

13/// A revision is path-safe — rejects path-injection payloads before
14/// the value reaches worktree paths or status URLs. Deliberately not
15/// hex-only: the shape comes from the forge/VCS, not the domain.
16pub fn is_valid_sha(sha: &str) -> bool {
17    (7..=64).contains(&sha.len())
18        && !sha.contains("..")
19        && !sha.contains('/')
20        && sha
21            .bytes()
22            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_' | b'.'))
23}

25/// How a repo's pending work collapses: `latest` keeps one pending
26/// slot (newest wins — CI gates), `all` runs every push in order
27/// (FIFO, bounded). Per-repo via `[[repo]] queue = "..."`.
28#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize, Default)]
29#[serde(rename_all = "lowercase")]
30pub enum QueuePolicy {
31    #[default]
32    Latest,
33    All,
34}
35
36/// Backlog cap for `all` — overflow evicts the oldest pending sha.
37pub const MAX_PENDING: usize = 32;
38
39/// Bounded dedup set for delivery ids — a replayed webhook reuses the
40/// same id, so `insert` returning false means "already handled".
41/// Oldest ids are evicted past `cap`; memory-only (replays matter
42/// within process uptime).
43pub struct Seen {
44    ids: std::collections::HashSet<String>,
45    order: VecDeque<String>,
46    cap: usize,
47}
48
49impl Seen {
50    pub fn new(cap: usize) -> Self {
51        Seen {
52            ids: Default::default(),
53            order: VecDeque::new(),
54            cap,
55        }
56    }
57    /// true = fresh, false = replay/duplicate.
58    pub fn insert(&mut self, id: String) -> bool {
59        if !self.ids.insert(id.clone()) {
60            return false;
61        }
62        self.order.push_back(id);
63        while self.order.len() > self.cap {
64            if let Some(old) = self.order.pop_front() {
65                self.ids.remove(&old);
66            }
67        }
68        true
69    }
70}
71
72impl Default for Seen {
73    fn default() -> Self {
74        Self::new(10_000)
75    }
76}

78/// Queue state. The running child handle lives in the runner/CLI layer
79/// (`ChildSlot`) so enqueue can kill it without blocking on this lock.
80#[derive(Default)]
81pub struct Queue {
82    pub pending: VecDeque<String>,
83    pub running: Option<String>,
84    pub cancelled: bool,
85    pub policy: QueuePolicy,
86}
87
88impl Queue {
89    pub fn with_policy(policy: QueuePolicy) -> Self {
90        Queue {
91            policy,
92            ..Default::default()
93        }
94    }
95
96    /// `latest`: replace the pending sha (old is returned for a
97    /// superseded status). `all`: append; a full backlog evicts the
98    /// oldest, also returned as superseded. `Duplicate` = deduped.
99    pub fn enqueue(&mut self, sha: String) -> EnqueueResult {
100        if self.running.as_deref() == Some(sha.as_str()) || self.pending.iter().any(|s| s == &sha) {
101            return EnqueueResult::Duplicate;
102        }
103        let superseded = match self.policy {
104            QueuePolicy::Latest => self.pending.pop_front(),
105            QueuePolicy::All if self.pending.len() >= MAX_PENDING => self.pending.pop_front(),
106            QueuePolicy::All => None,
107        };
108        self.pending.push_back(sha);
109        EnqueueResult::Queued { superseded }
110    }
111
112    /// Next sha to run, FIFO.
113    pub fn dequeue(&mut self) -> Option<String> {
114        self.pending.pop_front()
115    }
116}
117
118pub enum EnqueueResult {
119    Queued { superseded: Option<String> },
120    Duplicate,
121}

199/// Resolved view used by the watcher — always a list, even for the
200/// legacy single-repo form.
201#[derive(Debug, Clone)]
202pub struct WatchedRepo {
203    pub name: String,
204    pub dir: PathBuf,
205    pub branch: String,
206    pub status_repo: String,
207    /// Global `[runner]` or the repo's `[repo.runner]` override.
208    pub runner: RunnerCfg,
209    pub policy: QueuePolicy,
210}
211
212impl Config {
213    pub fn watched(&self) -> Result<Vec<WatchedRepo>, String> {
214        let canon = |p: &str| {
215            std::fs::canonicalize(p).unwrap_or_else(|_| p.into())
216        };
217        if !self.repos.is_empty() {
218            return self
219                .repos
220                .iter()
221                .map(|r| {
222                    let runner = r.runner.clone().unwrap_or_else(|| self.runner.clone());
223                    // `all` keeps every push — killing the in-flight run
224                    // would drop a commit that was never superseded.
225                    if r.queue == QueuePolicy::All && runner.cancel_running {
226                        return Err(format!(
227                            "repo {}: queue = \"all\" conflicts with \
228                         runner.cancel_running",
229                            r.name
230                        ));
231                    }
232                    Ok(WatchedRepo {
233                        name: r.name.clone(),
234                        dir: canon(&r.dir.to_string_lossy()).into(),
235                        branch: r.branch.clone(),
236                        status_repo: r.status_repo().to_string(),
237                        runner,
238                        policy: r.queue,
239                    })
240                })
241                .collect();
242        }
243        Err("no [[repo]] entries configured".into())
244    }
245}

368/// Publish order for workspace members: members sorted so every
369/// member's path-deps on other members come first. Returns an error
370/// name on a dependency cycle.
371pub fn publish_order(members: &[String], deps: &[(String, String)]) -> Result<Vec<String>, String> {
372    let member_set: std::collections::HashSet<&str> = members.iter().map(String::as_str).collect();
373    let mut done: std::collections::HashSet<String> = Default::default();
374    let mut out = Vec::with_capacity(members.len());
375    while out.len() < members.len() {
376        let mut progressed = false;
377        for m in members {
378            if done.contains(m) {
379                continue;
380            }
381            let ready = deps
382                .iter()
383                .filter(|(a, _)| a == m)
384                .all(|(_, d)| !member_set.contains(d.as_str()) || done.contains(d));
385            if ready {
386                done.insert(m.clone());
387                out.push(m.clone());
388                progressed = true;
389            }
390        }
391        if !progressed {
392            return Err("dependency cycle in workspace".into());
393        }
394    }
395    Ok(out)
396}

398#[cfg(test)]
399mod tests {
400    use super::publish_order;
401
402    #[test]
403    fn order_respects_path_deps() {
404        let members = vec!["bupha".into(), "bupha-core".into(), "bupha-github".into()];
405        let deps = vec![
406            ("bupha".into(), "bupha-github".into()),
407            ("bupha".into(), "bupha-core".into()),
408            ("bupha-github".into(), "bupha-core".into()),
409        ];
410        let order = publish_order(&members, &deps).unwrap();
411        let pos = |n: &str| order.iter().position(|x| x == n).unwrap();
412        assert!(pos("bupha-core") < pos("bupha-github"));
413        assert!(pos("bupha-github") < pos("bupha"));
414    }
415
416    #[test]
417    fn cycle_errors() {
418        let members = vec!["a".into(), "b".into()];
419        let deps = vec![("a".into(), "b".into()), ("b".into(), "a".into())];
420        assert!(publish_order(&members, &deps).is_err());
421    }
