1//! bupha-core — domain types, `bupha.toml` config, and the latest-wins
2//! queue every runner impl shares.
3//!
4//! The queue is the batch semantic for fast-pushing repos: at most one
5//! run in flight plus one pending sha — newer pushes supersede the
6//! pending one (skipped shas should get a "superseded" status from the
7//! caller). N commits during a run = 1 follow-up run on the newest.
8
9use serde::Deserialize;
10use std::collections::VecDeque;
11use std::path::PathBuf;
12
13/// Where pushes come from. Implementations parse their own payload
14/// format; the queue only sees shas.
15pub trait Trigger: Send + Sync {
16    /// Is this request the kind of event we run CI on (e.g. a push)?
17    fn is_event(&self, headers: &[(String, String)]) -> bool;
18    fn verify(&self, body: &[u8], headers: &[(String, String)]) -> bool;
19    /// Parse the payload — `Push` to enqueue, `Skip` for well-formed
20    /// events with nothing to run (e.g. branch deletion), `None` for
21    /// malformed bodies.
22    fn parse_push(&self, body: &[u8]) -> Option<PushOutcome>;
23    /// Unique delivery id from the request headers (e.g. GitHub's
24    /// `X-GitHub-Delivery`). Used for replay dedup — `None` means the
25    /// trigger has no such concept.
26    fn delivery_id(&self, _headers: &[(String, String)]) -> Option<String> {
27        None
28    }
29}
30
31/// What a parsed payload asks the watcher to do.
32pub enum PushOutcome {
33    /// (repo_name, git_ref, sha) — repo is `repository.full_name`;
34    /// the caller filters ref per watched repo.
35    Push { repo: String, git_ref: String, sha: String },
36    /// Well-formed event that needs no run (deleted branch, zero sha).
37    Skip,
38}
39
40/// A sha is hex of plausible length — rejects path-injection payloads
41/// before the value reaches worktree paths or status URLs.
42pub fn is_valid_sha(sha: &str) -> bool {
43    (7..=64).contains(&sha.len())
44        && sha.bytes().all(|b| b.is_ascii_hexdigit())
45}

47/// Where run outcomes are reported (commit statuses, logs, chat, …).
48pub trait StatusSink: Send + Sync {
49    fn context(&self) -> &str;
50    fn post(&self, sha: &str, state: &str, desc: &str);
51}

53/// How a repo's pending work collapses: `latest` keeps one pending
54/// slot (newest wins — CI gates), `all` runs every push in order
55/// (FIFO, bounded). Per-repo via `[[repos]] queue = "..."`.
56#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize, Default)]
57#[serde(rename_all = "lowercase")]
58pub enum QueuePolicy {
59    #[default]
60    Latest,
61    All,
62}
63
64/// Backlog cap for `all` — overflow evicts the oldest pending sha.
65pub const MAX_PENDING: usize = 32;
66
67/// Bounded dedup set for delivery ids — a replayed webhook reuses the
68/// same id, so `insert` returning false means "already handled".
69/// Oldest ids are evicted past `cap`; memory-only (replays matter
70/// within process uptime).
71pub struct Seen {
72    ids: std::collections::HashSet<String>,
73    order: VecDeque<String>,
74    cap: usize,
75}
76
77impl Seen {
78    pub fn new(cap: usize) -> Self {
79        Seen { ids: Default::default(), order: VecDeque::new(), cap }
80    }
81    /// true = fresh, false = replay/duplicate.
82    pub fn insert(&mut self, id: String) -> bool {
83        if !self.ids.insert(id.clone()) {
84            return false;
85        }
86        self.order.push_back(id);
87        while self.order.len() > self.cap {
88            if let Some(old) = self.order.pop_front() {
89                self.ids.remove(&old);
90            }
91        }
92        true
93    }
94}
95
96impl Default for Seen {
97    fn default() -> Self {
98        Self::new(10_000)
99    }
100}

102/// Queue state. The running child handle lives in the runner/CLI layer
103/// (`ChildSlot`) so enqueue can kill it without blocking on this lock.
104#[derive(Default)]
105pub struct Queue {
106    pub pending: VecDeque<String>,
107    pub running: Option<String>,
108    pub cancelled: bool,
109    pub policy: QueuePolicy,
110}
111
112impl Queue {
113    pub fn with_policy(policy: QueuePolicy) -> Self {
114        Queue { policy, ..Default::default() }
115    }
116
117    /// `latest`: replace the pending sha (old is returned for a
118    /// superseded status). `all`: append; a full backlog evicts the
119    /// oldest, also returned as superseded. `Duplicate` = deduped.
120    pub fn enqueue(&mut self, sha: String) -> EnqueueResult {
121        if self.running.as_deref() == Some(sha.as_str())
122            || self.pending.iter().any(|s| s == &sha)
123        {
124            return EnqueueResult::Duplicate;
125        }
126        let superseded = match self.policy {
127            QueuePolicy::Latest => self.pending.pop_front(),
128            QueuePolicy::All if self.pending.len() >= MAX_PENDING =>
129                self.pending.pop_front(),
130            QueuePolicy::All => None,
131        };
132        self.pending.push_back(sha);
133        EnqueueResult::Queued { superseded }
134    }
135
136    /// Next sha to run, FIFO.
137    pub fn dequeue(&mut self) -> Option<String> {
138        self.pending.pop_front()
139    }
140}
141
142pub enum EnqueueResult {
143    Queued { superseded: Option<String> },
144    Duplicate,
145}

177/// One watched repository under [[repos]].
178#[derive(Debug, Deserialize, Clone)]

206/// Resolved view used by the watcher — always a list, even for the
207/// legacy single-repo form.
208#[derive(Debug, Clone)]
209pub struct WatchedRepo {
210    pub name: String,
211    pub dir: PathBuf,
212    pub branch: String,
213    pub status_repo: String,
214    /// Global [runner] or the repo's [repos.runner] override.
215    pub runner: RunnerCfg,
216    pub policy: QueuePolicy,
217}
218
219impl Config {
220    pub fn watched(&self) -> Result<Vec<WatchedRepo>, String> {
221        if !self.repos.is_empty() {
222            return self.repos.iter().map(|r| {
223                let runner = r.runner.clone()
224                    .unwrap_or_else(|| self.runner.clone());
225                // `all` keeps every push — killing the in-flight run
226                // would drop a commit that was never superseded.
227                if r.queue == QueuePolicy::All && runner.cancel_running {
228                    return Err(format!(
229                        "repo {}: queue = \"all\" conflicts with \
230                         runner.cancel_running", r.name));
231                }
232                Ok(WatchedRepo {
233                    name: r.name.clone(),
234                    dir: r.dir.clone(),
235                    branch: r.branch.clone(),
236                    status_repo: r.status_repo().to_string(),
237                    runner,
238                    policy: r.queue,
239                })
240            }).collect();
241        }
242        if let Some(r) = &self.repo {
243            let name = if self.status.repo.is_empty() {
244                return Err("single-repo config needs status.repo".into());
245            } else {
246                self.status.repo.clone()
247            };
248            return Ok(vec![WatchedRepo {
249                name, dir: r.dir.clone(),
250                branch: r.branch.clone(),
251                status_repo: self.status.repo.clone(),
252                runner: self.runner.clone(),
253                policy: QueuePolicy::Latest,
254            }]);
255        }
256        Err("no [[repos]] entries and no [repo] configured".into())
257    }
258}

400impl Config {
401    /// Load order: explicit path → $BUPHA_CONFIG → ./bupha.toml →
402    /// ~/.config/bupha/bupha.toml.
403    pub fn load(path: Option<&str>) -> Result<Config, String> {
404        let candidates: Vec<PathBuf> = match path.map(str::to_string)
405            .or_else(|| std::env::var("BUPHA_CONFIG").ok())
406        {
407            Some(p) => vec![PathBuf::from(p)],
408            None => vec![
409                PathBuf::from("bupha.toml"),
410                dirs_home().join(".config/bupha/bupha.toml"),
411            ],
412        };
413        for p in candidates {
414            if p.exists() {
415                let text = std::fs::read_to_string(&p)
416                    .map_err(|e| format!("read {}: {e}", p.display()))?;
417                return toml::from_str(&text)
418                    .map_err(|e| format!("parse {}: {e}", p.display()));
419            }
420        }
421        Err("no bupha.toml found (search: $BUPHA_CONFIG, ., ~/.config/bupha/)"
422            .into())
423    }
424}
425
426fn dirs_home() -> PathBuf {
427    std::env::var("HOME").map(PathBuf::from).unwrap_or_else(|_| ".".into())
