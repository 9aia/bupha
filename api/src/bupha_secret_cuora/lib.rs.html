1//! Cuora impl of `Secrets` — the canonical secrets story (T-056).
2//! Thin wrapper over the shared [`hypha_secret_cuora`] adapter,
3//! pinned to the `bupha` scope.
4//!
5//! Each bupha secret NAME resolves to a fixed cuora ref:
6//! `cuora://bupha/<NAME>` — e.g. `BUPHA_WH_9AIA_BUPHA` maps to
7//! `cuora://bupha/BUPHA_WH_9AIA_BUPHA`. Values are fetched through
8//! cuora so audit events, guardrails, and vault policy apply
9//! exactly as for every other consumer; bupha never stores a value
10//! itself.
11//!
12//! Two transports, selected by `BUPHA_CUORA_TRANSPORT` (the
13//! `<PRODUCT>_CUORA_TRANSPORT` convention — see `hypha-secret-cuora`):
14//!
15//! - `cli` (default) — shell `cuora get`/`put` per call. Secret
16//!   resolution happens a handful of times per run/verb, so a
17//!   per-call process is cheap and needs no shared mutable state.
18//! - `serve` — a persistent `cuora serve` child speaking the
19//!   ndjson JSON-RPC wire (`cuora-serve/1`) on stdio. Spawned lazily
20//!   on first use, reused for the client's lifetime.
21//!
22//! A missing `cuora` binary, a denied op, or an absent ref all
23//! answer `None` — "not here" — so the caller's chain falls through
24//! to the env-file leg exactly like a missing ref on a headless
25//! box, on either transport.
26//!
27//! ## Sealed-pack fallback (T-053)
28//!
29//! The `bupha` scope is already the canonical fleet surface, so
30//! there is no middle scope here — but the CI-webhook secrets are
31//! migrating into the sealed tamias pack `secrets-ci-webhooks`
32//! (vault-less, age-sealed; see cuora's `cuora://<pack>/<secret>`
33//! read path). Resolution is therefore a two-scope chain, vault
34//! first:
35//!
36//! 1. `cuora://bupha/<NAME>` — the vault copy;
37//! 2. `cuora://secrets-ci-webhooks/<NAME>` — the sealed pack;
38//! 3. the caller's next leg (env file).
39//!
40//! Vault-first ordering means a name present in both keeps
41//! resolving to the vault value — packs are the migration target,
42//! not an override. `set` always writes the vault scope: pack
43//! contents are authored via `tamias`/`cuora seal`, never this
44//! store.
45//!
46//! Env knobs: `BUPHA_CUORA_BIN` overrides the binary path (tests
47//! stub it); `BUPHA_CUORA_TRANSPORT` picks `cli`/`serve`;
48//! `CUORA_*` and `--config` resolution stay cuora's own — the child
49//! inherits the environment, so `CUORA_ENV` picks the vault env —
50//! bupha does not pin one.
51
52pub use hypha_secret_cuora::{CuoraServe, CuoraTransport};

54use hypha_secret::Secrets;
55use std::path::PathBuf;
56use std::sync::OnceLock;

58/// Ref namespace — every bupha secret name lives under
59/// `cuora://bupha/…` so `cuora list` shows the whole fleet surface.
60pub const NAMESPACE: &str = "bupha";
61
62/// Sealed-pack fallback namespace — the `secrets-ci-webhooks`
63/// tamias pack holding the `BUPHA_WH_*` webhook secrets
64/// (vault-less, age-sealed). Consulted only when
65/// `cuora://bupha/<NAME>` misses.
66pub const PACK_NAMESPACE: &str = "secrets-ci-webhooks";
67
68/// The env knobs this shim reads — re-exported for consumers that
69/// build their own transport legs.
70pub const BIN_ENV: &str = "BUPHA_CUORA_BIN";
71pub const TRANSPORT_ENV: &str = "BUPHA_CUORA_TRANSPORT";
72
73/// `Secrets` over cuora — CLI or `cuora serve` transport per
74/// [`CuoraTransport`]. Thin wrapper over
75/// [`hypha_secret_cuora::CuoraSecrets`] pinned to the `bupha`
76/// scope plus the `secrets-ci-webhooks` sealed-pack fallback
77/// (see the module docs); the env knobs stay bupha's.
78///
79/// Clones share the serve child — see
80/// [`CuoraSecrets::shared_default`] for why that matters.
81#[derive(Clone)]
82pub struct CuoraSecrets {
83    /// The `cuora` binary — `BUPHA_CUORA_BIN` when set, else `cuora`
84    /// resolved on PATH. Kept as a field (not just inside `inner`)
85    /// because consumers document it as the test seam.
86    pub bin: PathBuf,
87    inner: hypha_secret_cuora::CuoraSecrets,
88    /// Sealed-pack scope — consulted only on a vault miss (see the
89    /// module docs' pack-fallback chain).
90    pack: hypha_secret_cuora::CuoraSecrets,
91}
92
93impl Default for CuoraSecrets {
94    fn default() -> Self {
95        Self::from_env()
96    }
97}
98
99impl std::fmt::Debug for CuoraSecrets {
100    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
101        f.debug_struct("CuoraSecrets")
102            .field("bin", &self.bin)
103            .field("transport", &self.inner.transport())
104            .field("pack_scope", &PACK_NAMESPACE)
105            .finish_non_exhaustive()
106    }
107}
108
109impl CuoraSecrets {
110    /// The env-selected store — `BUPHA_CUORA_BIN` +
111    /// `BUPHA_CUORA_TRANSPORT`.
112    fn from_env() -> Self {
113        Self::with_transport(
114            std::env::var_os(BIN_ENV).map_or_else(|| PathBuf::from("cuora"), PathBuf::from),
115            CuoraTransport::from_env(TRANSPORT_ENV),
116        )
117    }
118
119    /// A store over `bin` with the env-selected transport.
120    #[must_use]
121    pub fn with_bin(bin: PathBuf) -> Self {
122        Self::with_transport(bin, CuoraTransport::from_env(TRANSPORT_ENV))
123    }
124
125    /// A store over `bin` on an explicit transport — the test seam
126    /// and the "I know what I'm doing" constructor.
127    #[must_use]
128    pub fn with_transport(bin: PathBuf, transport: CuoraTransport) -> Self {
129        Self {
130            bin: bin.clone(),
131            inner: hypha_secret_cuora::CuoraSecrets::new(NAMESPACE)
132                .with_bin(bin.clone())
133                .with_transport(transport),
134            pack: hypha_secret_cuora::CuoraSecrets::new(PACK_NAMESPACE)
135                .with_bin(bin)
136                .with_transport(transport),
137        }
138    }
139
140    /// The process-wide default store.
141    ///
142    /// Worth sharing under the serve transport: `CuoraSecrets` is
143    /// cheap to construct but the serve child's whole point is
144    /// persistence — a fresh store per lookup would spawn and reap
145    /// `cuora serve` on every secret read, worse than the CLI. Under
146    /// `cli` sharing is a no-op behaviorally.
147    #[must_use]
148    pub fn shared_default() -> &'static Self {
149        static DEFAULT: OnceLock<CuoraSecrets> = OnceLock::new();
150        DEFAULT.get_or_init(CuoraSecrets::from_env)
151    }
152}
153
154/// Build the ref for a secret name. A free fn so tests pin the
155/// mapping without constructing the store.
156///
157/// `BUPHA_WH_9AIA_BUPHA` → `cuora://bupha/BUPHA_WH_9AIA_BUPHA`.
158#[must_use]
159pub fn rref(name: &str) -> String {
160    hypha_secret_cuora::rref(NAMESPACE, name)
161}
162
163impl Secrets for CuoraSecrets {
164    /// Vault scope first, then the sealed pack — a name present in
165    /// both keeps resolving to the vault value, so the pack (the
166    /// migration target) never overrides the vault.
167    fn get(&self, name: &str) -> Option<String> {
168        self.inner.get(name).or_else(|| self.pack.get(name))
169    }
170
171    /// Deploy-time migration — `cuora put` reads the value on stdin
172    /// (never argv) on the CLI leg; under serve it rides the
173    /// request's `value` field, still never argv.
174    fn set(&self, name: &str, value: &str) -> std::io::Result<()> {
175        self.inner.set(name, value)
176    }
177}
