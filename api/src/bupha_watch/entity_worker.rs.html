1//! entity_worker — the `ci-job` consumption loop (P-002 §4.1): the
2//! entity-plane replacement for the retired redis `WorkQueue` pull path (T-049).
3//!
4//! One worker process polls `EntityApi::list_entities("ci-job")`,
5//! filters pending jobs this box can run (known key + `runsOn` ⊆
6//! `[worker] labels`), CAS-claims via `ci.claim`, executes through
7//! the same `run_job` pipeline every other host uses (worktree →
8//! runner → status sink), renews the lease on `run_job`'s ticker,
9//! and settles with `ci.complete`. Claim loss (409 fence mismatch)
10//! abandons the run — a peer owns it; a `cancelRequested` flag on
11//! the entity kills the child and settles `cancelled`.
12//!
13//! The loop is split into a narrow synchronous [`JobApi`] port (so
14//! the state machine unit-tests against an in-memory fake) and
15//! [`EntityJobApi`], the `hypha_node_api::EntityApi` adapter that
16//! bridges the async trait onto a current-thread tokio runtime —
17//! `HttpEntityApi` is blocking-ureq inside `spawn_blocking`, which
18//! needs a runtime context bupha otherwise doesn't carry.
19

41/// Default claim lease — the plane's `DEFAULT_LEASE_SECS` (5 min,
42/// renewed ~every 2 min by `run_job`'s ticker).
43const DEFAULT_LEASE_SECS: u64 = 300;
44
45/// `BUPHA_ENTITY_LEASE_SECS` overrides the claim lease — same env
46/// override style as `BUPHA_RENEW_TICKS`/`BUPHA_WORKER_POLL_MS`.
47fn lease_secs() -> u64 {
48    std::env::var("BUPHA_ENTITY_LEASE_SECS")
49        .ok()
50        .and_then(|v| v.parse::<u64>().ok())
51        .filter(|n| *n > 0)
52        .unwrap_or(DEFAULT_LEASE_SECS)
53}
54
55/// Idle poll between scans — `BUPHA_WORKER_POLL_MS`, default 1s.
56fn poll_ms() -> u64 {
57    std::env::var("BUPHA_WORKER_POLL_MS")
58        .ok()
59        .and_then(|v| v.parse::<u64>().ok())
60        .unwrap_or(1000)
61}

65/// A `ci-job` entity as the worker sees it — the wire id (ops target
66/// it verbatim) plus the parsed indexDoc.
67#[derive(Debug, Clone)]
68pub struct EntityJob {
69    /// `ci:{key}:{sha}:{attempt}` — contains `/` via the repo part;
70    /// `HttpEntityApi` percent-encodes it into the path.
71    pub id: String,
72    /// The parsed + validated indexDoc (`CiJob::from_index_doc` —
73    /// the same trust boundary the plane applies at create).
74    pub job: CiJob,
75}
76
77/// A won `ci.claim` — every holder op (`renew`/`release`/`complete`)
78/// must echo `worker` + `fence`; a mismatch is a 409 = claim lost.
79#[derive(Debug, Clone)]
80pub struct HeldClaim {
81    pub id: String,
82    pub worker: String,
83    pub fence: u64,
84}
85
86/// `JobApi::claim` resolution — `Lost` covers every "someone else
87/// owns it" shape (409 heldBy, label-gate 409, entity gone); the
88/// caller rescans rather than distinguishing.
89#[derive(Debug)]
90pub enum ClaimOutcome {
91    Won(HeldClaim),
92    Lost,
93}
94
95/// The narrow synchronous port the claim loop speaks. [`EntityJobApi`]
96/// adapts `hypha_node_api::EntityApi` (async, ureq-backed);
97/// tests substitute an in-memory fake.
98pub trait JobApi: Send + Sync {
99    /// `list_entities("ci-job")` → parsed pending-capable docs.
100    fn list_jobs(&self) -> io::Result<Vec<EntityJob>>;
101    /// `GET /api/entities/{id}` → the current doc (cancel/holder
102    /// checks). `None` = unknown id.
103    fn get_job(&self, id: &str) -> io::Result<Option<EntityJob>>;
104    /// `ci.claim {worker, leaseSecs}` — CAS `pending→running`.
105    fn claim(&self, id: &str, worker: &str, lease_secs: u64) -> io::Result<ClaimOutcome>;
106    /// `ci.renew {worker, fence, leaseSecs}` — `false` = fence
107    /// mismatch (claim lost) OR `cancelRequested` observed; the
108    /// caller disambiguates via `get_job` after the run aborts.
109    fn renew(&self, claim: &HeldClaim, lease_secs: u64) -> io::Result<bool>;
110    /// `ci.complete {worker, fence, result, description?}` — terminal
111    /// transition. `Err` on transport failure or 409 (fence lost —
112    /// the peer that re-claimed owns the settle).
113    fn complete(
114        &self,
115        claim: &HeldClaim,
116        result: &str,
117        description: Option<&str>,
118    ) -> io::Result<()>;
119    /// `ci.release {worker, fence, reason?}` — graceful
120    /// `running→pending` (shutdown path).
121    fn release(&self, claim: &HeldClaim, reason: &str) -> io::Result<()>;
122    /// Best-effort terminal notification — the worker POSTs a typed
123    /// `ci.job.completed` event to the node's event-ingest route so
124    /// campaigns can react (pr-heal). Called only after a successful
125    /// `ci.complete`; failures are logged, never propagated — the
126    /// settle already happened. Default no-op.
127    fn emit_completed(&self, _job: &EntityJob, _result: &str) {}
128}

132/// `JobApi` over `hypha_node_api::EntityApi` — the production impl
133/// (`HttpEntityApi` against a hypha/peregrin node; any other
134/// `EntityApi` impl works too, e.g. a hub's `SyncEntityApi`).
135///
136/// The trait is `async_trait` and `HttpEntityApi` hops onto
137/// `spawn_blocking`, so calls need a tokio runtime — this type owns
138/// a current-thread one and `block_on`s each call. bupha stays
139/// sync/no-runtime everywhere else; the runtime is private to the
140/// worker.
141pub struct EntityJobApi {
142    api: Arc<dyn EntityApi>,
143    rt: tokio::runtime::Runtime,
144    /// Node origin + credential for the event-ingest route — set by
145    /// [`EntityJobApi::http`], `None` for `new` (non-HTTP planes have
146    /// no ingest endpoint to call). The origin is the FULL
147    /// `/api/events/inject` URL — [`EntityJobApi::with_events_url`]
148    /// retargets it (e.g. `campaign serve`'s shared listener) without
149    /// moving the entity ops.
150    emit: Option<(String, Option<String>)>,
151}
152
153impl EntityJobApi {
154    /// Wrap any `EntityApi` impl.
155    ///
156    /// # Errors
157    /// Fails if the current-thread runtime can't be built.
158    pub fn new(api: Arc<dyn EntityApi>) -> io::Result<Self> {
159        let rt = tokio::runtime::Builder::new_current_thread()
160            .enable_all()
161            .build()
162            .map_err(io::Error::other)?;
163        Ok(Self {
164            api,
165            rt,
166            emit: None,
167        })
168    }
169
170    /// Convenience for the common case — ureq `HttpEntityApi`
171    /// against `base` (e.g. `http://127.0.0.1:18787`) with an
172    /// optional bearer token (paired node credential). The base +
173    /// token are kept for [`JobApi::emit_completed`].
174    ///
175    /// # Errors
176    /// Fails if the runtime can't be built.
177    pub fn http(base: &str, token: Option<String>) -> io::Result<Self> {
178        let mut api = Self::new(Arc::new(hypha_node_api::api::HttpEntityApi::new(
179            base,
180            token.clone(),
181        )))?;
182        api.emit = Some((
183            format!("{}/api/events/inject", base.trim_end_matches('/')),
184            token,
185        ));
186        Ok(api)
187    }
188
189    /// Retarget the `ci.job.completed` emit — `url` is the FULL
190    /// ingest endpoint (e.g. `campaign serve`'s
191    /// `http://127.0.0.1:29173/api/events/inject`), not a node
192    /// origin. The bearer credential stays the entity node's
193    /// `node_token_env` token — deployments point both planes at the
194    /// same `PEREGRIN_TOKEN`. No-op-ish when `url` is `None` (keeps
195    /// the `{node}/api/events/inject` default).
196    #[must_use]
197    pub fn with_events_url(mut self, url: Option<String>) -> Self {
198        if let (Some(events), Some((emit_url, _))) = (&url, &mut self.emit) {
199            *emit_url = events.clone();
200        }
201        self
202    }
203
204    fn submit(&self, id: &str, kind: &str, payload: serde_json::Value) -> io::Result<()> {
205        self.submit_result(id, kind, payload).map(|_| ())
206    }
207
208    /// `submit_op_result` — same op, but the node's response body
209    /// (or `Queued`) survives the hop. `ci.claim` is the one op whose
210    /// body carries state the caller needs (`fence`).
211    fn submit_result(
212        &self,
213        id: &str,
214        kind: &str,
215        payload: serde_json::Value,
216    ) -> io::Result<OpResult> {
217        let api = Arc::clone(&self.api);
218        let id = id.to_string();
219        let op = EntityOp {
220            kind: kind.to_string(),
221            payload,
222        };
223        self.rt
224            .block_on(async move { api.submit_op_result(&id, &op).await })
225            .map_err(io::Error::other)
226    }
227
228    /// `submit_op`'s error channel is a formatted string
229    /// (`"submit op: HTTP 409 — {body}"`) — the heldBy detail rides
230    /// in the body; the loop only needs the status class.
231    fn is_conflict(e: &io::Error) -> bool {
232        let msg = e.to_string();
233        msg.contains("HTTP 409") || msg.contains("HTTP 404")
234    }
235
236    /// The pre-`submit_op_result` claim path — re-read the entity doc
237    /// the claim just wrote. Ops serialize node-side, so the read is
238    /// consistent; kept as the fallback for body-less responses.
239    fn claim_from_doc(&self, id: &str, worker: &str) -> io::Result<ClaimOutcome> {
240        match self.get_job(id)? {
241            Some(ej)
242                if ej.job.state == JobState::Running
243                    && ej.job.claimed_by.as_deref() == Some(worker) =>
244            {
245                Ok(ClaimOutcome::Won(HeldClaim {
246                    id: id.to_string(),
247                    worker: worker.to_string(),
248                    fence: ej.job.fence,
249                }))
250            }
251            // Claim accepted but the doc disagrees — treat as
252            // lost rather than run unfenced.
253            _ => Ok(ClaimOutcome::Lost),
254        }
255    }
256}
257
258/// `ci.claim`'s op body → `HeldClaim` — `{claimed, fence, job}` where
259/// `job` is the post-claim indexDoc (P-002 §1). Returns `None` when
260/// the body doesn't prove a won claim for `worker` — the caller falls
261/// back to the entity re-read rather than running unfenced.
262fn claim_from_body(id: &str, worker: &str, body: &Value) -> Option<HeldClaim> {
263    if body.get("claimed").and_then(Value::as_bool) != Some(true) {
264        return None;
265    }
266    // Top-level `fence` must agree with the doc's — they come from
267    // the same transaction; a mismatch means a foreign body.
268    let doc_fence = body.pointer("/job/fence").and_then(Value::as_u64);
269    let body_fence = body.get("fence").and_then(Value::as_u64);
270    if doc_fence.is_none() || body_fence.is_none() || doc_fence != body_fence {
271        return None;
272    }
273    let now = chrono::Utc::now().to_rfc3339();
274    let job = CiJob::from_index_doc(body.get("job")?, &now).ok()?;
275    if job.state != JobState::Running || job.claimed_by.as_deref() != Some(worker) {
276        return None;
277    }
278    Some(HeldClaim {
279        id: id.to_string(),
280        worker: worker.to_string(),
281        fence: job.fence,
282    })
283}

285/// `EntityDto.index_doc` → `EntityJob` — the same validation the
286/// plane applies at create; hostile docs drop out of the candidate
287/// set rather than reaching git argv.
288fn parse_job(dto: &EntityDto) -> Option<EntityJob> {
289    if dto.kind != CI_JOB_KIND {
290        return None;
291    }
292    let now = chrono::Utc::now().to_rfc3339();
293    let mut job = CiJob::from_index_doc(&dto.index_doc, &now).ok()?;
294    job.id = dto.id.clone();
295    Some(EntityJob {
296        id: dto.id.clone(),
297        job,
298    })
299}

301impl JobApi for EntityJobApi {
302    fn list_jobs(&self) -> io::Result<Vec<EntityJob>> {
303        let api = Arc::clone(&self.api);
304        let dtos = self
305            .rt
306            .block_on(api.list_entities(Some(CI_JOB_KIND)))
307            .map_err(io::Error::other)?;
308        Ok(dtos.iter().filter_map(parse_job).collect())
309    }
310
311    fn get_job(&self, id: &str) -> io::Result<Option<EntityJob>> {
312        let api = Arc::clone(&self.api);
313        let id = id.to_string();
314        match self.rt.block_on(api.entity(&id)) {
315            Ok(dto) => Ok(parse_job(&dto)),
316            // 404 = gone — not an error for the poll/claim paths.
317            Err(e) if e.contains("HTTP 404") => Ok(None),
318            Err(e) => Err(io::Error::other(e)),
319        }
320    }
321
322    fn claim(&self, id: &str, worker: &str, lease_secs: u64) -> io::Result<ClaimOutcome> {
323        let payload = json!({ "worker": worker, "leaseSecs": lease_secs });
324        match self.submit_result(id, "ci.claim", payload) {
325            Ok(OpResult::Applied(Some(body))) => {
326                // Fast path: the op body is `{claimed, fence, job}` —
327                // the post-claim indexDoc, no entity re-read needed.
328                if let Some(held) = claim_from_body(id, worker, &body) {
329                    return Ok(ClaimOutcome::Won(held));
330                }
331                // Body didn't prove the win (non-ci-job plane, older
332                // node, opaque response) — fall back to the re-read.
333                self.claim_from_doc(id, worker)
334            }
335            Ok(OpResult::Applied(None)) => self.claim_from_doc(id, worker),
336            // Hub outbox-parked the claim — no node response exists
337            // yet, so the win can't be proven. Report Lost: the next
338            // poll resubmits (the CAS makes the replay idempotent)
339            // and the claim settles once the owning node is back.
340            Ok(OpResult::Queued) => Ok(ClaimOutcome::Lost),
341            Err(e) if Self::is_conflict(&e) => Ok(ClaimOutcome::Lost),
342            Err(e) => Err(e),
343        }
344    }
345
346    fn renew(&self, claim: &HeldClaim, lease_secs: u64) -> io::Result<bool> {
347        let payload = json!({
348            "worker": claim.worker,
349            "fence": claim.fence,
350            "leaseSecs": lease_secs,
351        });
352        match self.submit(&claim.id, "ci.renew", payload) {
353            Ok(()) => {
354                // The 200 proves holder+fence; the extra read is for
355                // `cancelRequested` — the worker's kill switch (the
356                // renew loop is the only place a running job can
357                // observe it, P-002 §2 `ci.cancel`).
358                match self.get_job(&claim.id)? {
359                    Some(ej) => Ok(ej.job.state == JobState::Running
360                        && ej.job.claimed_by.as_deref() == Some(claim.worker.as_str())
361                        && !ej.job.cancel_requested),
362                    None => Ok(false),
363                }
364            }
365            Err(e) if Self::is_conflict(&e) => Ok(false),
366            Err(e) => Err(e),
367        }
368    }
369
370    fn complete(
371        &self,
372        claim: &HeldClaim,
373        result: &str,
374        description: Option<&str>,
375    ) -> io::Result<()> {
376        let mut payload = json!({
377            "worker": claim.worker,
378            "fence": claim.fence,
379            "result": result,
380        });
381        if let Some(d) = description {
382            payload["description"] = json!(d);
383        }
384        self.submit(&claim.id, "ci.complete", payload)
385    }
386
387    fn release(&self, claim: &HeldClaim, reason: &str) -> io::Result<()> {
388        let payload = json!({
389            "worker": claim.worker,
390            "fence": claim.fence,
391            "reason": reason,
392        });
393        self.submit(&claim.id, "ci.release", payload)
394    }
395
396    fn emit_completed(&self, job: &EntityJob, result: &str) {
397        // `emit.0` is the full ingest URL (`http()` defaults it to
398        // `{node}/api/events/inject`; `with_events_url` retargets).
399        let Some((url, token)) = &self.emit else {
400            return;
401        };
402        let j = &job.job;
403        let body = json!({
404            "type": "ci.job.completed",
405            "payload": {
406                "job": job.id,
407                "key": j.key,
408                "repo": j.repo,
409                "ref": j.git_ref,
410                "sha": j.sha,
411                "event": j.event,
412                "sender": j.sender,
413                "deliveryId": j.delivery_id,
414                "result": result,
415                "attempt": j.attempt,
416            },
417            // Settle is once-per-(job,fence-epoch); the key dedups a
418            // retried emit without masking a genuine re-run (new
419            // attempt = new entity id).
420            "idempotencyKey": format!("{}:{}", job.id, result),
421        });
422        let res = match token {
423            Some(t) => ureq::post(url)
424                .header("authorization", &format!("Bearer {t}"))
425                .send_json(&body),
426            None => ureq::post(url).send_json(&body),
427        };
428        if let Err(e) = res {
429            eprintln!("ci.job.completed emit {}: {e}", job.id);
430        }
431    }
432}

436/// `TryClaim` over a held entity claim — `run_job`'s renew ticker
437/// speaks this trait; every call maps onto `ci.renew`. `try_claim`/
438/// `is_done`/`mark_done` are single-box no-ops here: the entity CAS
439/// already happened and there is no peer-done set to consult.
440pub struct EntityClaim<'a> {
441    api: &'a dyn JobApi,
442    held: HeldClaim,
443    lease_secs: u64,
444}
445
446impl<'a> EntityClaim<'a> {
447    fn new(api: &'a dyn JobApi, held: HeldClaim, lease_secs: u64) -> Self {
448        Self {
449            api,
450            held,
451            lease_secs,
452        }
453    }
454}
455
456impl TryClaim for EntityClaim<'_> {
457    fn try_claim(&self, _repo: &str, _sha: &str, _worker: &str) -> io::Result<Option<Fence>> {
458        // The claim was already won via `ci.claim` — hand back its
459        // fence so callers that inspect it see the real token.
460        Ok(Some(Fence(self.held.fence)))
461    }
462
463    fn renew(&self, _repo: &str, _sha: &str, _worker: &str) -> io::Result<bool> {
464        self.api.renew(&self.held, self.lease_secs)
465    }
466}

470/// How one claimed job ended. `Skip` = not runnable after all
471/// (unknown key drift, hostile sha) — the loop releases the claim
472/// instead of settling a fake result.
473#[derive(Debug, Clone, Copy, PartialEq, Eq)]
474pub enum ExecEnd {
475    /// Run completed locally; `false` = failed.
476    Finished(bool),
477    /// Lease lost or cancel observed mid-run — `run_job` killed the
478    /// child; the loop decides complete-vs-abandon from the entity.
479    ClaimLost,
480    /// Release the claim untouched.
481    Skip,
482}

484/// The execution half of the loop — owns runner/sink/worktree
485/// plumbing. [`WatchExec`] is the production impl (the same
486/// `bupha-watch` builders `bupha watch`/`tamias-driver-bupha worker`
487/// use); tests
488/// substitute a scripted fake.
489pub trait JobExec: Send + Sync {
490    /// This worker's claim identity (`<host>:<pid>`) — goes into
491    /// `ci.claim`'s `worker` field.
492    fn worker(&self) -> &str;
493    /// Run one claimed job end-to-end. `claim` renews the entity
494    /// lease on `run_job`'s ticker.
495    fn exec(&self, job: &EntityJob, claim: &EntityClaim<'_>) -> ExecEnd;
496}

498/// Production executor — the same wiring `bupha watch` uses: per-key
499/// status sinks, per-key runners,
500/// telemetry/attest/audit, one shared child slot (one build at a
501/// time per box). Queue persistence does NOT apply — entity state
502/// *is* the queue.
503pub struct WatchExec {
504    cfg: Arc<Config>,
505    watched: Arc<Vec<WatchedRepo>>,
506    sinks: std::collections::HashMap<String, Arc<dyn StatusSink>>,
507    runners: std::collections::HashMap<String, Arc<dyn Runner>>,
508    telemetry: Arc<dyn Telemetry>,
509    attest: Option<Arc<dyn Attest>>,
510    shared: Mutex<Shared>,
511    child: ChildSlot,
512    worker: String,
513    /// Git credential helper for fork-`cloneUrl` fetches —
514    /// materialized once from `[worker] fork_token_env` (`None` =
515    /// fork fetches ride the box's existing git auth).
516    fork_helper: Option<std::path::PathBuf>,
517}
518
519impl WatchExec {
520    /// Wire the executor against `cfg` — same fail-fast config
521    /// errors `bupha watch` raises.
522    ///
523    /// # Panics
524    /// Exits the process on an unloadable config or unknown runner/
525    /// telemetry/audit kind — identical to the CLI paths.
526    #[must_use]
527    pub fn new(cfg: Arc<Config>) -> Self {
528        let token = secret(&cfg.runner.token_env).unwrap_or_default();
529        let watched: Vec<WatchedRepo> = cfg.watched().unwrap_or_else(|e| {
530            eprintln!("config: {e}");
531            std::process::exit(2);
532        });
533        let watched = Arc::new(watched);
534        let audit = build_audit(&cfg);
535        let sinks = build_sinks(&cfg, &watched, &token, audit.as_ref());
536        let runners = build_runners(&cfg, &watched);
537        let telemetry = build_telemetry(&cfg);
538        let attest = build_attest(&cfg);
539        let fork_helper = fork_credential_helper(&cfg);
540        let _ = std::fs::create_dir_all(&cfg.worktree.base);
541        Self {
542            cfg,
543            watched,
544            sinks,
545            runners,
546            telemetry,
547            attest,
548            shared: Mutex::new(Shared::default()),
549            child: Arc::new(Mutex::new(None)),
550            worker: worker_id(),
551            fork_helper,
552        }
553    }
554}

556impl JobExec for WatchExec {
557    fn worker(&self) -> &str {
558        &self.worker
559    }
560
561    fn exec(&self, ej: &EntityJob, claim: &EntityClaim<'_>) -> ExecEnd {
562        let Some(repo) = self.watched.iter().find(|r| r.key == ej.job.key) else {
563            eprintln!("ci-job {}: unknown key {:?} — releasing", ej.id, ej.job.key);
564            return ExecEnd::Skip;
565        };
566        // Ref/event gate — the same match the standalone ingest
567        // applies (`w.branch == git_ref && w.events.contains(event)`,
568        // T-068). `candidates` pre-filters, but exec is also the
569        // trust boundary for direct callers: a ci-job enqueued for a
570        // ref this job doesn't watch must not run a sha that merely
571        // exists in repo.dir.
572        if repo.branch != ej.job.git_ref || !repo.events.iter().any(|e| e == &ej.job.event) {
573            eprintln!(
574                "ci-job {}: ref/event {}/{:?} doesn't match {} — releasing",
575                ej.id, ej.job.git_ref, ej.job.event, repo.key
576            );
577            return ExecEnd::Skip;
578        }
579        let sha = ej.job.sha.as_str();
580        // Last-line validation before the sha names worktree dirs and
581        // reaches git argv — the plane validated at create, but the
582        // wire is a trust boundary at every hop.
583        if !is_valid_sha(sha) {
584            eprintln!("ci-job {}: refusing unsafe sha {sha:?}", ej.id);
585            return ExecEnd::Skip;
586        }
587        let (Some(sink), Some(runner)) = (self.sinks.get(&repo.key), self.runners.get(&repo.key))
588        else {
589            eprintln!("ci-job {}: no sink/runner for key {:?}", ej.id, repo.key);
590            return ExecEnd::Skip;
591        };
592        self.telemetry.event(&RunEvent::Started {
593            repo: repo.key.clone(),
594            sha: sha.to_string(),
595        });
596        let t0 = std::time::Instant::now();
597        // Fork-PR checkout override (T-073): `cloneUrl`/`headRef`
598        // name the remote+ref the head sha is fetched from when it
599        // isn't in `repo.dir` — absent for pushes and same-repo PRs.
600        let fetch = ej.job.clone_url.as_deref().map(|url| FetchTarget {
601            url,
602            git_ref: ej.job.head_ref.as_deref(),
603            credential_helper: self.fork_helper.as_deref(),
604        });
605        let end = run_job(
606            &self.cfg,
607            repo,
608            &**sink,
609            &**runner,
610            &self.shared,
611            &Worker {
612                child: &self.child,
613                claim,
614                id: &self.worker,
615            },
616            sha,
617            fetch,
618            &self.attest,
619        );
620        let wall = t0.elapsed();
621        settle_run(&self.cfg, &*self.telemetry, &repo.key, sha, &end, wall);
622        match end {
623            JobEnd::Finished(ok) => ExecEnd::Finished(ok),
624            JobEnd::ClaimLost => ExecEnd::ClaimLost,
625        }
626    }
627}

631/// What one keys this worker may claim: known job keys + the label
632/// set `[worker] labels` advertises + the per-key ref/event gate the
633/// standalone ingest applies (`branch`/`events` — T-068). Built once
634/// from the config — `runsOn` matching is the same set-equality the
635/// plane re-checks server-side (P-002 §4: the caller pre-filters,
636/// the CAS makes the pick race-safe).
637#[derive(Debug, Default)]
638pub struct ClaimFilter {
639    pub keys: HashSet<String>,
640    pub labels: Vec<String>,
641    /// key → (watched `branch`, `events`) — the entity job's
642    /// `git_ref`/`event` must match exactly like a webhook push does.
643    /// Keys absent here (hand-built filters in tests) skip the gate.
644    specs: std::collections::HashMap<String, (String, Vec<String>)>,
645}
646
647impl ClaimFilter {
648    /// From a loaded config — `cfg.watched()` keys + worker labels +
649    /// each key's branch/event spec.
650    ///
651    /// # Errors
652    /// Propagates `Config::watched` validation failures.
653    pub fn from_config(cfg: &Config) -> Result<Self, String> {
654        let watched = cfg.watched()?;
655        Ok(Self {
656            keys: watched.iter().map(|w| w.key.clone()).collect(),
657            labels: cfg.worker.labels.clone(),
658            specs: watched
659                .iter()
660                .map(|w| (w.key.clone(), (w.branch.clone(), w.events.clone())))
661                .collect(),
662        })
663    }
664}
665
666/// The standalone-ingest match (`w.branch == git_ref &&
667/// w.events.contains(event)`) applied to an entity doc — T-068: the
668/// entity path must not run a sha pushed to a ref this job doesn't
669/// watch.
670fn ref_event_match(spec: &(String, Vec<String>), job: &CiJob) -> bool {
671    spec.0 == job.git_ref && spec.1.iter().any(|e| e == &job.event)
672}
673
674/// Pending candidates in claim order — oldest `enqueuedAt` first
675/// (`all`-lane FIFO; `latest` lanes hold ≤1 pending by construction,
676/// so the sort is a no-op there). Drops terminal/running/foreign/
677/// label-mismatched/hostile docs.
678#[must_use]
679pub fn candidates(jobs: &[EntityJob], filter: &ClaimFilter) -> Vec<EntityJob> {
680    let mut v: Vec<EntityJob> = jobs
681        .iter()
682        .filter(|ej| {
683            ej.job.state == JobState::Pending
684                && filter.keys.contains(&ej.job.key)
685                && filter
686                    .specs
687                    .get(&ej.job.key)
688                    .is_none_or(|spec| ref_event_match(spec, &ej.job))
689                && ej.job.runs_on.iter().all(|l| filter.labels.contains(l))
690                && is_valid_sha(&ej.job.sha)
691        })
692        .cloned()
693        .collect();
694    v.sort_by(|a, b| {
695        a.job
696            .enqueued_at
697            .cmp(&b.job.enqueued_at)
698            .then_with(|| a.id.cmp(&b.id))
699    });
700    v
701}

703/// Outcome of one poll→claim→run→settle cycle.
704#[derive(Debug, Clone, Copy, PartialEq, Eq)]
705pub enum Step {
706    /// Shutdown requested — the loop exits.
707    Shutdown,
708    /// Nothing claimable this scan (or a transient api error).
709    Idle,
710    /// A job was claimed and run to its end.
711    Ran,
712}
713
714/// Settle one won claim — the `ci.complete`/`ci.release`/abandon
715/// decision tree (P-002 §2 semantics over the entity doc).
716fn settle(api: &dyn JobApi, exec: &dyn JobExec, ej: &EntityJob, held: HeldClaim) {
717    let lease = lease_secs();
718    let claim = EntityClaim::new(api, held.clone(), lease);
719    let end = exec.exec(ej, &claim);
720    match end {
721        ExecEnd::Skip => {
722            if let Err(e) = api.release(&held, "not runnable on this worker") {
723                eprintln!("ci.release {}: {e}", held.id);
724            }
725        }
726        ExecEnd::Finished(ok) => {
727            // A cancel that landed after the last renew tick still
728            // wins — we are still the holder, so `cancelled` is the
729            // honest result (the plane keeps `cancelRequested` until
730            // someone settles).
731            let cancelled = api
732                .get_job(&held.id)
733                .ok()
734                .flatten()
735                .is_some_and(|e| e.job.cancel_requested);
736            let (result, desc) = if cancelled {
737                ("cancelled", "ci cancelled")
738            } else if ok {
739                ("success", "ci passed")
740            } else {
741                ("failure", "ci failed")
742            };
743            if let Err(e) = api.complete(&held, result, Some(desc)) {
744                // 409 = fence lost mid/post-run — a peer owns the
745                // settle now; anything else is a transport problem.
746                // Either way there is nothing safe to retry here.
747                eprintln!("ci.complete {}: {e} — abandoning", held.id);
748            } else {
749                api.emit_completed(ej, result);
750            }
751        }
752        ExecEnd::ClaimLost => {
753            // renew returned false for one of two reasons: a real
754            // fence loss (peer re-claimed — abandon silently) or
755            // `cancelRequested` (we are still the holder — settle
756            // `cancelled`). The entity doc disambiguates.
757            let cancel_pending = matches!(
758                api.get_job(&held.id),
759                Ok(Some(e))
760                    if e.job.state == JobState::Running
761                        && e.job.claimed_by.as_deref() == Some(exec.worker())
762                        && e.job.cancel_requested
763            );
764            if cancel_pending {
765                if let Err(e) = api.complete(&held, "cancelled", Some("ci cancelled")) {
766                    eprintln!("ci.complete {}: {e} — claim lost, abandoning", held.id);
767                } else {
768                    api.emit_completed(ej, "cancelled");
769                }
770            }
771        }
772    }
773}
774
775/// One poll→claim→run→settle cycle — exposed for tests; [`run`]
776/// loops it until `shutdown`.
777pub fn step(
778    filter: &ClaimFilter,
779    api: &dyn JobApi,
780    exec: &dyn JobExec,
781    shutdown: &dyn Fn() -> bool,
782) -> Step {
783    if shutdown() {
784        return Step::Shutdown;
785    }
786    let jobs = match api.list_jobs() {
787        Ok(j) => j,
788        Err(e) => {
789            eprintln!("ci-job poll: {e}");
790            return Step::Idle;
791        }
792    };
793    for cand in candidates(&jobs, filter) {
794        match api.claim(&cand.id, exec.worker(), lease_secs()) {
795            Ok(ClaimOutcome::Won(held)) => {
796                println!("ci-job claimed {} ({})", cand.id, held.worker);
797                settle(api, exec, &cand, held);
798                return Step::Ran;
799            }
800            // Lost the CAS — a peer owns it; rescan picks the next.
801            Ok(ClaimOutcome::Lost) => continue,
802            Err(e) => {
803                eprintln!("ci.claim {}: {e}", cand.id);
804                return Step::Idle;
805            }
806        }
807    }
808    Step::Idle
809}
810
811/// The worker loop — poll until `shutdown` (the e2e
812/// `BUPHA_WATCH_SHUTDOWN_FILE` hook in production; systemd's
813/// SIGTERM kills between jobs since runs are synchronous). Sleeps
814/// `poll` between scans — one job at a time per box, same as
815/// `tamias-driver-bupha worker`.
816pub fn run(
817    filter: &ClaimFilter,
818    api: &dyn JobApi,
819    exec: &dyn JobExec,
820    shutdown: &dyn Fn() -> bool,
821) {
822    let poll = Duration::from_millis(poll_ms());
823    loop {
824        if step(filter, api, exec, shutdown) == Step::Shutdown {
825            break;
826        }
827        std::thread::sleep(poll);
828    }
829}

1210        assert!(cands[1].id.ends_with("bbb2222ccc33:1"));

1213        // job — only the two unlabeled `o/n` pendings remain.

1221            specs: Default::default(),
1222        };
