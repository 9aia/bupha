1//! bupha-sturn — the sturn side of the trigger port. Reads
2//! `refs/sturn/*` straight out of a clone via git plumbing: the WAL
3//! is the state (sturn's bus is only a cache), so polling refs is
4//! the robust ingest surface — it survives restarts, catches up
5//! after `git fetch`, and needs no daemon on the CI box.
6//!
7//! What a CI-worthy event looks like: `sturn snap` appends an
8//! `Op::Snapshot` event commit to `refs/sturn/wal/<ws>` (each event
9//! is a commit whose tree holds `event.json`) and moves
10//! `refs/sturn/ws/<ws>` to the snap commit. The poller diffs WAL
11//! heads, walks new event commits, and turns Snapshot markers into
12//! (branch ref, sha, sender) pushes for the shared queue.
13//!
14//! Sender attestation: `actor` in an event is self-asserted. When a
15//! signer policy exists (the repo's `sturn.toml [signing]
16//! allowed_signers`, else `~/.config/sturn/allowed_signers`) each
17//! event's SSH signature is verified (`ssh-keygen -Y verify`,
18//! namespace `sturn-event`) — a signed event emits its actor, an
19//! unsigned or bad-sig event emits an empty sender (which `[access]`
20//! policies will deny). With no signer policy anywhere the sender is
21//! the bare actor claim — `[access] kind = "all"` compatible, but
22//! not an identity an allowlist should trust.
23

33fn git(repo: &Path, args: &[&str]) -> io::Result<String> {
34    let o = Command::new("git")
35        .arg("-C")
36        .arg(repo)
37        .args(args)
38        .stdout(Stdio::piped())
39        .stderr(Stdio::piped())
40        .output()?;
41    if o.status.success() {
42        Ok(String::from_utf8_lossy(&o.stdout).trim().to_string())
43    } else {
44        Err(io::Error::other(format!(
45            "git {}: {}",
46            args.join(" "),
47            String::from_utf8_lossy(&o.stderr).trim()
48        )))
49    }
50}

52/// ws tip refs — `refs/sturn/ws/<ws>` holds the commit of that ws's
53/// last snap (sturn-ws-git hardcodes this prefix; only the WAL
54/// prefix is configurable in sturn.toml).
55pub const WS_PREFIX: &str = "refs/sturn/ws";

57/// The sturn.toml knobs the poller needs. Missing file or missing
58/// fields fall back to sturn's own defaults.
59#[derive(Debug, Clone)]
60pub struct Meta {
61    /// Short branch `sturn snap` lands commits on ("main").
62    pub branch: String,
63    /// WAL ref prefix — one ref per workspace beneath it.
64    pub wal_prefix: String,
65    /// `[signing] allowed_signers` path verbatim (may contain `~`).

67}

96/// Read `<dir>/sturn.toml` — absent/unparseable = sturn defaults.
97pub fn meta(dir: &Path) -> Meta {
98    let cfg: Option<SturnToml> = std::fs::read_to_string(dir.join("sturn.toml"))
99        .ok()
100        .and_then(|s| toml::from_str(&s).ok());
101    Meta {
102        branch: cfg
103            .as_ref()
104            .and_then(|c| c.repo.branch.clone())
105            .unwrap_or_else(|| "main".into()),
106        wal_prefix: cfg
107            .as_ref()
108            .and_then(|c| c.wal.ref_prefix.clone())
109            .unwrap_or_else(|| "refs/sturn/wal".into()),
110        allowed_signers: cfg.and_then(|c| c.signing.allowed_signers.clone()),
111    }
112}

114/// Resolve the allowed_signers file for a repo — the sturn.toml
115/// `[signing]` path (with `~` expanded) or sturn's own default
116/// `~/.config/sturn/allowed_signers`, whichever exists first.
117/// `None` = no signer policy anywhere → actor names are
118/// self-asserted, never attested.
119fn signers_for(meta: &Meta) -> Option<PathBuf> {
120    let expand = |p: &str| -> PathBuf {
121        match p.strip_prefix("~/") {
122            Some(rest) => std::env::var("HOME")
123                .map(PathBuf::from)
124                .unwrap_or_else(|_| ".".into())
125                .join(rest),
126            _ => PathBuf::from(p),
127        }
128    };
129    if let Some(p) = meta.allowed_signers.as_deref().filter(|s| !s.is_empty()) {
130        let path = expand(p);
131        if path.exists() {
132            return Some(path);
133        }
134    }
135    let fallback = std::env::var("HOME")
136        .map(PathBuf::from)
137        .unwrap_or_else(|_| ".".into())
138        .join(".config/sturn/allowed_signers");
139    fallback.exists().then_some(fallback)
140}

142/// The branch ref a snap commit targets — what watched jobs' `branch`

144pub fn branch_ref(meta: &Meta) -> String {
145    format!("refs/heads/{}", meta.branch)

148/// Fetch `refs/sturn/*` so wal/ws refs reflect the remote. `remote`
149/// empty = local-only mode — ops land in the watched clone itself
150/// (the single-box dogfooding case), nothing to fetch.
151pub fn fetch(repo: &Path, remote: &str) -> bool {
152    if remote.is_empty() {
153        return true;
154    }
155    Command::new("git")
156        .arg("-C")
157        .arg(repo)
158        .args(["fetch", remote, "+refs/sturn/*:refs/sturn/*"])
159        .stdout(Stdio::null())
160        .stderr(Stdio::null())
161        .status()
162        .map(|s| s.success())
163        .unwrap_or(false)
164}

166/// (ws, head sha) for every `<wal_prefix>/<ws>` ref — the pollable
167/// surface. Sorted for deterministic ticks.
168pub fn wal_heads(repo: &Path, wal_prefix: &str) -> Vec<(String, String)> {
169    let Ok(out) = git(
170        repo,
171        &[
172            "for-each-ref",
173            "--format=%(refname) %(objectname)",
174            wal_prefix,
175        ],
176    ) else {
177        return vec![];
178    };
179    let mut v: Vec<(String, String)> = out
180        .lines()
181        .filter_map(|l| {
182            let (r, sha) = l.split_once(' ')?;
183            Some((
184                r.strip_prefix(&format!("{wal_prefix}/"))?.to_string(),
185                sha.to_string(),
186            ))
187        })
188        .collect();
189    v.sort();
190    v
191}

193/// One WAL event, minimized — the poller needs the dedup id, the
194/// actor (access policy's `sender`), and whether it is a Snapshot.
195pub struct Event {
196    /// Event id — the event blob's content address; falls back to
197    /// the WAL commit sha for events written before id assignment.
198    pub id: String,
199    /// Self-asserted actor name — attested only when `sig` verifies
200    /// against the repo's allowed_signers (see [`verified_actor`]).
201    pub actor: String,
202    pub snapshot: bool,
203    /// SSH signature (PEM) over the canonical event — present only
204    /// when the WAL writer signed (`[signing] key` in sturn.toml).
205    sig: Option<String>,
206    /// Canonical signing payload: the event with `id` and `sig`
207    /// cleared — exactly what `ssh-keygen -Y sign` covered.
208    signing_bytes: Vec<u8>,
209}
210
211impl Event {
212    /// `actor` iff the signature verifies against `signers` —
213    /// `ssh-keygen -Y verify`, namespace `sturn-event` (sturn's own
214    /// scheme). `None` = unsigned or failed → the source must emit
215    /// an unattested (empty) sender instead of trusting the name.
216    pub fn verified_actor(&self, signers: &Path) -> Option<String> {
217        let sig = self.sig.as_deref()?;
218        let tmp = std::env::temp_dir();
219        let tag = format!("bupha-vfy-{}", std::process::id());
220        let (data, sigf) = (tmp.join(&tag), tmp.join(format!("{tag}.sig")));
221        std::fs::write(&data, &self.signing_bytes).ok()?;
222        std::fs::write(&sigf, sig).ok()?;
223        let st = Command::new("ssh-keygen")
224            .args(["-Y", "verify", "-f"])
225            .arg(signers)
226            .args(["-I", &self.actor, "-n", "sturn-event", "-s"])
227            .arg(&sigf)
228            .stdin(Stdio::from(std::fs::File::open(&data).ok()?))
229            .stdout(Stdio::null())
230            .stderr(Stdio::null())
231            .status();
232        let _ = std::fs::remove_file(&data);
233        let _ = std::fs::remove_file(&sigf);
234        st.ok().filter(|s| s.success()).map(|_| self.actor.clone())
235    }
236}

238/// The event.json payload — `op` kept raw so the signing bytes can be
239/// rebuilt verbatim (signature covers the serialized form, not a
240/// re-serialization that might reorder fields).
241#[derive(Deserialize)]
242struct RawEvent {
243    #[serde(default)]
244    id: String,
245    #[serde(default)]
246    ws: String,
247    #[serde(default)]
248    actor: String,
249    #[serde(default)]
250    ts: u64,
251    #[serde(default)]
252    sig: Option<String>,
253    op: Box<serde_json::value::RawValue>,
254}
255
256#[derive(Deserialize)]
257struct OpTag {
258    op: String,
259}
260
261/// The canonical form `ssh-keygen` signed — sturn-core's
262/// `Event::signing_bytes`: `{"id":"","ws":…,"actor":…,"ts":…,"op":…}`
263/// in field order, `sig` dropped.
264fn signing_bytes(ev: &RawEvent) -> Vec<u8> {
265    use serde::Serialize;
266    #[derive(Serialize)]
267    struct View<'a> {
268        id: &'a str,
269        ws: &'a str,
270        actor: &'a str,
271        ts: u64,
272        op: &'a serde_json::value::RawValue,
273    }
274    serde_json::to_vec(&View {
275        id: "",
276        ws: &ev.ws,
277        actor: &ev.actor,
278        ts: ev.ts,
279        op: &ev.op,
280    })
281    .unwrap_or_default()
282}

284/// Events appended to `wal_ref` after `from` (None = the whole log),
285/// oldest first — `git rev-list <from>..<ref> --reverse`, then each
286/// commit's `event.json`. Err on plumbing failure so the caller can
287/// fall back (e.g. `from` GC'd after a compact).
288pub fn events_since(repo: &Path, wal_ref: &str, from: Option<&str>) -> io::Result<Vec<Event>> {
289    let range = match from {
290        Some(f) if !f.is_empty() => format!("{f}..{wal_ref}"),
291        _ => wal_ref.to_string(),
292    };
293    let commits = git(repo, &["rev-list", "--reverse", &range])?;
294    let mut out = Vec::new();
295    for c in commits.lines().filter(|l| !l.is_empty()) {
296        let json = git(repo, &["cat-file", "-p", &format!("{c}:event.json")])?;
297        let ev: RawEvent = serde_json::from_str(&json)
298            .map_err(|e| io::Error::other(format!("bad event in {c}: {e}")))?;
299        // Backfill id for events written before id assignment — the
300        // id *is* the event blob's content address (sturn-wal-git
301        // does the same via sturn_odb::object_id).
302        let id = if ev.id.is_empty() {
303            git(repo, &["rev-parse", &format!("{c}:event.json")]).unwrap_or_else(|_| c.to_string())
304        } else {
305            ev.id.clone()
306        };
307        let snapshot = serde_json::from_str::<OpTag>(ev.op.get())
308            .map(|t| t.op == "snapshot")
309            .unwrap_or(false);
310        out.push(Event {
311            id,
312            actor: ev.actor.clone(),
313            snapshot,
314            sig: ev.sig.clone(),
315            signing_bytes: signing_bytes(&ev),
316        });
317    }
318    Ok(out)
319}

321/// `refs/sturn/ws/<ws>` — the commit that ws's last snap produced.
322pub fn ws_tip(repo: &Path, ws: &str) -> Option<String> {
323    git(
324        repo,
325        &["rev-parse", "-q", "--verify", &format!("{WS_PREFIX}/{ws}")],
326    )
327    .ok()
328    .filter(|s| !s.is_empty())
329}

331/// Poller state — `"<repo>|<ws>"` → last-seen WAL head. Persisted
332/// under worktree.base so a restarted watcher resumes at the delta
333/// instead of replaying the log or skipping events it never saw.
334pub fn load_heads(path: &Path) -> HashMap<String, String> {
335    std::fs::read_to_string(path)
336        .ok()
337        .and_then(|s| serde_json::from_str(&s).ok())
338        .unwrap_or_default()
339}
340
341pub fn save_heads(path: &Path, heads: &HashMap<String, String>) -> io::Result<()> {
342    let tmp = path.with_extension("json.tmp");
343    std::fs::write(&tmp, serde_json::to_string(heads).unwrap_or_default())?;
344    std::fs::rename(&tmp, path)
345}

347/// The sturn `Source` impl — polls each `sturn = true` repo's clone
348/// for WAL movement (`refs/sturn/wal/<ws>`), reads new event
349/// commits, and on an `Op::Snapshot` marker emits the ws tip
350/// commit — the same queue every other source feeds. Polls refs,
351/// not the bus: the WAL is the state (the bus is a cache), so this
352/// needs no daemon and catches up across restarts via the persisted
353/// heads file.
354pub struct SturnSource {
355    /// (repo name, clone dir) per sturn-enabled repo — one poll
356    /// entry each; a repo's jobs share the clone.
357    pub repos: Vec<(String, PathBuf)>,
358    /// Remote to fetch `refs/sturn/*` from — empty = local-only.
359    pub remote: String,
360    /// WAL head poll interval, ms.
361    pub poll_ms: u64,
362    /// Persisted per-ws heads file — `"<repo>|<ws>"` → last-seen
363    /// WAL head.
364    pub heads_path: PathBuf,
365    /// Event-id dedup — compaction rebuilds WAL commits, but event
366    /// ids are blob addresses and survive the rewrite.
367    seen: Mutex<bupha_core::Seen>,
368}
369
370impl SturnSource {
371    /// `repos` = one entry per repo NAME (jobs share the clone);
372    /// `heads_path` lives under `worktree.base` on the caller.
373    pub fn new(
374        repos: Vec<(String, PathBuf)>,
375        remote: String,
376        poll_ms: u64,
377        heads_path: PathBuf,
378    ) -> Self {
379        SturnSource {
380            repos,
381            remote,
382            poll_ms,
383            heads_path,
384            seen: Mutex::new(bupha_core::Seen::default()),
385        }
386    }
387}
388
389impl Source for SturnSource {
390    fn name(&self) -> &str {
391        "sturn-wal"
392    }
393
394    fn run(&self, emit: Emit, stop: Stop) {
395        let mut heads = load_heads(&self.heads_path);
396        if self.repos.is_empty() {
397            eprintln!("[trigger.sturn] set but no [[repo]] has sturn = true — poller disabled");
398            return;
399        }
400        println!(
401            "sturn trigger polling {} repo(s) every {}ms",
402            self.repos.len(),
403            self.poll_ms
404        );
405        loop {
406            if stop() {
407                return;
408            }
409            for (name, dir) in &self.repos {
410                let meta = meta(dir);
411                // Signer policy resolved per tick — sturn.toml may
412                // gain [signing] between polls.
413                let signers = signers_for(&meta);
414                if !fetch(dir, &self.remote) {
415                    eprintln!("sturn fetch {name}: failed — retrying next tick");
416                    continue;
417                }
418                for (ws, head) in wal_heads(dir, &meta.wal_prefix) {
419                    let key = format!("{name}|{ws}");
420                    let prev = heads.get(&key).cloned();
421                    if prev.as_deref() == Some(head.as_str()) {
422                        continue;
423                    }
424                    let wal_ref = format!("{}/{ws}", meta.wal_prefix);
425                    // prev GC'd (compact + prune) → fall back to the
426                    // whole log; event ids dedup the replay in `seen`.
427                    let evs = events_since(dir, &wal_ref, prev.as_deref())
428                        .or_else(|_| events_since(dir, &wal_ref, None))
429                        .unwrap_or_default();
430                    for ev in evs {
431                        if !ev.snapshot {
432                            continue;
433                        }
434                        if !self.seen.lock().unwrap().insert(format!("sturn:{}", ev.id)) {
435                            continue;
436                        }
437                        let Some(sha) = ws_tip(dir, &ws) else {
438                            continue;
439                        };
440                        // `actor` is self-asserted in the WAL — it only
441                        // becomes an attested sender when the event's
442                        // SSH signature verifies against the repo's
443                        // allowed_signers. Without a signer policy the
444                        // sender stays a claim; with one, an unsigned
445                        // or bad-sig event goes out unattested ("").
446                        let sender = match &signers {
447                            Some(p) => ev.verified_actor(p).unwrap_or_else(|| {
448                                eprintln!(
449                                    "sturn event {}: unsigned/bad sig — \
450                                     unattested sender",
451                                    ev.id
452                                );
453                                String::new()
454                            }),
455                            None => ev.actor.clone(),
456                        };
457                        let git_ref = branch_ref(&meta);
458                        match emit(&Push {
459                            repo: name.clone(),
460                            git_ref: git_ref.clone(),
461                            sha: sha.clone(),
462                            sender,
463                        }) {
464                            Verdict::Queued => {
465                                println!("sturn snap {name} ws={ws} → {:.12} queued", sha)
466                            }
467                            Verdict::NoMatch => {
468                                println!("sturn snap {name} ws={ws} — no job watches {git_ref}")
469                            }
470                            _ => {}
471                        }
472                    }
473                    heads.insert(key, head);
474                }
475            }
476            let _ = save_heads(&self.heads_path, &heads);
477            std::thread::sleep(Duration::from_millis(self.poll_ms));
478        }
479    }
480}

486    #[test]
487    fn signing_bytes_match_sturn_canonical_form() {
488        // sturn-core signs {id:"",ws,actor,ts,op} — sig stripped, id
489        // cleared, compact JSON in field order. The op object must
490        // ride through verbatim (RawValue), not re-serialized.
491        let ev: RawEvent = serde_json::from_str(
492            r#"{"id":"abc","ws":"w1","actor":"alice","ts":42,"op":{"op":"snapshot","message":"m"},"sig":"SIG"}"#,
493        )
494        .unwrap();
495        let bytes = signing_bytes(&ev);
496        assert_eq!(
497            String::from_utf8(bytes).unwrap(),
498            r#"{"id":"","ws":"w1","actor":"alice","ts":42,"op":{"op":"snapshot","message":"m"}}"#
499        );
500    }

502    #[test]

507        .unwrap();
508        assert_eq!(ev.sig.as_deref(), Some("s"));
