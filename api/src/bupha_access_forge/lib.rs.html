1//! Forge impl of `Access` — "does the pusher have write access to
2//! this repo?" via the forge's collaborator-permission API. Same
3//! trust model hosted CI uses: policy lives in the forge, not a file.
4//!
5//! github impl: `GET /repos/{repo}/collaborators/{user}/permission`
6//! (200 = allowed, 404 = not a collaborator → deny).
7
8use bupha_access::Access;
9use std::collections::HashMap;
10use std::sync::Mutex;
11
12/// Checks the forge's collaborator permission for (sender, repo) —
13/// caches positive results in-process (a collaborator doesn't lose
14/// write mid-run realistically).
15pub struct ForgeAccess {
16    pub api_base: String,
17    pub token: String,
18    pub min_role: String,
19    cache: Mutex<HashMap<(String, String), bool>>,
20}
21
22impl ForgeAccess {
23    pub fn new(api_base: &str, token: &str) -> Self {
24        Self {
25            api_base: api_base.to_string(),
26            token: token.to_string(),
27            min_role: "push".into(),
28            cache: Mutex::new(HashMap::new()),
29        }
30    }
31}
32
33/// Percent-encode one URL path segment — sender/repo can come from
34/// signed forge payloads (well-formed logins) or self-asserted
35/// source identities (sturn actors, arbitrary strings). Unreserved
36/// chars only, everything else is %-escaped so a hostile value can't
37/// reshape the request path or smuggle a query.
38fn encode_segment(s: &str) -> String {
39    // "." and ".." are legal unreserved chars but dangerous as whole
40    // path segments — force-encode them.
41    if s == "." || s == ".." {
42        return s.bytes().map(|b| format!("%{b:02X}")).collect();
43    }
44    s.bytes()
45        .map(|b| {
46            if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_' | b'.' | b'~') {
47                (b as char).to_string()
48            } else {
49                format!("%{b:02X}")
50            }
51        })
52        .collect()
53}
54
55impl Access for ForgeAccess {
56    fn allow(&self, sender: &str, repo: &str) -> bool {
57        let key = (sender.to_string(), repo.to_string());
58        if let Some(&ok) = self.cache.lock().unwrap().get(&key) {
59            return ok;
60        }
61        // github: GET collaborator permission → role ∈ {admin, write}
62        let url = format!(
63            "{}/repos/{}/collaborators/{}/permission",
64            self.api_base,
65            // "owner/name" is two path segments — encode each.
66            repo.split('/')
67                .map(encode_segment)
68                .collect::<Vec<_>>()
69                .join("/"),
70            encode_segment(sender)
71        );
72        let ok = ureq::get(&url)
73            .header("Authorization", &format!("Bearer {}", self.token))
74            .header("Accept", "application/vnd.github+json")
75            .header("User-Agent", "bupha")
76            .call()
77            .map(|r| {
78                let body: serde_json::Value = r.into_body().read_json().unwrap_or_default();
79                let p = body
80                    .get("permission")
81                    .and_then(|v| v.as_str())
82                    .unwrap_or_default();
83                matches!(p, "admin" | "write" | "maintain")
84            })
85            .unwrap_or(false);
86        // Only positives are cached — a denied sender may legitimately
87        // gain access later (invite accepted), and a transient API
88        // failure must not pin a false deny for the process's life.
89        if ok {
90            self.cache.lock().unwrap().insert(key, true);
91        }
92        ok
93    }
94}

96#[cfg(test)]
97mod tests {
98    use super::encode_segment;
99
100    #[test]
101    fn segment_encoding() {
102        assert_eq!(encode_segment("alice"), "alice");
103        assert_eq!(encode_segment("a-b_c.d~e"), "a-b_c.d~e");
104        // "../x" is ONE segment — "/" encodes, leaving ".." harmless.
105        assert_eq!(encode_segment("../x"), "..%2Fx");
106        // A bare ".." segment would traverse — force-encode it.
107        assert_eq!(encode_segment(".."), "%2E%2E");
108        assert_eq!(encode_segment("."), "%2E");
109        assert_eq!(encode_segment("a?b=c"), "a%3Fb%3Dc");
110        assert_eq!(encode_segment("a b"), "a%20b");
111    }
