6use serde::{Deserialize, Serialize};
7use std::collections::{BTreeMap, HashSet, VecDeque};
8use std::io;
9use std::sync::Mutex;

11/// One repo's persisted slice: pending backlog + the sha that was
12/// running when the process stopped (re-queued at the front).
13#[derive(Debug, Default, Clone, Serialize, Deserialize)]
14/// Serializable per-repo queue snapshot — what a store writes.
15pub struct PersistedQueue {
16    pub pending: VecDeque<String>,
17    pub running: Option<String>,
18}
19
20/// Queue persistence — load/save the full per-repo map.
21pub trait QueueStore: Send + Sync {
22    /// Repo name → persisted queue. Missing store = empty map.
23    fn load(&self) -> io::Result<BTreeMap<String, PersistedQueue>>;
24    /// Must be crash-safe — tmp file + rename, never a torn write.
25    fn persist(&self, queues: &BTreeMap<String, PersistedQueue>) -> io::Result<()>;
26    /// Persist ONE repo's queue without touching other keys — a
27    /// watcher only holds its own repos' queues, so a whole-map
28    /// write would clobber a peer watcher's entries in a shared
29    /// store. Default: read-modify-write via load+persist (correct
30    /// for disjoint writers; stores with real per-key ops override).
31    fn persist_key(&self, key: &str, q: &PersistedQueue) -> io::Result<()> {
32        let mut m = self.load()?;
33        m.insert(key.to_string(), q.clone());
34        self.persist(&m)
35    }
36}

38/// A job pushed for the fleet — the sha to run plus the labels a
39/// claiming worker must carry.
40#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
41pub struct Job {
42    /// Watched key — "owner/name" or "owner/name:job".
43    pub key: String,
44    pub sha: String,
45    /// Label requirements — a worker must carry ALL of them to
46    /// claim. Empty = claimable by anyone.
47    #[serde(default)]
48    pub runs_on: Vec<String>,
49}
50
51/// A key a worker is willing to run — its queue policy decides
52/// claim order (newest-only for `latest`, FIFO for `all`).
53pub struct KeyPolicy {
54    pub key: String,
55    /// `queue = "latest"` semantics: only the newest entry matters;
56    /// older ones are superseded and pruned on scan.
57    pub latest: bool,
58}
59
60/// What a claim scan produced.
61pub enum Claim {
62    /// A job was leased for this worker; `superseded` entries were
63    /// pruned and deserve honest "superseded" statuses.
64    Won { job: Job, superseded: Vec<Job> },
65    /// Nothing claimable, but stale entries were pruned — report
66    /// them as superseded, then scan again.
67    Pruned(Vec<Job>),
68}
69
70/// Shared pull queue — N producers (watchers) enqueue, M consumers
71/// (workers) claim with a lease. A dead worker's leased job becomes
72/// claimable again when its lease expires; `complete` settles it.
73pub trait WorkQueue: Send + Sync {
74    /// Push a job. `false` = the sha is already queued (dedup).
75    fn enqueue(&self, job: &Job) -> io::Result<bool>;
76    /// Scan `keys` for a claimable job: not done, labels satisfied
77    /// by `worker_labels`, lease won. Latest-policy keys prune
78    /// entries superseded by the claimed/newest sha.
79    fn claim_next(
80        &self,
81        worker: &str,
82        worker_labels: &[String],
83        keys: &[KeyPolicy],
84    ) -> io::Result<Option<Claim>>;
85    /// Settle a claimed job — marks done, removes the list entry,
86    /// releases the lease. No-op if `worker` no longer holds it.
87    fn complete(&self, job: &Job, worker: &str) -> io::Result<()>;
88}

90/// In-memory WorkQueue — same-process semantics for single-box use
91/// and tests. Claims can't expire (no TTL), so a claimed entry
92/// leaves the list until completed.
93#[derive(Default)]
94pub struct MemWorkQueue {
95    lists: Mutex<BTreeMap<String, VecDeque<Job>>>,
96    held: Mutex<HashSet<(String, String)>>,
97    done: Mutex<HashSet<(String, String)>>,
98}
99
100fn labels_fit(job: &Job, worker_labels: &[String]) -> bool {
101    job.runs_on.iter().all(|r| worker_labels.contains(r))
102}
103
104impl WorkQueue for MemWorkQueue {
105    fn enqueue(&self, job: &Job) -> io::Result<bool> {
106        let mut l = self.lists.lock().unwrap();
107        let q = l.entry(job.key.clone()).or_default();
108        if q.iter().any(|j| j.sha == job.sha) {
109            return Ok(false);
110        }
111        q.push_back(job.clone());
112        Ok(true)
113    }
114
115    fn claim_next(
116        &self,
117        worker: &str,
118        worker_labels: &[String],
119        keys: &[KeyPolicy],
120    ) -> io::Result<Option<Claim>> {
121        let _ = worker; // in-process claims aren't fenced
122        let mut lists = self.lists.lock().unwrap();
123        let mut held = self.held.lock().unwrap();
124        let done = self.done.lock().unwrap();
125        for kp in keys {
126            let Some(q) = lists.get_mut(&kp.key) else {
127                continue;
128            };
129            // Prune completed entries off the front as we scan.
130            q.retain(|j| !done.contains(&(kp.key.clone(), j.sha.clone())));
131            if kp.latest {
132                // Newest live entry is the only candidate; older ones
133                // are superseded by it (or already pruned if it's held).
134                let Some(i) = q
135                    .iter()
136                    .rposition(|j| !held.contains(&(kp.key.clone(), j.sha.clone())))
137                else {
138                    continue;
139                };
140                if !labels_fit(&q[i], worker_labels) {
141                    continue;
142                }
143                let job = q[i].clone();
144                let superseded: Vec<Job> = q.drain(..i).collect();
145                held.insert((kp.key.clone(), job.sha.clone()));
146                return Ok(Some(Claim::Won { job, superseded }));
147            }
148            // `all`: oldest claimable entry wins.
149            for j in q.iter() {
150                if held.contains(&(kp.key.clone(), j.sha.clone())) {
151                    continue;
152                }
153                if !labels_fit(j, worker_labels) {
154                    continue;
155                }
156                held.insert((kp.key.clone(), j.sha.clone()));
157                return Ok(Some(Claim::Won {
158                    job: j.clone(),
159                    superseded: vec![],
160                }));
161            }
162        }
163        Ok(None)
164    }
165
166    fn complete(&self, job: &Job, worker: &str) -> io::Result<()> {
167        let _ = worker;
168        self.lists
169            .lock()
170            .unwrap()
171            .get_mut(&job.key)
172            .map(|q| q.retain(|j| j.sha != job.sha));
173        self.held
174            .lock()
175            .unwrap()
176            .remove(&(job.key.clone(), job.sha.clone()));
177        self.done
178            .lock()
179            .unwrap()
180            .insert((job.key.clone(), job.sha.clone()));
181        Ok(())
182    }
183}
