1//! script — render expanded [`Job`]s into one bash program. Pure
2//! string building; the runner spawns it with `bash -c` in the
3//! worktree. Jobs are scheduled by a ready-queue over the `needs`
4//! DAG: a job group launches as soon as EVERY group it needs has
5//! settled — never waiting on unrelated jobs at the same depth.
6//! Groups that can overlap an unrelated group run as `set -m`
7//! background process groups, each in a private `git worktree`
8//! checkout (the same isolation parallel matrix legs use); groups
9//! comparable with every other group provably run alone and keep
10//! the sequential shared-worktree path. Matrix legs of one job id
11//! additionally fan out inside their group, throttled by
12//! `strategy.max-parallel`, or fall back to the sequential path
13//! when `max-parallel: 1`.
14

18/// Everything the generated program needs from the run context.
19pub struct ScriptCtx<'a> {
20    /// Absolute worktree path (exported as BUPHA_WORKDIR).
21    pub workdir: &'a str,
22    pub sha: &'a str,
23    pub git_ref: &'a str,
24    /// "owner/name".
25    pub repository: &'a str,
26    /// Per-run dir (outside the worktree) each expanded job drops a
27    /// `<ji>` status record into for the runner's post-run
28    /// `status_reports` pass. Empty disables reporting.
29    pub report: &'a str,
30}

32/// Single-quote a literal for safe embedding in bash.
33pub(crate) fn sq(s: &str) -> String {
34    format!("'{}'", s.replace('\'', "'\\''"))
35}

37/// `cd` target: `ws` (a quoted bash word for the workspace root —
38/// the worktree for sequential jobs, a per-leg `git worktree` for
39/// parallel matrix legs) + optional repo-relative working-directory.
40/// Deferred `${{ }}` output refs become `$(cat …)` lookups — the
41/// pieces are quoted so the result stays one word.
42fn cd_target(ws: &str, wd: Option<&str>, ji: usize, wf: &str) -> Result<String, String> {
43    match wd {
44        Some(w) if !w.is_empty() => Ok(format!("{ws}/{}", shq_exprs(w, ji, wf)?)),
45        _ => Ok(ws.to_string()),
46    }
47}

49/// `export` lines for an env list. Values may carry deferred output
50/// refs — each becomes a quoted `$(cat …)` piece inside the word.
51fn exports(pairs: &[(String, String)], ji: usize, wf: &str) -> Result<String, String> {
52    pairs
53        .iter()
54        .map(|(k, v)| Ok(format!("export {k}={}", shq_exprs(v, ji, wf)?)))
55        .collect::<Result<Vec<_>, String>>()
56        .map(|lines| lines.join("\n"))
57}

59/// Pick a heredoc delimiter the script can't collide with.
60fn heredoc_delim(script: &str, base: &str) -> String {
61    let mut d = base.to_string();
62    while script.lines().any(|l| l == d) {
63        d.push('X');
64    }
65    d
66}

68/// The shell GitHub would use for a `run` step: `shell:` maps to a
69/// known template, default is bash with GH's flags.
70fn shell_cmd(shell: Option<&str>) -> Result<&'static str, String> {
71    // A `shell:` value can't carry a `$(cat …)` lookup — it names
72    // the interpreter, not a word. Deferred output refs here are a
73    // hard error rather than a silently literal `${{ … }}`.
74    if shell.map(|s| s.contains("${{")).unwrap_or(false) {
75        return Err("`shell:` cannot contain `${{ }}` expressions".to_string());
76    }
77    match shell {
78        None | Some("bash") => Ok("bash --noprofile --norc -eo pipefail"),
79        Some("sh") => Ok("sh -e"),
80        Some(other) => Err(format!(
81            "unsupported `shell: {other}` — v1 supports bash (default) and sh"
82        )),
83    }
84}

98/// Compile a [`Cond`] to a bash and-or list whose exit status is
99/// the truth value (0 = true). Used directly in `if …; then`.
100fn cond_bool(c: &Cond, cx: &CondCx) -> Result<String, String> {
101    Ok(match c {
102        Cond::Lit(s) => format!("bupha_truthy {}", sq(s)),
103        Cond::Dyn(d) => format!("bupha_truthy {}", dyn_word(d, cx)?),
104        Cond::Call { name, args } => match name.as_str() {
105            "success" => format!("bupha_needs_ok{}", idx_args(cx.needs)),
106            "failure" => format!("bupha_needs_failed{}", idx_args(cx.needs)),
107            "cancelled" => format!("bupha_needs_cancelled{}", idx_args(cx.needs)),
108            "always" => ":".to_string(),
109            "contains" | "startswith" | "endswith" => {
110                let (a, b) = (&args[0], &args[1]);
111                // GHA's contains/startsWith/endsWith are
112                // case-insensitive — `${var,,}` lowercases.
113                let pat = match name.as_str() {
114                    "contains" => "*\"${_b_i,,}\"*",
115                    "startswith" => "\"${_b_i,,}\"*",
116                    _ => "*\"${_b_i,,}\"",
117                };
118                format!(
119                    "( _b_s={}; _b_i={}; [[ \"${{_b_s,,}}\" == {pat} ]] )",
120                    cond_word(a, cx)?,
121                    cond_word(b, cx)?
122                )
123            }
124            _ => return Err(format!("unsupported `if` function `{name}()`")),
125        },
126        Cond::Bin { op, l, r } => match op {
127            BinOp::And => format!("( {} ) && ( {} )", cond_bool(l, cx)?, cond_bool(r, cx)?),
128            BinOp::Or => format!("( {} ) || ( {} )", cond_bool(l, cx)?, cond_bool(r, cx)?),
129            BinOp::Eq | BinOp::Ne => {
130                let cmp = if *op == BinOp::Eq { "==" } else { "!=" };
131                format!(
132                    "( _b_a={}; _b_b={}; [[ \"${{_b_a,,}}\" {cmp} \"${{_b_b,,}}\" ]] )",
133                    cond_word(l, cx)?,
134                    cond_word(r, cx)?
135                )
136            }
137        },
138        Cond::Not(inner) => format!("! ( {} )", cond_bool(inner, cx)?),
139    })
140}

142/// Compile a [`Cond`] to a bash *word* evaluating to its string
143/// value. Boolean-valued nodes yield `true`/`false` — matching how
144/// GHA stringifies booleans in comparisons.
145fn cond_word(c: &Cond, cx: &CondCx) -> Result<String, String> {
146    Ok(match c {
147        Cond::Lit(s) => sq(s),
148        Cond::Dyn(d) => dyn_word(d, cx)?,
149        _ => format!(
150            "\"$(if {}; then echo true; else echo false; fi)\"",
151            cond_bool(c, cx)?
152        ),
153    })
154}

161/// A deferred [`DynRef`] as a quoted bash word.
162fn dyn_word(d: &DynRef, cx: &CondCx) -> Result<String, String> {
163    Ok(match d {
164        DynRef::NeedOutput { job, name } => {
165            output_lookup(&format!("needs.{job}.outputs.{name}"), cx.ji, true, cx.wf)?
166        }
167        DynRef::StepOutput { step, name } => {
168            output_lookup(&format!("steps.{step}.outputs.{name}"), cx.ji, true, cx.wf)?
169        }
170        DynRef::NeedResult { job } => {
171            format!("\"{}\"", needs_result_lookup(job, cx.wf))
172        }
173        DynRef::StepConclusion { step } => {
174            // Step `run:` scripts execute in a child `bash` fed by
175            // a heredoc — shell functions don't cross that boundary,
176            // so conclusions live in files like outputs do.
177            format!(
178                "\"$(cat \"$BUPHA_GHA_TMP/stepconcl/{}/{}\" 2>/dev/null || :)\"",
179                cx.ji, step
180            )
181        }
182        DynRef::Secret { name } => secret_lookup(name),
183    })
184}

186/// A bound reusable-workflow secret as a quoted bash word:
187/// `printenv <env-name>` at run time — the value never enters the
188/// generated script (the `secrets: inherit` / `${{ secrets.X }}`

191fn secret_lookup(name: &str) -> String {
192    format!("\"$(printenv {} 2>/dev/null || :)\"", sq(name))

195/// Render all jobs into one bash script. Scheduling is a
196/// ready-queue over the `needs` DAG: a job group launches as soon
197/// as every group it needs has *settled* (its `jobsynced/<g>`
198/// marker exists — written only after the parent's status fold, so
199/// gates and `needs.*` lookups always read final state). There are
200/// no depth barriers — an unrelated straggler never delays a
201/// dependent's launch. A group that is incomparable with at least
202/// one other group in the DAG (neither ancestor nor descendant —
203/// the two can genuinely overlap) launches as a `set -m` background
204/// subshell in a private `git worktree`; a group comparable with
205/// every other group provably runs alone and keeps the sequential
206/// shared-worktree path inline in the scheduler loop. Matrix legs
207/// of one job id — adjacent expanded jobs — either fan out as
208/// background process groups inside their group (each leg in a
209/// private `git worktree`, throttled by `strategy.max-parallel`)
210/// or run sequentially when the group has one leg or
211/// `max-parallel: 1`. Cross-group concurrency is capped by
212/// `$BUPHA_GHA_MAX_PARALLEL` (0/unset = unlimited).
213pub fn render(jobs: &[Job], ctx: &ScriptCtx) -> Result<String, String> {
214    // job key → every expanded index (matrix legs) — feeds
215    // `BUPHA_GHA_JOBIDX`, the runtime table the `bupha_job_failed`
216    // fail-fast helper scans. Keyed by workflow+id: `needs` never
217    // crosses workflow files, but two workflows can declare the
218    // same job id and their aggregates must not collide.
219    let mut job_idx: HashMap<String, Vec<usize>> = HashMap::new();
220    for (i, j) in jobs.iter().enumerate() {
221        job_idx.entry(job_key(j)).or_default().push(i);
222    }
223    // Consecutive runs of same-(workflow,id) expanded jobs are one
224    // matrix group. Topo order keeps a job's legs adjacent: legs
225    // share `needs`, so they become ready in the same pass and emit
226    // in index order — nothing can interleave.
227    let mut groups: Vec<Vec<usize>> = Vec::new();
228    let mut i = 0;
229    while i < jobs.len() {
230        let mut e = i + 1;
231        while e < jobs.len() && jobs[e].id == jobs[i].id && jobs[e].workflow == jobs[i].workflow {
232            e += 1;
233        }
234        groups.push((i..e).collect());
235        i = e;
236    }
237    // A group fans out unless it's a single leg or pinned to
238    // `max-parallel: 1` (the sequential path, kept for workflows
239    // that rely on legs sharing the worktree).
240    let is_parallel = |g: usize| -> bool {
241        let legs = &groups[g];
242        legs.len() > 1 && jobs[legs[0]].max_parallel != Some(1)
243    };
244    let group_parallel: Vec<bool> = (0..groups.len()).map(&is_parallel).collect();
245    let any_parallel = group_parallel.iter().any(|p| *p);
246    // Job-level `timeout-minutes` runs the body under a watchdog —
247    // monitor mode gives each background job its own process group
248    // so `kill -- -$pid` takes the whole tree down. Parallel matrix
249    // legs need it too: fail-fast kills leg groups by pgid. So do
250    // background-scheduled groups: group subshells host their own
251    // `&` jobs.
252    let any_job_timeout = jobs.iter().any(|j| j.timeout_seconds.is_some());
253
254    // --- needs-DAG over job groups ------------------------------------
255    // expanded-job index → its group, then per-group dependency
256    // edges (a group depends on every group containing one of its
257    // legs' `needs` targets). Topo order keeps every dependency
258    // edge pointing at an earlier group.
259    let gcount = groups.len();
260    let mut group_of = vec![0usize; jobs.len()];
261    for (g, legs) in groups.iter().enumerate() {
262        for &i in legs {
263            group_of[i] = g;
264        }
265    }
266    // gdeps[g] = sorted unique groups g must wait for (also the
267    // readiness probe's argument list).
268    let mut gdeps: Vec<Vec<usize>> = vec![Vec::new(); gcount];
269    for (g, legs) in groups.iter().enumerate() {
270        let mut set = BTreeSet::new();
271        for &i in legs {
272            for &d in &jobs[i].needs {
273                let dg = group_of[d];
274                if dg != g {
275                    set.insert(dg);
276                }
277            }
278        }
279        gdeps[g] = set.into_iter().collect();
280    }
281    // anc[g] = bitset of g's transitive dependencies (incl. g) —
282    // deps always point backward, so one forward pass with
283    // bitset-union suffices.
284    let words = gcount.div_ceil(64);
285    let mut anc = vec![vec![0u64; words]; gcount];
286    for g in 0..gcount {
287        anc[g][g / 64] |= 1 << (g % 64);
288        for &d in &gdeps[g] {
289            debug_assert!(d < g, "needs edge must point backward in topo order");
290            // d < g: split the borrow so dst (anc[g]) and src
291            // (anc[d]) can be zipped mutably/immutably.
292            let (head, tail) = anc.split_at_mut(g);
293            for (dst, src) in tail[0].iter_mut().zip(head[d].iter()) {
294                *dst |= *src;
295            }
296        }
297    }
298    let is_anc = |set: &[u64], i: usize| (set[i / 64] >> (i % 64)) & 1 == 1;
299    // A group may overlap another iff they are incomparable —
300    // neither is a transitive dependency of the other. Only those
301    // groups need private worktrees + background subshells; a group
302    // comparable with every other provably runs alone (when it
303    // becomes ready, every other group is either settled or not
304    // yet launchable), so it keeps the shared-worktree path.
305    let mut may_overlap = vec![false; gcount];
306    for g in 0..gcount {
307        for h in 0..gcount {
308            if h != g && !is_anc(&anc[h], g) && !is_anc(&anc[g], h) {
309                may_overlap[g] = true;
310                break;
311            }
312        }
313    }
314    let any_bg = may_overlap.iter().any(|m| *m);
315
316    let mut out = String::with_capacity(8192);
317    out.push_str("#!/usr/bin/env bash\n");
318    out.push_str("# Generated by bupha gha runner — .github/workflows as CI.\n");
319    out.push_str("set -uo pipefail\n");
320    if any_job_timeout || any_parallel || any_bg {
321        out.push_str("set -m\n");
322    }
323    out.push_str(&format!("export GITHUB_EVENT_NAME={}\n", sq("push")));
324    out.push_str(&format!("export GITHUB_SHA={}\n", sq(ctx.sha)));
325    out.push_str(&format!("export GITHUB_REF={}\n", sq(ctx.git_ref)));
326    out.push_str(&format!(
327        "export GITHUB_REF_NAME={}\n",
328        sq(short_ref(ctx.git_ref))
329    ));
330    out.push_str(&format!(
331        "export GITHUB_REPOSITORY={}\n",
332        sq(ctx.repository)
333    ));
334    out.push_str(&format!("export BUPHA_WORKDIR={}\n", sq(ctx.workdir)));
335    out.push_str("declare -a BUPHA_GHA_STATUS=()\n");
336    out.push_str("declare -A BUPHA_GHA_JOBIDX=()\n");
337    for (id, idxs) in &job_idx {
338        let list = idxs
339            .iter()
340            .map(usize::to_string)
341            .collect::<Vec<_>>()
342            .join(" ");
343        out.push_str(&format!("BUPHA_GHA_JOBIDX[{}]={}\n", sq(id), sq(&list)));
344    }
345    out.push_str("BUPHA_GHA_FAILS=0\n");
346    // Per-run scratch for the $GITHUB_* command files — the dialect
347    // equivalent of the runner's per-job temp dir. One tmp dir for
348    // the whole run; each job gets its own subdir so env-file writes
349    // scope to the job like on GitHub. `legs/`+`legexit/` are the
350    // parallel-matrix workspaces and per-leg exit records;
351    // `jobs/` holds the private worktrees of concurrently-scheduled
352    // job groups; `joboutpick/` records which leg's outputs
353    // `needs.*` reads.
354    out.push_str("BUPHA_GHA_TMP=$(mktemp -d)\n");
355    out.push_str("export BUPHA_GHA_TMP\n");
356    out.push_str("mkdir -p \"$BUPHA_GHA_TMP/jobstatus\" \"$BUPHA_GHA_TMP/stepconcl\" \"$BUPHA_GHA_TMP/jobresult\" \"$BUPHA_GHA_TMP/joboutpick\" \"$BUPHA_GHA_TMP/legexit\" \"$BUPHA_GHA_TMP/legs\" \"$BUPHA_GHA_TMP/jobs\"\n");
357    if any_parallel || any_bg {
358        // Backstop for job worktrees: normally each is
359        // `git worktree remove`d at harvest; if the script dies
360        // mid-run the EXIT trap removes whatever legs/*+jobs/*
361        // dirs remain so the common git dir doesn't leak prunable
362        // entries. Unmatched globs stay literal and fail the
363        // `[ -d ]` guard.
364        out.push_str("trap 'for _d in \"$BUPHA_GHA_TMP\"/legs/*/ \"$BUPHA_GHA_TMP\"/jobs/*/; do [ -d \"$_d\" ] && git -C \"$BUPHA_WORKDIR\" worktree remove --force \"${_d%/}\" 2>/dev/null; done; rm -rf \"$BUPHA_GHA_TMP\"' EXIT\n");
365    } else {
366        out.push_str("trap 'rm -rf \"$BUPHA_GHA_TMP\"' EXIT\n");
367    }
368    out.push_str("export BUPHA_GHA_OUTPUTS=\"$BUPHA_GHA_TMP/outputs.txt\"\n");
369    out.push_str(": > \"$BUPHA_GHA_OUTPUTS\"\n");
370    if !ctx.report.is_empty() {
371        out.push_str(&format!(
372            "export BUPHA_GHA_REPORT={}\nmkdir -p \"$BUPHA_GHA_REPORT\"\n",
373            sq(ctx.report)
374        ));
375    }
376    out.push_str(ABSORB_FNS);
377    if any_parallel {
378        // Parallel-matrix scheduler state — each matrix group gets
379        // its own copy inside its subshell (top-level groups run in
380        // the main shell), so one run/fail pair per group suffices.
381        out.push_str(
382            "BUPHA_MP_RUN=\"\"\nBUPHA_MP_FAIL=0\ndeclare -A BUPHA_MP_PID=() BUPHA_MP_WD=()\n",
383        );
384        out.push_str(MP_FNS);
385    }
386    if any_bg {
387        out.push_str(JP_FNS);
388    }
389
390    if any_bg {
391        out.push_str(&render_scheduler(
392            jobs,
393            &groups,
394            &group_parallel,
395            &gdeps,
396            &may_overlap,
397            ctx,
398        )?);
399    } else {
400        // No incomparable pair — the needs DAG is a total order,
401        // so every group runs alone on the shared worktree.
402        for g in 0..groups.len() {
403            if group_parallel[g] {
404                out.push_str(&render_parallel_group(jobs, &groups[g], ctx, false)?);
405            } else {
406                for &i in &groups[g] {
407                    out.push_str(&render_seq_job(
408                        &jobs[i],
409                        i,
410                        &groups[g],
411                        ctx,
412                        &sq(ctx.workdir),
413                    )?);
414                }
415            }
416        }
417    }
418    out.push_str("if [ \"$BUPHA_GHA_FAILS\" -gt 0 ]; then\n  echo \"gha: workflow(s) failed\"\n  exit 1\nfi\necho \"gha: all jobs passed\"\n");
419    Ok(out)
420}

422/// One expanded job's sequential block: run gate (fail-fast cancel
423/// outermost when `strategy.fail-fast`), the `( set -e; … )` body
424/// optionally under the `timeout-minutes` watchdog, `outputs:`
425/// publication, the group's `needs.*` aggregates, and the per-job
426/// status report. `ws` is the quoted workspace root — the shared
427/// worktree on the sequential path, a private `git worktree` inside
428/// a concurrent level's group subshell. Inside a subshell the
429/// `BUPHA_GHA_STATUS`/`BUPHA_GHA_FAILS` writes stay local; the
430/// `jobstatus/`+aggregate files `bupha_set_status` & co. write are
431/// the channel the parent settles from.
432fn render_seq_job(
433    j: &Job,
434    i: usize,
435    legs: &[usize],
436    ctx: &ScriptCtx,
437    ws: &str,
438) -> Result<String, String> {
439    let mut out = String::new();
440    let label = job_label(j);
441    out.push_str(&format!(
442        "echo {}\n",
443        shq_exprs(&format!("gha: job '{label}' starting"), i, &j.workflow)?
444    ));
445
446    // The run gate: `if:` when present (it already carries the
447    // implicit success() when the author didn't name a status
448    // function), else the implicit all-needs-passed check.
449    let gate = job_gate(j, i)?;
450    let fail_fast = j.fail_fast;
451
452    // Gate opening — three shapes: fail-fast cancel outermost,
453    // then the run gate, else unconditional.
454    if fail_fast {
455        out.push_str(&format!(
456            "if bupha_job_failed {}; then\n  bupha_set_status {i} 3\n  echo {}\n",
457            sq(&job_key(j)),
458            shq_exprs(
459                &format!("gha: job '{label}' cancelled — fail-fast"),
460                i,
461                &j.workflow
462            )?
463        ));
464        if let Some((g, _)) = &gate {
465            out.push_str(&format!("elif {g}; then\n"));
466        } else {
467            out.push_str("else\n");
468        }
469    } else if let Some((g, _)) = &gate {
470        out.push_str(&format!("if {g}; then\n"));
471    }
472
473    // Job body — the same `( set -e; … )` + status + outputs
474    // block as before, optionally under a timeout watchdog.
475    let timed = j.timeout_seconds;
476    out.push_str("  (\n    set -e\n");
477    out.push_str(&job_body(j, i, ctx, ws)?);
478    if let Some(secs) = timed {
479        // Watchdog: the job body runs in a background process
480        // group (set -m); the killer sleeps N seconds then
481        // TERMs the whole group. `wait` still reports the real
482        // exit status — 143 when killed.
483        out.push_str(&format!(
484            "  ) &\n  bupha_pid=$!\n  ( sleep {secs}; touch \"$BUPHA_GHA_TMP/timeout_{i}\"; kill -TERM -- -\"$bupha_pid\" 2>/dev/null ) &\n  bupha_wd=$!\n  wait \"$bupha_pid\"\n  bupha_set_status {i} \"$?\"\n  kill \"$bupha_wd\" 2>/dev/null\n  wait \"$bupha_wd\" 2>/dev/null\n  if [ -f \"$BUPHA_GHA_TMP/timeout_{i}\" ]; then rm -f \"$BUPHA_GHA_TMP/timeout_{i}\"; echo {}; fi\n",
485            shq_exprs(&format!("gha: job '{label}' timed out"), i, &j.workflow)?
486        ));
487    } else {
488        out.push_str("  )\n");
489        out.push_str(&format!("  bupha_set_status {i} \"$?\"\n"));
490    }
491    // Job `outputs:` evaluate once the job's steps are done —
492    // steps.*.outputs files are already on disk, needs.*.outputs
493    // read the producers' jobouts/ dirs. Skipped/failed jobs
494    // publish nothing, matching GitHub.
495    let pass = job_output_lines(j, i)?;
496    out.push_str(&format!(
497        "  if [ \"${{BUPHA_GHA_STATUS[{i}]}}\" -eq 0 ]; then\n{pass}    echo {}\n  else\n    echo {}\n    BUPHA_GHA_FAILS=$((BUPHA_GHA_FAILS+1))\n  fi\n",
498        shq_exprs(&format!("gha: job '{label}' passed"), i, &j.workflow)?,
499        shq_exprs(&format!("gha: job '{label}' failed"), i, &j.workflow)?,
500    ));
501    if let Some((_, reason)) = &gate {
502        out.push_str(&format!(
503            "else\n  bupha_set_status {i} 2\n  echo {}\nfi\n",
504            shq_exprs(
505                &format!("gha: job '{label}' skipped — {reason}"),
506                i,
507                &j.workflow
508            )?
509        ));
510    } else if fail_fast {
511        out.push_str("fi\n");
512    }
513    // Refresh needs.<id>.result + the needs.*.outputs pick now
514    // that this leg's status is final — downstream jobs and
515    // steps read the aggregate files. Emitted per leg; both
516    // converge once the group's last leg settles.
517    let list = legs
518        .iter()
519        .map(usize::to_string)
520        .collect::<Vec<_>>()
521        .join(" ");
522    let sid = job_key(j);
523    out.push_str(&format!("bupha_agg_result {} {list}\n", sq(&sid)));
524    out.push_str(&format!("bupha_pick_out {} {list}\n", sq(&sid)));
525    // Per-job status record for the runner's post-run
526    // `status_reports` pass — every gate path has already
527    // settled this leg's jobstatus file by now.
528    out.push_str(&format!(
529        "bupha_report {i} {} {}\n",
530        sq(&status_suffix(j)),
531        shq_exprs(&j.label, i, &j.workflow)?
532    ));
533    Ok(out)
534}

536/// The ready-queue scheduler loop, emitted when at least one pair
537/// of job groups is incomparable in the `needs` DAG. Each pass:
538/// (1) reap every finished background group — fold leg statuses
539/// back (`bupha_sync_job`), run fail accounting, remove the private
540/// worktree, drop the `jobsynced/<g>` marker that unblocks
541/// dependents; (2) launch every unlaunched group whose dependency
542/// groups are all synced — background groups fork a `set -m`
543/// subshell in a private worktree (throttled by
544/// `$BUPHA_GHA_MAX_PARALLEL`, 0 = unlimited), while groups
545/// comparable with every other run inline on the shared worktree
546/// (they provably never overlap anything); (3) `wait -n` until the
547/// next subshell exits. `jobdone/<g>` is written by the subshell as
548/// its last act; `jobsynced/<g>` is written by the parent only
549/// after the status fold, so a dependent's fork-time
550/// `BUPHA_GHA_STATUS` snapshot and its `needs.*` file lookups
551/// always see final producer state. A subshell that dies without
552/// `jobdone` (crash, not SIGKILL — the subshell sets no trap) is
553/// detected via `kill -0` on its pid once `wait -n` reaps it and
554/// settled as failure leg-by-leg. A pass that launches nothing
555/// with nothing running means a `needs` cycle — impossible after
556/// plan's topo sort, but the guard exits loudly rather than hang.
557/// There is no cross-job fail-fast — matching GHA,
558/// `strategy.fail-fast` only cancels legs of its own job id.
559fn render_scheduler(
560    jobs: &[Job],
561    groups: &[Vec<usize>],
562    group_parallel: &[bool],
563    gdeps: &[Vec<usize>],
564    may_overlap: &[bool],
565    ctx: &ScriptCtx,
566) -> Result<String, String> {
567    let mut out = String::new();
568    out.push_str("mkdir -p \"$BUPHA_GHA_TMP/jobdone\" \"$BUPHA_GHA_TMP/jobsynced\"\n");
569    out.push_str("BUPHA_GHA_MAXPAR=${BUPHA_GHA_MAX_PARALLEL:-0}\n");
570    out.push_str("declare -A BUPHA_JQ_L=() BUPHA_JQ_D=() BUPHA_JQ_PID=()\n");
571    out.push_str(&format!(
572        "BUPHA_JQ_LEFT={} BUPHA_JQ_RUNNING=0 BUPHA_JQ_PROGRESS=0\nwhile [ \"$BUPHA_JQ_LEFT\" -gt 0 ]; do\n",
573        groups.len()
574    ));
575    for (g, m) in may_overlap.iter().enumerate() {
576        if *m {
577            out.push_str(&render_group_reap(g, groups, group_parallel)?);
578        }
579    }
580    out.push_str("BUPHA_JQ_PROGRESS=0\n");
581    for g in 0..groups.len() {
582        out.push_str(&render_group_launch(
583            jobs,
584            g,
585            groups,
586            group_parallel,
587            gdeps,
588            may_overlap,
589            ctx,
590        )?);
591    }
592    out.push_str(
593        "if [ \"$BUPHA_JQ_LEFT\" -gt 0 ]; then\n  if [ \"$BUPHA_JQ_RUNNING\" -gt 0 ]; then\n    wait -n 2>/dev/null\n  elif [ \"$BUPHA_JQ_PROGRESS\" -eq 0 ]; then\n    echo \"gha: job scheduler deadlock — unmet needs\"\n    exit 1\n  fi\nfi\ndone\nwait 2>/dev/null\n",
594    );
595    Ok(out)
596}

598/// The reap half of one scheduler pass for a background-capable
599/// group: when its `jobdone/<g>` marker exists (or its pid is gone
600/// without one — a crashed subshell), fold every leg's status back
601/// into the parent array + fail accounting, remove the group's
602/// private worktree (non-parallel groups; parallel legs clean
603/// themselves inside their subshell's harvest), drop
604/// `jobsynced/<g>` so dependents become ready.
605fn render_group_reap(
606    g: usize,
607    groups: &[Vec<usize>],
608    group_parallel: &[bool],
609) -> Result<String, String> {
610    let legs = &groups[g];
611    let mut out = String::new();
612    out.push_str(&format!(
613        "if [ -n \"${{BUPHA_JQ_L[{g}]:-}}\" ] && [ -z \"${{BUPHA_JQ_D[{g}]:-}}\" ] && {{ [ -f \"$BUPHA_GHA_TMP/jobdone/{g}\" ] || ! kill -0 \"${{BUPHA_JQ_PID[{g}]:-0}}\" 2>/dev/null; }}; then\n"
614    ));
615    out.push_str(&format!(
616        "BUPHA_JQ_D[{g}]=1\nBUPHA_JQ_RUNNING=$((BUPHA_JQ_RUNNING-1))\nBUPHA_JQ_LEFT=$((BUPHA_JQ_LEFT-1))\n"
617    ));
618    for &i in legs {
619        // Fold the subshell-written jobstatus word back into the
620        // parent's array + fail accounting (missing words settle
621        // as failure — a crashed subshell must not read skipped).
622        out.push_str(&format!("bupha_sync_job {i}\n"));
623        out.push_str(&format!(
624            "case \"${{BUPHA_GHA_STATUS[{i}]:-2}}\" in 0|2|3) ;; *) BUPHA_GHA_FAILS=$((BUPHA_GHA_FAILS+1));; esac\n"
625        ));
626    }
627    if !group_parallel[g] {
628        let key = legs[0];
629        out.push_str(&format!("bupha_jp_clean \"$BUPHA_GHA_TMP/jobs/{key}\"\n"));
630    }
631    // Last: the marker dependents' `bupha_jq_ready` probes — only
632    // now are statuses, jobresult/joboutpick aggregates and report
633    // records all final.
634    out.push_str(&format!("touch \"$BUPHA_GHA_TMP/jobsynced/{g}\"\n"));
635    out.push_str("fi\n");
636    Ok(out)
637}

639/// The launch half of one scheduler pass for a single group,
640/// guarded by `!launched && bupha_jq_ready <dep groups>`. Sets
641/// `BUPHA_JQ_L`/`BUPHA_JQ_PROGRESS` up front; on a
642/// `$BUPHA_GHA_MAX_PARALLEL` cap hit it unsets `BUPHA_JQ_L` so a
643/// later pass retries. Background groups (`may_overlap`) fork a
644/// `set -m` subshell — matrix groups fan legs out inside it with
645/// worktrees pre-created parent-side (serializing git's lock),
646/// plain groups run their legs sequentially in one shared private
647/// worktree under `$BUPHA_GHA_TMP/jobs/<first-leg>`; a failed
648/// worktree add settles every leg as failure parent-side (with
649/// report records + aggregates) rather than let subshells cd into
650/// nothing. Inline groups run their body synchronously and mark
651/// themselves done immediately — safe because a group comparable
652/// with every other provably runs alone.
653fn render_group_launch(
654    jobs: &[Job],
655    g: usize,
656    groups: &[Vec<usize>],
657    group_parallel: &[bool],
658    gdeps: &[Vec<usize>],
659    may_overlap: &[bool],
660    ctx: &ScriptCtx,
661) -> Result<String, String> {
662    let legs = &groups[g];
663    let mut out = String::new();
664    let ready = if gdeps[g].is_empty() {
665        String::new()
666    } else {
667        let args = gdeps[g]
668            .iter()
669            .map(usize::to_string)
670            .collect::<Vec<_>>()
671            .join(" ");
672        format!(" && bupha_jq_ready {args}")
673    };
674    out.push_str(&format!(
675        "if [ -z \"${{BUPHA_JQ_L[{g}]:-}}\" ]{ready}; then\nBUPHA_JQ_L[{g}]=1\nBUPHA_JQ_PROGRESS=1\n"
676    ));
677    if !may_overlap[g] {
678        // Inline — runs alone on the shared worktree.
679        if group_parallel[g] {
680            out.push_str(&render_parallel_group(jobs, legs, ctx, false)?);
681        } else {
682            for &i in legs {
683                out.push_str(&render_seq_job(&jobs[i], i, legs, ctx, &sq(ctx.workdir))?);
684            }
685        }
686        out.push_str(&format!(
687            "touch \"$BUPHA_GHA_TMP/jobsynced/{g}\"\nBUPHA_JQ_D[{g}]=1\nBUPHA_JQ_LEFT=$((BUPHA_JQ_LEFT-1))\nfi\n"
688        ));
689        return Ok(out);
690    }
691    // Background — cap check first; a full cap leaves the group
692    // unlaunched for a later pass.
693    out.push_str(
694        "if [ \"$BUPHA_GHA_MAXPAR\" -le 0 ] || [ \"$BUPHA_JQ_RUNNING\" -lt \"$BUPHA_GHA_MAXPAR\" ]; then\n",
695    );
696    if group_parallel[g] {
697        // Pre-create every leg worktree parent-side — git's
698        // worktree lock can't take concurrent `add`s from the
699        // group subshells. A failed pre-create is retried (and
700        // honestly failed) by the leg's own launch check.
701        for &i in legs {
702            out.push_str(&format!(
703                "git -C \"$BUPHA_WORKDIR\" worktree add --detach --quiet \"$BUPHA_GHA_TMP/legs/{i}\" HEAD 2>/dev/null || true\n"
704            ));
705        }
706        out.push_str("( set -m\n");
707        out.push_str(&render_parallel_group(jobs, legs, ctx, true)?);
708        out.push_str(&format!(
709            "touch \"$BUPHA_GHA_TMP/jobdone/{g}\"\n) &\nBUPHA_JQ_PID[{g}]=$!\nBUPHA_JQ_RUNNING=$((BUPHA_JQ_RUNNING+1))\n"
710        ));
711    } else {
712        let key = legs[0];
713        let wt = format!("\"$BUPHA_GHA_TMP/jobs/{key}\"");
714        out.push_str(&format!(
715            "if git -C \"$BUPHA_WORKDIR\" worktree add --detach --quiet \"$BUPHA_GHA_TMP/jobs/{key}\" HEAD; then\n( set -m\n"
716        ));
717        for &i in legs {
718            out.push_str(&render_seq_job(&jobs[i], i, legs, ctx, &wt)?);
719        }
720        out.push_str(&format!(
721            "touch \"$BUPHA_GHA_TMP/jobdone/{g}\"\n) &\nBUPHA_JQ_PID[{g}]=$!\nBUPHA_JQ_RUNNING=$((BUPHA_JQ_RUNNING+1))\nelse\n"
722        ));
723        // Provisioning failed — settle every leg as a failure
724        // (with report records + aggregates) rather than let the
725        // subshells cd into nothing, then mark the group done
726        // parent-side: no subshell exists to write jobdone.
727        let list = legs
728            .iter()
729            .map(usize::to_string)
730            .collect::<Vec<_>>()
731            .join(" ");
732        let sid = job_key(&jobs[key]);
733        for &i in legs {
734            let j = &jobs[i];
735            out.push_str(&format!(
736                "  bupha_set_status {i} 1\n  echo {}\n  bupha_report {i} {} {}\n",
737                shq_exprs(
738                    &format!(
739                        "gha: job '{}' failed — git worktree add failed",
740                        job_label(j)
741                    ),
742                    i,
743                    &j.workflow
744                )?,
745                sq(&status_suffix(j)),
746                shq_exprs(&j.label, i, &j.workflow)?
747            ));
748        }
749        out.push_str(&format!(
750            "  bupha_agg_result {} {list}\n  bupha_pick_out {} {list}\n  touch \"$BUPHA_GHA_TMP/jobsynced/{g}\"\n  BUPHA_JQ_D[{g}]=1\n  BUPHA_JQ_LEFT=$((BUPHA_JQ_LEFT-1))\nfi\n",
751            sq(&sid),
752            sq(&sid)
753        ));
754    }
755    out.push_str("else\nunset \"BUPHA_JQ_L[");
756    out.push_str(&g.to_string());
757    out.push_str("]\"\nfi\nfi\n");
758    Ok(out)
759}

770/// A job's run gate: `if:` when present (it already carries the
771/// implicit success() when the author didn't name a status
772/// function), else the implicit all-needs-passed check. Returns
773/// the bash condition plus the skip reason for the log line.
774fn job_gate(j: &Job, i: usize) -> Result<Option<(String, &'static str)>, String> {
775    // `gate_needs` (reusable-call publishers) is the subset of
776    // `needs` the job's own gate may read: the call site's needs.
777    // The imported callee jobs are scheduling edges only — their
778    // failures surface through the publisher's check step, not the
779    // gate (a failed callee job must fail the call, not skip it).
780    let gate_needs = j.gate_needs.as_deref().unwrap_or(&j.needs);
781    if let Some(cond) = &j.cond {
782        let cx = CondCx {
783            ji: i,
784            needs: gate_needs,
785            wf: &j.workflow,
786        };
787        Ok(Some((cond_bool(cond, &cx)?, "if condition false")))
788    } else if !gate_needs.is_empty() {
789        let conds = gate_needs
790            .iter()
791            .map(|&d| format!("[ \"${{BUPHA_GHA_STATUS[{d}]:-2}}\" -eq 0 ]"))
792            .collect::<Vec<_>>()
793            .join(" && ");
794        Ok(Some((conds, "needs did not pass")))
795    } else {
796        Ok(None)
797    }
798}

800/// The job-body interior shared by the sequential path and
801/// parallel-matrix legs: workflow export, the per-job `$GITHUB_*`
802/// scratch files (`ws` is the quoted workspace root — the worktree
803/// for sequential jobs, the leg's own `git worktree` for parallel
804/// legs), job env exports, then every step.
805fn job_body(j: &Job, ji: usize, ctx: &ScriptCtx, ws: &str) -> Result<String, String> {
806    let mut out = String::new();
807    out.push_str(&format!("    export GITHUB_WORKFLOW={}\n", sq(&j.workflow)));
808    // $GITHUB_* file commands: point each at a per-job scratch
809    // file. Steps append with `>>`; absorb helpers fold them back
810    // after every step (env → exports, path → PATH, output →
811    // per-step files under stepouts/ for the steps.*/needs.*
812    // expression lookups).
813    out.push_str(&format!(
814        "    BUPHA_JOB_TMP=\"$BUPHA_GHA_TMP/job_{ji}\"\n    mkdir -p \"$BUPHA_JOB_TMP\"\n    export RUNNER_TEMP=\"$BUPHA_JOB_TMP\"\n    export GITHUB_WORKSPACE={ws}\n    export GITHUB_ENV=\"$BUPHA_JOB_TMP/github_env\"\n    export GITHUB_OUTPUT=\"$BUPHA_JOB_TMP/github_output\"\n    export GITHUB_PATH=\"$BUPHA_JOB_TMP/github_path\"\n    export GITHUB_STEP_SUMMARY=\"$BUPHA_JOB_TMP/step_summary.md\"\n    : > \"$GITHUB_ENV\"; : > \"$GITHUB_OUTPUT\"; : > \"$GITHUB_PATH\"; : > \"$GITHUB_STEP_SUMMARY\"\n"
815    ));
816    if !j.env.is_empty() {
817        out.push_str(&format!(
818            "    {}\n",
819            exports(&j.env, ji, &j.workflow)?.replace('\n', "\n    ")
820        ));
821    }
822    // Composite-`uses:` deadlines: arm each group's timestamp file
823    // (epoch = now + total seconds) just before its first tagged
824    // step. Keys are per-job (`deadline_<ji>_<safe key>`), so
825    // matrix legs and parallel jobs can't collide; arming is
826    // unconditional — a fully `if:`-skipped group leaves a harmless
827    // file behind.
828    let mut armed: HashSet<&str> = HashSet::new();
829    for (si, s) in j.steps.iter().enumerate() {
830        for (key, secs) in step_deadlines(s) {
831            if armed.insert(key.as_str()) {
832                out.push_str(&format!(
833                    "    echo $(( $(date +%s) + {secs} )) > \"$BUPHA_GHA_TMP/deadline_{ji}_{}\"\n",
834                    safe_ctx(key)
835                ));
836            }
837        }
838        out.push_str(&render_step(s, ji, si, j, ctx, ws)?);
839    }
840    Ok(out)
841}

847        Step::Run(r) => &r.deadlines,
848        _ => &[],

866/// One parallel matrix group: legs fan out as `set -m` background
867/// process groups, each in a private `git worktree` checkout under
868/// `$BUPHA_GHA_TMP/legs/<ji>` (created parent-side to serialize
869/// git's worktree lock, removed at harvest or by the EXIT-trap
870/// backstop). `max-parallel` throttles launches through a
871/// `wait -n` + harvest loop; `fail-fast` kills surviving leg
872/// groups on the first real failure and settles them cancelled.
873/// Per-leg `if:`/needs gates evaluate parent-side before launch;
874/// leg bodies record their exit code via an EXIT trap into
875/// `legexit/<ji>` so even TERM-killed legs report (→ cancelled
876/// under fail-fast). Statuses settle parent-side at harvest —
877/// the `BUPHA_GHA_STATUS` array can't cross the leg subshells.
878fn render_parallel_group(
879    jobs: &[Job],
880    legs: &[usize],
881    ctx: &ScriptCtx,
882    wt_precreated: bool,
883) -> Result<String, String> {
884    let n = legs.len();
885    let first = &jobs[legs[0]];
886    let maxp = first.max_parallel.unwrap_or(n as u32).max(1);
887    let ff = u8::from(first.fail_fast);
888    let mut out = String::new();
889    let label = job_label(first);
890    out.push_str(&format!(
891        "echo {}\n",
892        shq_exprs(
893            &format!("gha: job '{label}' — {n} matrix legs, max-parallel {maxp}"),
894            legs[0],
895            &first.workflow,
896        )?
897    ));
898    for &i in legs {
899        let j = &jobs[i];
900        let label = job_label(j);
901        out.push_str(&format!(
902            "echo {}\n",
903            shq_exprs(&format!("gha: job '{label}' starting"), i, &j.workflow)?
904        ));
905        let launch = render_leg_launch(j, i, ctx, maxp, ff, &label, wt_precreated)?;
906        match job_gate(j, i)? {
907            Some((g, reason)) => {
908                out.push_str(&format!(
909                    "if {g}; then\n{launch}  else\n    bupha_set_status {i} 2\n    echo {}\n  fi\n",
910                    shq_exprs(
911                        &format!("gha: job '{label}' skipped — {reason}"),
912                        i,
913                        &j.workflow
914                    )?
915                ));
916            }
917            None => out.push_str(&launch),
918        }
919    }
920    // Drain: reap every launched leg, then settle the group's
921    // aggregates and per-leg status records.
922    out.push_str(&format!(
923        "while [ -n \"$BUPHA_MP_RUN\" ]; do\n  wait -n 2>/dev/null\n  bupha_mp_harvest {ff}\ndone\nBUPHA_MP_FAIL=0\n"
924    ));
925    let list = legs
926        .iter()
927        .map(usize::to_string)
928        .collect::<Vec<_>>()
929        .join(" ");
930    let sid = job_key(first);
931    out.push_str(&format!("bupha_agg_result {} {list}\n", sq(&sid)));
932    out.push_str(&format!("bupha_pick_out {} {list}\n", sq(&sid)));
933    for &i in legs {
934        let j = &jobs[i];
935        out.push_str(&format!(
936            "bupha_report {i} {} {}\n",
937            sq(&status_suffix(j)),
938            shq_exprs(&j.label, i, &j.workflow)?
939        ));
940    }
941    Ok(out)
942}

944/// The launch block for one parallel matrix leg (emitted inside
945/// its run gate): throttle to `max-parallel`, honour a pending
946/// fail-fast cancel, `git worktree add` the leg's private
947/// checkout, then background the leg body as its own process
948/// group with an optional `timeout-minutes` watchdog.
949fn render_leg_launch(
950    j: &Job,
951    i: usize,
952    ctx: &ScriptCtx,
953    maxp: u32,
954    ff: u8,
955    label: &str,
956    wt_precreated: bool,
957) -> Result<String, String> {
958    let legws = format!("\"$BUPHA_GHA_TMP/legs/{i}\"");
959    let mut out = String::new();
960    // Throttle: while all slots are busy, block on the next leg
961    // exit and harvest whoever finished (a harvest may also trip
962    // fail-fast, which drops us into the cancel branch below).
963    out.push_str(&format!(
964        "  while [ \"$BUPHA_MP_FAIL\" -eq 0 ] && [ \"$(bupha_mp_slots)\" -ge {maxp} ]; do\n    wait -n 2>/dev/null\n    bupha_mp_harvest {ff}\n  done\n"
965    ));
966    // `wt_precreated`: a concurrent level already ran `git worktree
967    // add` for every leg parent-side (serializing git's lock) — the
968    // launch only checks the dir exists, retrying the add itself
969    // when the pre-create failed.
970    let add = if wt_precreated {
971        format!(
972            "[ -e \"$BUPHA_GHA_TMP/legs/{i}/.git\" ] || git -C \"$BUPHA_WORKDIR\" worktree add --detach --quiet \"$BUPHA_GHA_TMP/legs/{i}\" HEAD"
973        )
974    } else {
975        format!(
976            "git -C \"$BUPHA_WORKDIR\" worktree add --detach --quiet \"$BUPHA_GHA_TMP/legs/{i}\" HEAD"
977        )
978    };
979    out.push_str(&format!(
980        "  if [ \"$BUPHA_MP_FAIL\" -eq 1 ]; then\n    bupha_set_status {i} 3\n    echo {}\n  elif {add}; then\n",
981        shq_exprs(&format!("gha: job '{label}' cancelled — fail-fast"), i, &j.workflow)?
982    ));
983    // The leg body: the EXIT trap records the exit code (even on
984    // TERM), the TERM trap converts kills to 143; the inner
985    // `( set -e … )` runs preamble+steps exactly like a
986    // sequential job body.
987    out.push_str("    (\n      trap 'printf \"%s\" \"$?\" > \"$BUPHA_GHA_TMP/legexit/");
988    out.push_str(&i.to_string());
989    out.push_str("\"' EXIT\n      trap 'exit 143' TERM\n      (\n        set -e\n");
990    out.push_str(&job_body(j, i, ctx, &legws)?);
991    out.push_str("      )\n      bupha_rc=$?\n      if [ \"$bupha_rc\" -eq 0 ]; then\n");
992    // Job `outputs:` publish per leg into jobouts/<ji> — the
993    // joboutpick file later selects the highest-index successful
994    // leg for needs.<job>.outputs.*.
995    for line in job_output_lines(j, i)?.lines() {
996        out.push_str("        ");
997        out.push_str(line.trim_start());
998        out.push('\n');
999    }
1000    out.push_str(&format!(
1001        "        echo {}\n      else\n        echo {}\n      fi\n      exit \"$bupha_rc\"\n    ) &\n    BUPHA_MP_PID[{i}]=$!\n",
1002        shq_exprs(&format!("gha: job '{label}' passed"), i, &j.workflow)?,
1003        shq_exprs(&format!("gha: job '{label}' failed"), i, &j.workflow)?
1004    ));
1005    if let Some(secs) = j.timeout_seconds {
1006        // Same watchdog shape as sequential jobs, keyed per leg:
1007        // the killer TERMs the leg's process group after N secs.
1008        out.push_str(&format!(
1009            "    ( sleep {secs}; touch \"$BUPHA_GHA_TMP/timeout_{i}\"; kill -TERM -- -\"${{BUPHA_MP_PID[{i}]}}\" 2>/dev/null ) &\n    BUPHA_MP_WD[{i}]=$!\n"
1010        ));
1011    }
1012    out.push_str(&format!("    BUPHA_MP_RUN=\"$BUPHA_MP_RUN {i}\"\n"));
1013    out.push_str(&format!(
1014        "  else\n    bupha_set_status {i} 1\n    echo {}\n  fi\n",
1015        shq_exprs(
1016            &format!("gha: job '{label}' failed — git worktree add failed"),
1017            i,
1018            &j.workflow
1019        )?
1020    ));
1021    Ok(out)
1022}

1062fn render_step(
1063    s: &Step,
1064    ji: usize,
1065    si: usize,
1066    job: &Job,
1067    ctx: &ScriptCtx,
1068    ws: &str,
1069) -> Result<String, String> {
1070    let mut out = String::new();
1071    match s {
1072        Step::Checkout { id, cond } => {
1073            let key = step_key(id.as_deref(), si);
1074            let gate = step_gate(cond, ji, job)?;
1075            if let Some(g) = &gate {
1076                out.push_str(&format!("    if {g}; then\n"));
1077            }
1078            out.push_str("    (\n      set -eo pipefail\n");
1079            out.push_str(&format!(
1080                "      cd {}\n",
1081                cd_target(ws, None, ji, &job.workflow)?
1082            ));
1083            if bupha_core::is_valid_sha(ctx.sha) {
1084                out.push_str(&format!(
1085                    "      head_sha=$(git rev-parse HEAD)\n      case \"$head_sha\" in {}*) ;; *) echo \"gha: actions/checkout — HEAD $head_sha != pushed {}\"; exit 1;; esac\n",
1086                    ctx.sha,
1087                    ctx.sha,
1088                ));
1089            }
1090            out.push_str(
1091                "      echo \"gha: actions/checkout — worktree already at pushed sha\"\n    )\n",
1092            );
1093            out.push_str(&step_concl_write(ji, &key, "success"));
1094            if gate.is_some() {
1095                out.push_str(&format!(
1096                    "    else\n{}    echo {}\n    fi\n",
1097                    step_concl_write(ji, &key, "skipped"),
1098                    shq_exprs(
1099                        &format!(
1100                            "gha: step '{}' skipped — if condition false",
1101                            step_label(id.as_deref(), si)
1102                        ),
1103                        ji,
1104                        &job.workflow
1105                    )?
1106                ));
1107            }
1108        }
1109        Step::Run(r) => {
1110            let shell = shell_cmd(r.shell.as_deref().or(job.shell.as_deref()))?;
1111            let wd = r
1112                .working_directory
1113                .as_deref()
1114                .or(job.working_directory.as_deref());
1115            if wd.map(|w| w.starts_with('/')).unwrap_or(false) {
1116                return Err(format!(
1117                    "absolute working-directory `{wd:?}` is unsupported"
1118                ));
1119            }
1120            let key = step_key(r.id.as_deref(), si);
1121            let gate = step_gate(&r.cond, ji, job)?;
1122            if let Some(g) = &gate {
1123                out.push_str(&format!("    if {g}; then\n"));
1124            }
1125            out.push_str("    (\n      set -eo pipefail\n");
1126            out.push_str(&format!(
1127                "      cd {}\n",
1128                cd_target(ws, wd, ji, &job.workflow)?
1129            ));
1130            if !r.env.is_empty() {
1131                out.push_str(&format!(
1132                    "      {}\n",
1133                    exports(&r.env, ji, &job.workflow)?.replace('\n', "\n      ")
1134                ));
1135            }
1136            let delim = heredoc_delim(&r.script, &format!("BUPHA_EOF_{ji}_{si}"));
1137            // `timeout-minutes` wraps the step's shell — GNU
1138            // timeout signals the child's whole process group, so
1139            // grandchildren (cargo, npm, …) die with it. Composite
1140            // deadlines stack on top: the step's shell gets the
1141            // smallest remaining budget across its own timeout and
1142            // every armed deadline file (computed at run time).
1143            let timeout = if r.deadlines.is_empty() {
1144                r.timeout_seconds
1145                    .map(|s| format!("timeout --kill-after=5 {s} "))
1146                    .unwrap_or_default()
1147            } else {
1148                let explicit = r.timeout_seconds.unwrap_or(0);
1149                let files = r
1150                    .deadlines
1151                    .iter()
1152                    .map(|(k, _)| format!("\"$BUPHA_GHA_TMP/deadline_{ji}_{}\"", safe_ctx(k)))
1153                    .collect::<Vec<_>>()
1154                    .join(" ");
1155                format!("timeout --kill-after=5 \"$(bupha_step_timeout {explicit} {files})\" ")
1156            };
1157            out.push_str(&format!("      {timeout}{shell} <<'{delim}'\n"));
1158            out.push_str(&script_body(&r.script, ji, &job.workflow)?);
1159            if !r.script.ends_with('\n') {
1160                out.push('\n');
1161            }
1162            out.push_str(&format!("{delim}\n    )\n"));
1163            // Fold the step's $GITHUB_* file writes back: env
1164            // entries export for later steps, PATH lines prepend,
1165            // outputs land as per-name files under stepouts/<job>/
1166            // <step-id>/ — the store steps.*.outputs.* and job
1167            // outputs: read. A failed step never reaches this —
1168            // `set -e` ends the job first, matching the runner's
1169            // stop-on-failure v1 semantics.
1170            out.push_str("    bupha_absorb_env_file \"$GITHUB_ENV\"\n");
1171            out.push_str("    bupha_absorb_path_file \"$GITHUB_PATH\"\n");
1172            out.push_str(&format!(
1173                "    bupha_absorb_output_file \"$GITHUB_OUTPUT\" {} {}\n",
1174                ji,
1175                sq(&key)
1176            ));
1177            out.push_str(&step_concl_write(ji, &key, "success"));
1178            if gate.is_some() {
1179                out.push_str(&format!(
1180                    "    else\n{}    echo {}\n    fi\n",
1181                    step_concl_write(ji, &key, "skipped"),
1182                    shq_exprs(
1183                        &format!(
1184                            "gha: step '{}' skipped — if condition false",
1185                            step_label(r.id.as_deref(), si)
1186                        ),
1187                        ji,
1188                        &job.workflow
1189                    )?
1190                ));
1191            }
1192        }
1193        Step::Publish { id, cond, outputs } => {
1194            // A composite action's output publisher: each declared
1195            // output's template evaluates now (its `steps.*`/`needs.*`
1196            // refs read the stores the inlined steps just wrote) and
1197            // lands as a file under stepouts/<job>/<uses-id>/ — the
1198            // same store `steps.<id>.outputs.*` reads. Also records
1199            // the uses-step's conclusion under its id.
1200            let key = step_key(Some(id.as_str()), si);
1201            let gate = step_gate(cond, ji, job)?;
1202            if let Some(g) = &gate {
1203                out.push_str(&format!("    if {g}; then\n"));
1204            }
1205            out.push_str(&format!(
1206                "    mkdir -p \"$BUPHA_GHA_TMP/stepouts/{ji}/{key}\"\n"
1207            ));
1208            for (name, tmpl) in outputs {
1209                let word = shq_exprs(tmpl, ji, &job.workflow)?;
1210                out.push_str(&format!(
1211                    "    printf '%s' {word} > \"$BUPHA_GHA_TMP/stepouts/{ji}/{key}/{name}\"\n    printf '%s=%s\\n' {} {word} >> \"$BUPHA_GHA_OUTPUTS\"\n",
1212                    sq(name)
1213                ));
1214            }
1215            out.push_str(&step_concl_write(ji, &key, "success"));
1216            if gate.is_some() {
1217                out.push_str(&format!(
1218                    "    else\n{}    echo {}\n    fi\n",
1219                    step_concl_write(ji, &key, "skipped"),
1220                    shq_exprs(
1221                        &format!("gha: step '{id}' skipped — if condition false"),
1222                        ji,
1223                        &job.workflow
1224                    )?
1225                ));
1226            }
1227        }
1228    }
1229    Ok(out)
1230}

1253/// A deferred `${{ needs.*.outputs.* }}` / `${{ steps.*.outputs.* }}`
1254/// ref → the bash that reads its value: `cat` of the file the
1255/// producer's absorb wrote. `quoted` wraps the substitution so the
1256/// result stays one word (env exports, echo args); unquoted leaves
1257/// word-splitting to the step's own quoting, matching how GitHub
1258/// substitutes the raw value into `run:` scripts.
1259fn output_lookup(inner: &str, ji: usize, quoted: bool, wf: &str) -> Result<String, String> {
1260    let sub = if let Some(rest) = inner.strip_prefix("needs.") {
1261        if let Some((job, name)) = rest.split_once(".outputs.") {
1262            if job.is_empty() || name.is_empty() {
1263                return Err(format!(
1264                    "unsupported deferred expression `${{{{ {inner} }}}}`"
1265                ));
1266            }
1267            // `joboutpick/<key>` holds the expanded index of the
1268            // job's highest-index *successful* leg — the
1269            // deterministic `needs.<job>.outputs.*` rule under
1270            // parallel matrix (legs finish out of order, so
1271            // "last to write" is meaningless). Empty pick → the
1272            // path can't resolve → empty, like an output the
1273            // producer never declared.
1274            let pick = ref_key(wf, job);
1275            format!(
1276                "$(cat \"$BUPHA_GHA_TMP/jobouts/$(cat \"$BUPHA_GHA_TMP/joboutpick/{pick}\" 2>/dev/null)/{name}\" 2>/dev/null || :)"
1277            )
1278        } else if let Some(job) = rest.strip_suffix(".result") {
1279            if job.is_empty() {
1280                return Err(format!(
1281                    "unsupported deferred expression `${{{{ {inner} }}}}`"
1282                ));
1283            }
1284            needs_result_lookup(job, wf)
1285        } else {
1286            return Err(format!(
1287                "unsupported deferred expression `${{{{ {inner} }}}}`"
1288            ));
1289        }
1290    } else if let Some(rest) = inner.strip_prefix("steps.") {
1291        if let Some((sid, name)) = rest.split_once(".outputs.") {
1292            if sid.is_empty() || name.is_empty() {
1293                return Err(format!(
1294                    "unsupported deferred expression `${{{{ {inner} }}}}`"
1295                ));
1296            }
1297            format!("$(cat \"$BUPHA_GHA_TMP/stepouts/{ji}/{sid}/{name}\" 2>/dev/null || :)")
1298        } else if let Some(sid) = rest.strip_suffix(".conclusion") {
1299            if sid.is_empty() {
1300                return Err(format!(
1301                    "unsupported deferred expression `${{{{ {inner} }}}}`"
1302                ));
1303            }
1304            format!("$(cat \"$BUPHA_GHA_TMP/stepconcl/{ji}/{sid}\" 2>/dev/null || :)")
1305        } else {
1306            return Err(format!(
1307                "unsupported deferred expression `${{{{ {inner} }}}}`"
1308            ));
1309        }
1310    } else if let Some(name) = inner.strip_prefix("secrets.") {
1311        // Internal marker emitted by plan for env pass-through
1312        // secrets (`secrets: inherit`, `${{ secrets.X }}` forwards):
1313        // `name` is the runner env var to read — never a literal
1314        // secret name the callee declared. Charset re-checked here
1315        // so a malformed marker can't smuggle shell into `printenv`.
1316        if name.is_empty() || !name.chars().all(|c| c.is_ascii_alphanumeric() || c == '_') {
1317            return Err(format!(
1318                "unsupported deferred expression `${{{{ {inner} }}}}`"
1319            ));
1320        }
1321        format!("$(printenv {} 2>/dev/null || :)", sq(name))
1322    } else {
1323        return Err(format!(
1324            "unsupported deferred expression `${{{{ {inner} }}}}`"
1325        ));
1326    };
1327    Ok(if quoted { format!("\"{sub}\"") } else { sub })
1328}

1343/// Path-safe form of a job id for the `jobresult/`+`joboutpick/`
1344/// stores — ids are author-controlled YAML keys, so anything
1345/// outside `[A-Za-z0-9_-]` folds to `-` (a `..` or `/` must never
1346/// reach a `cat`/`>` path under `$BUPHA_GHA_TMP`).
1347fn safe_ctx(s: &str) -> String {
1348    let out: String = s
1349        .chars()
1350        .map(|c| {
1351            if c.is_ascii_alphanumeric() || c == '-' || c == '_' {
1352                c
1353            } else {
1354                '-'
1355            }
1356        })
1357        .collect();
1358    if out.is_empty() {
1359        "_".to_string()
1360    } else {
1361        out
1362    }
1363}

1385/// Turn an expanded template into a bash word for positions outside
1386/// `run:` scripts (env exports, echo args, cd targets): literal
1387/// stretches single-quoted, deferred output refs as quoted `$(cat …)`
1388/// pieces — adjacent pieces concatenate into one word.
1389fn shq_exprs(tpl: &str, ji: usize, wf: &str) -> Result<String, String> {
1390    let mut out = String::new();
1391    let mut rest = tpl;
1392    while let Some(start) = rest.find("${{") {
1393        out.push_str(&sq(&rest[..start]));
1394        let after = &rest[start + 3..];
1395        let Some(end) = after.find("}}") else {
1396            return Err("unclosed `${{` in rendered template".to_string());
1397        };
1398        out.push_str(&output_lookup(after[..end].trim(), ji, true, wf)?);
1399        rest = &after[end + 2..];
1400    }
1401    out.push_str(&sq(rest));
1402    Ok(out)
1403}

1405/// Rewrite a `run:` script's deferred output refs into `$(cat …)`.
1406/// Literal text passes through verbatim — the quoted heredoc keeps
1407/// it literal for the step's own shell, so the substitution lands
1408/// exactly where GitHub's pre-compiled expression would have put
1409/// the value (inside the user's own quoting).
1410fn script_body(script: &str, ji: usize, wf: &str) -> Result<String, String> {
1411    let mut out = String::with_capacity(script.len());
1412    let mut rest = script;
1413    while let Some(start) = rest.find("${{") {
1414        out.push_str(&rest[..start]);
1415        let after = &rest[start + 3..];
1416        let Some(end) = after.find("}}") else {
1417            return Err("unclosed `${{` in rendered script".to_string());
1418        };
1419        out.push_str(&output_lookup(after[..end].trim(), ji, false, wf)?);
1420        rest = &after[end + 2..];
1421    }
1422    out.push_str(rest);
1423    Ok(out)
1424}

1433/// Bash helpers emitted once per generated program. They implement
1434/// the GitHub env-file contract the runner's `absorb_env_files`
1435/// implements in Rust: `NAME=value` lines, `NAME<<DELIM` heredocs,
1436/// `#` comments; each file is truncated after folding so the next
1437/// step starts clean. Names must be shell identifiers — anything
1438/// else is skipped rather than eval'd.
1439const ABSORB_FNS: &str = r#"
1440bupha_ltrim() { local s="$1"; printf '%s' "${s#"${s%%[![:space:]]*}"}"; }
1441bupha_rtrim() { local s="$1"; printf '%s' "${s%"${s##*[![:space:]]}"}"; }
1442
1443bupha_absorb_env_file() {
1444  local f="$1" line name val delim l
1445  [ -f "$f" ] || return 0
1446  while IFS= read -r line || [ -n "$line" ]; do
1447    case "$line" in ''|'#'*) continue ;; esac
1448    if [[ "$line" == *'<<'* && "$line" != *=* ]]; then
1449      name=$(bupha_rtrim "${line%%'<<'*}")
1450      delim=$(bupha_rtrim "$(bupha_ltrim "${line#*'<<'}")")
1451      delim="${delim//\'/}"
1452      delim="${delim//\"/}"
1453      val=""
1454      while IFS= read -r l; do
1455        if [ "$l" = "$delim" ]; then break; fi
1456        val+="$l"$'\n'
1457      done
1458    elif [[ "$line" == *=* ]]; then
1459      name=$(bupha_rtrim "${line%%=*}")
1460      val="${line#*=}"
1461    else
1462      continue
1463    fi
1464    name=$(bupha_ltrim "$name")
1465    case "$name" in
1466      ''|[0-9]*|*[!A-Za-z0-9_]*) continue ;;
1467    esac
1468    export "$name=$val"
1469  done < "$f"
1470  : > "$f"
1471}

1473bupha_absorb_path_file() {
1474  local f="$1" line
1475  [ -f "$f" ] || return 0
1476  local -a dirs=()
1477  while IFS= read -r line || [ -n "$line" ]; do
1478    line=$(bupha_rtrim "$line")
1479    if [ -n "$line" ]; then dirs+=("$line"); fi
1480  done < "$f"
1481  # GitHub prepends each line in file order, so the LAST line ends
1482  # up front-most; later steps' entries outrank earlier ones
1483  # because this runs after every step.
1484  local i
1485  for ((i = 0; i < ${#dirs[@]}; i++)); do
1486    PATH="${dirs[$i]}:$PATH"
1487  done
1488  : > "$f"
1489}

1491bupha_absorb_output_file() {
1492  # $1 = the step's GITHUB_OUTPUT file, $2 = expanded-job index,
1493  # $3 = step key (its `id:` or an auto name). Each NAME=VALUE /
1494  # NAME<<DELIM entry becomes $BUPHA_GHA_TMP/stepouts/<job>/<step>/<name>
1495  # — the store `steps.<id>.outputs.*` and job `outputs:` read — and
1496  # the raw lines still append to the run's outputs.txt log.
1497  local f="$1" jikey="$2" sid="$3"
1498  [ -f "$f" ] || return 0
1499  if [ -s "$f" ]; then
1500    { printf '# step %s/%s\n' "$jikey" "$sid"; cat "$f"; } >> "$BUPHA_GHA_OUTPUTS"
1501  fi
1502  local dir="$BUPHA_GHA_TMP/stepouts/$jikey/$sid"
1503  local line name val delim l
1504  while IFS= read -r line || [ -n "$line" ]; do
1505    case "$line" in ''|'#'*) continue ;; esac
1506    if [[ "$line" == *'<<'* && "$line" != *=* ]]; then
1507      name=$(bupha_rtrim "${line%%'<<'*}")
1508      delim=$(bupha_rtrim "$(bupha_ltrim "${line#*'<<'}")")
1509      delim="${delim//\'/}"
1510      delim="${delim//\"/}"
1511      val=""
1512      while IFS= read -r l; do
1513        if [ "$l" = "$delim" ]; then break; fi
1514        val+="$l"$'\n'
1515      done
1516    elif [[ "$line" == *=* ]]; then
1517      name=$(bupha_rtrim "${line%%=*}")
1518      val="${line#*=}"
1519    else
1520      continue
1521    fi
1522    name=$(bupha_ltrim "$name")
1523    # Path-component safety: the expression charset never produces
1524    # `/`, but a hand-written GITHUB_OUTPUT could — skip rather than
1525    # write outside stepouts/.
1526    case "$name" in
1527      ''|.|..|*[!A-Za-z0-9_.-]*) continue ;;
1528    esac
1529    mkdir -p "$dir"
1530    printf '%s' "$val" > "$dir/$name"
1531  done < "$f"
1532  : > "$f"
1533}

1535# Composite-`uses:` step timeout: $1 = the step's own
1536# `timeout-minutes` seconds (0 = none); the rest are armed deadline
1537# files (epoch seconds, written by the job body before the group's
1538# first step). Prints the smallest positive budget left — the
1539# step's shell is wrapped in `timeout` with it, so a composite's
1540# `timeout-minutes` spans all its inlined steps like GitHub's
1541# per-step deadline.
1542bupha_step_timeout() {
1543  local best="$1"; shift
1544  local now d left f
1545  now=$(date +%s)
1546  for f in "$@"; do
1547    d=$(cat "$f" 2>/dev/null || echo 0)
1548    left=$(( d - now ))
1549    [ "$left" -lt 1 ] && left=1
1550    if [ "$best" -le 0 ] || [ "$left" -lt "$best" ]; then best="$left"; fi
1551  done
1552  [ "$best" -lt 1 ] && best=1
1553  echo "$best"
1554}

1718# Reap every finished leg of the running matrix group: read its
1719# legexit/<ji> record, kill its timeout watchdog, remove its
1720# worktree, settle jobstatus + FAILS accounting. $1 = the job's
1721# fail-fast flag. On the first real failure with fail-fast set,
1722# TERM every surviving leg's process group, wait them out, and
1723# settle them honestly: a leg that finished before the kill keeps
1724# its verdict, a TERM-killed one becomes cancelled (3), and legs
1725# that failed on their own still count as failures.
1726bupha_mp_harvest() {
1727  local ff="$1" i rc newrun="" failed=0
1728  for i in $BUPHA_MP_RUN; do
1729    if [ -f "$BUPHA_GHA_TMP/legexit/$i" ]; then
1730      rc=$(cat "$BUPHA_GHA_TMP/legexit/$i" 2>/dev/null)
1731      [ -n "$rc" ] || rc=1
1732      wait "${BUPHA_MP_PID[$i]}" 2>/dev/null
1733      if [ -n "${BUPHA_MP_WD[$i]:-}" ]; then
1734        kill "${BUPHA_MP_WD[$i]}" 2>/dev/null
1735        wait "${BUPHA_MP_WD[$i]}" 2>/dev/null
1736      fi
1737      if [ -f "$BUPHA_GHA_TMP/timeout_$i" ]; then
1738        rm -f "$BUPHA_GHA_TMP/timeout_$i"
1739        echo "gha: matrix leg $i timed out"
1740      fi
1741      bupha_mp_clean "$i"
1742      bupha_set_status "$i" "$rc"
1743      if [ "$rc" -ne 0 ]; then
1744        BUPHA_GHA_FAILS=$((BUPHA_GHA_FAILS+1))
1745        failed=1
1746      fi
1747      unset "BUPHA_MP_PID[$i]" "BUPHA_MP_WD[$i]"
1748    else
1749      newrun="$newrun $i"
1750    fi
1751  done
1752  BUPHA_MP_RUN="${newrun# }"
1753  if [ "$failed" -eq 1 ] && [ "$ff" -eq 1 ] && [ -n "$BUPHA_MP_RUN" ]; then
1754    BUPHA_MP_FAIL=1
1755    for i in $BUPHA_MP_RUN; do
1756      [ -n "${BUPHA_MP_PID[$i]:-}" ] && kill -TERM -- -"${BUPHA_MP_PID[$i]}" 2>/dev/null
1757      [ -n "${BUPHA_MP_WD[$i]:-}" ] && kill "${BUPHA_MP_WD[$i]}" 2>/dev/null
1758    done
1759    wait 2>/dev/null
1760    for i in $BUPHA_MP_RUN; do
1761      rc=$(cat "$BUPHA_GHA_TMP/legexit/$i" 2>/dev/null)
1762      bupha_mp_clean "$i"
1763      case "$rc" in
1764        0) bupha_set_status "$i" 0 ;;
1765        143|"") bupha_set_status "$i" 3 ;;
1766        *) bupha_set_status "$i" "$rc"
1767           BUPHA_GHA_FAILS=$((BUPHA_GHA_FAILS+1)) ;;
1768      esac
1769      unset "BUPHA_MP_PID[$i]" "BUPHA_MP_WD[$i]"
1770    done
1771    BUPHA_MP_RUN=""
1772  fi
1773}
