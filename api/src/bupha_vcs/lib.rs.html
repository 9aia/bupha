1//! Vcs port — VCS operations over clones and detached worktrees.
2//! Git impl in `bupha-git`; jj/hg = new impl crates.
3
4use std::path::Path;
5
6/// VCS operations the watcher and release flow need. All methods
7/// return bool success — callers decide whether a failure is
8/// retryable (fetch-miss) or fatal.
9pub trait Vcs: Send + Sync {
10    /// Fetch + prune remotes so `sha` can resolve.
11    fn sync(&self, repo: &Path) -> bool;
12    /// Does the object exist (after a sync attempt)?
13    fn has_commit(&self, repo: &Path, sha: &str) -> bool;
14    /// Detached worktree of `sha` at `dir`.
15    fn worktree_add(&self, repo: &Path, dir: &Path, sha: &str) -> bool;
16    /// Remove a worktree dir (best-effort — returns false on failure).
17    fn worktree_remove(&self, repo: &Path, dir: &Path) -> bool;
18    /// GC stale administrative worktree entries.
19    fn worktree_prune(&self, repo: &Path) -> bool;
20    /// Whether a tag exists locally.
21    fn tag_exists(&self, repo: &Path, tag: &str) -> bool;
22    /// Create a tag (no-op result check — false on failure).
23    fn tag(&self, repo: &Path, tag: &str) -> bool;
24    /// Push a tag to the remote.
25    fn push_tag(&self, repo: &Path, tag: &str) -> bool;
26}
